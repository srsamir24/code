<script>
  const $ = id => document.getElementById(id);
  const card = $('card'), num = $('num'), exp = $('exp'), cvc = $('cvc'), nameIn = $('name'), pay = $('pay');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const EASE = 'cubic-bezier(0.23, 1, 0.32, 1)';
  const params = new URLSearchParams(location.search);
  const still = params.get('still');

  const BRANDS = {
    none: { groups: [4, 4, 4, 4], cvc: 3, fill: ['oklch(0.38 0.015 60)', 'oklch(0.25 0.012 60)'], mark: '' },
    visa: { groups: [4, 4, 4, 4], cvc: 3, fill: ['oklch(0.42 0.13 262)', 'oklch(0.22 0.09 268)'], mark: '<b class="visa">VISA</b>' },
    mc:   { groups: [4, 4, 4, 4], cvc: 3, fill: ['oklch(0.4 0.08 165)', 'oklch(0.23 0.05 172)'], mark: '<span class="mc"><b></b><b></b></span>' },
    amex: { groups: [4, 6, 5], cvc: 4, fill: ['oklch(0.55 0.09 225)', 'oklch(0.36 0.08 235)'], mark: '<span class="amex">AMEX</span>' },
  };
  const fillOf = b => `linear-gradient(140deg, ${BRANDS[b].fill[0]}, ${BRANDS[b].fill[1]})`;

  function detect(d) {
    if (/^3[47]/.test(d)) return 'amex';
    if (/^4/.test(d)) return 'visa';
    if (/^5[1-5]/.test(d)) return 'mc';
    const n = +d.slice(0, 4);
    if (d.length >= 4 && n >= 2221 && n <= 2720) return 'mc';
    return 'none';
  }
  function luhn(d) {
    let sum = 0;
    for (let i = 0; i < d.length; i++) {
      let x = +d[d.length - 1 - i];
      if (i % 2) { x *= 2; if (x > 9) x -= 9; }
      sum += x;
    }
    return sum % 10 === 0;
  }

  /* ---------- rolling digits ---------- */
  function slots(el, groups) {
    el.innerHTML = groups.map(n => `<span class="g">${'<span class="s"><i>•</i></span>'.repeat(n)}</span>`).join('');
    return [...el.querySelectorAll('.s')];
  }
  function roll(slot, ch, animate) {
    const old = slot.lastElementChild;
    if (old.textContent === ch) return;
    const up = ch !== '•'; // digits roll up into place, deletions drop back down
    const neu = document.createElement('i'); neu.textContent = ch; slot.append(neu);
    if (!animate) return old.remove();
    const off = up ? 70 : -70;
    if (reduce) {
      neu.animate([{ opacity: 0 }, { opacity: 1 }], { duration: 160 });
      return old.remove();
    }
    old.animate([{ transform: 'none', opacity: 1 }, { transform: `translateY(${-off}%)`, opacity: 0, filter: 'blur(2px)' }], { duration: 200, easing: EASE }).onfinish = () => old.remove();
    neu.animate([{ transform: `translateY(${off}%)`, opacity: 0, filter: 'blur(2px)' }, { transform: 'none', opacity: 1, filter: 'blur(0)' }], { duration: 300, easing: EASE });
  }
  const paint = (list, str, animate = true) => list.forEach((s, i) => roll(s, str[i] ?? '•', animate));

  /* ---------- brand: colour field wipes in from the logo corner ---------- */
  let brand = null, numSlots = [], cvcSlots = [], cidSlots = [];
  function setBrand(b, animate = true) {
    if (b === brand) return;
    const prev = brand; brand = b;
    const B = BRANDS[b];
    if (!prev || BRANDS[prev].groups.join() !== B.groups.join()) numSlots = slots($('cNum'), B.groups);
    if (!prev || BRANDS[prev].cvc !== B.cvc) { cvcSlots = slots($('cCvc'), [B.cvc]); cidSlots = slots($('cid'), [B.cvc]); }
    card.classList.toggle('is-amex', b === 'amex');
    cvc.maxLength = B.cvc; cvc.placeholder = B.cvc === 4 ? '1234' : '123';
    $('cvcHint').textContent = b === 'amex' ? '4 on the front' : '3 on the back';
    card.querySelectorAll('.mark').forEach(m => {
      m.innerHTML = B.mark;
      if (animate && m.firstChild) m.firstChild.animate([{ opacity: 0, transform: 'scale(0.9)', filter: 'blur(2px)' }, { opacity: 1, transform: 'none', filter: 'blur(0)' }], { duration: 320, delay: 120, easing: EASE, fill: 'backwards' });
    });
    card.querySelectorAll('.face').forEach(face => {
      const base = face.querySelector('.base'), wipe = face.querySelector('.wipe');
      wipe.getAnimations().forEach(a => a.finish());
      if (!animate || !prev) { base.style.setProperty('--fill', fillOf(b)); return; }
      wipe.style.setProperty('--fill', fillOf(b));
      const at = getComputedStyle(face);
      const pos = `at ${at.getPropertyValue('--wx')} ${at.getPropertyValue('--wy')}`;
      const frames = reduce ? [{ opacity: 0, clipPath: `circle(150% ${pos})` }, { opacity: 1, clipPath: `circle(150% ${pos})` }]
                            : [{ clipPath: `circle(0% ${pos})` }, { clipPath: `circle(150% ${pos})` }];
      wipe.animate(frames, { duration: reduce ? 240 : 620, easing: 'cubic-bezier(0.77, 0, 0.175, 1)' }).onfinish = () => base.style.setProperty('--fill', fillOf(b));
    });
    onCvc();   // trims a 4-digit code to 3 and repaints the new slots
    if (animate) sync();
  }

  /* ---------- the 3D card: one spring per axis ---------- */
  const S = { rx: { x: 0, v: 0, t: 0 }, ry: { x: 0, v: 0, t: 0 } };
  let tilt = { x: 0, y: 0 }, flipped = false, raf = 0, last = 0;
  const shA = $('shA'), shC = $('shC');
  function render() {
    const rx = S.rx.x, ry = S.ry.x, r = ry * Math.PI / 180;
    const lift = 34 * Math.abs(Math.sin(r));          // the card lifts as it turns
    const cos = Math.abs(Math.cos(r));
    const side = Math.sin((ry - Math.round(ry / 180) * 180) * Math.PI / 180);
    card.style.transform = `translateZ(${lift}px) rotateX(${rx}deg) rotateY(${ry}deg)`;
    card.style.setProperty('--gx', `${50 + side * 60}%`);
    card.style.setProperty('--fx', `${side * 160 + rx * 4}px`);   // the foil bands slide as the card turns
    card.style.setProperty('--gy', `${50 - Math.sin(rx * Math.PI / 180) * 190}%`);
    card.style.setProperty('--shade', ((1 - cos) * 0.55).toFixed(3));
    card.style.setProperty('--ha', `${side * 220 + rx * 6}deg`);
    // shadow falls away from the light and away from the tilt, and spreads as the card lifts
    const sx = -side * 34, sy = 46 - rx * 1.1 + lift * 0.6;
    shA.style.transform = `translate(${sx}px, ${sy}px) scale(${0.86 * Math.max(cos, 0.18)}, ${0.8 - lift * 0.003})`;
    shA.style.opacity = (0.5 - lift * 0.004).toFixed(3);
    shC.style.transform = `translate(${sx * 0.3}px, ${12 - rx * 0.3 + lift * 0.25}px) scale(${0.94 * Math.max(cos, 0.12)}, 0.92)`;
    shC.style.opacity = (0.38 - lift * 0.008).toFixed(3);
  }
  // slide:js
  function step(now) {
    const dt = Math.min(0.032, Math.max(0, now - last) / 1e3);
    last = now;
    let busy = false;
    for (const a of Object.values(S)) {    // rx and ry
      const f = -180 * (a.x - a.t) - 19 * a.v;  // k, damping
      a.v += f * dt;
      a.x += a.v * dt;
      const live = Math.abs(a.x - a.t) + Math.abs(a.v) > .01;
      if (live) busy = true; else { a.x = a.t; a.v = 0; }
    }
    render();
    raf = busy ? requestAnimationFrame(step) : 0;
  }
  // /slide
  function sync() {
    S.rx.t = tilt.x;
    S.ry.t = (flipped ? 180 : 0) + tilt.y;  // flip = +180
    if (still) return;
    // reduced motion: no spring, straight to the target
    if (reduce) { S.rx.x = S.rx.t = 0; S.ry.x = S.ry.t = flipped ? 180 : 0; return render(); }
    if (!raf) raf = requestAnimationFrame(step);
  }
  function flip(on) { if (flipped !== on) { flipped = on; card.classList.toggle('cvc-on', on); sync(); } }

  // tilt toward the pointer, wherever it is on the page
  const scene = $('scene');
  addEventListener('pointermove', e => {
    if (still || reduce || e.pointerType === 'touch' && !e.buttons) return;
    demoTilt = false;
    const r = scene.getBoundingClientRect();
    const nx = Math.max(-1, Math.min(1, (e.clientX - r.left - r.width / 2) / (r.width / 2 + 140)));
    const ny = Math.max(-1, Math.min(1, (e.clientY - r.top - r.height / 2) / (r.height / 2 + 200)));
    tilt = { x: -ny * 14, y: nx * 18 }; sync();
  });
  document.documentElement.addEventListener('pointerleave', () => { tilt = { x: 0, y: 0 }; sync(); });

  /* ---------- inputs ---------- */
  const group = (d, groups) => { let out = [], i = 0; for (const n of groups) { if (i >= d.length) break; out.push(d.slice(i, i += n)); } return out.join(' '); };
  function err(id, input, msg) { $(id).textContent = msg; input.setAttribute('aria-invalid', msg ? 'true' : 'false'); }

  function onNumber(advance) {
    const caret = num.value.slice(0, num.selectionStart).replace(/\D/g, '').length;
    let d = num.value.replace(/\D/g, '');
    setBrand(detect(d));
    const max = BRANDS[brand].groups.reduce((a, b) => a + b);
    d = d.slice(0, max);
    num.value = group(d, BRANDS[brand].groups);
    if (document.activeElement === num) {          // keep the caret where the user typed
      let p = 0, seen = 0;
      while (seen < caret && p < num.value.length) { if (/\d/.test(num.value[p])) seen++; p++; }
      num.setSelectionRange(p, p);
    }
    paint(numSlots, d);
    err('numErr', num, d.length === max && !luhn(d) ? 'Looks like a typo' : '');
    if (advance && d.length === max && luhn(d)) exp.focus();
  }
  function checkExpiry(final) {
    const d = exp.value.replace(/\D/g, '');
    const m = +d.slice(0, 2), y = 2000 + +d.slice(2);
    const now = new Date();
    if (d.length >= 2 && (m < 1 || m > 12)) return 'Month 01–12';
    if (d.length === 4 && (y < now.getFullYear() || y === now.getFullYear() && m < now.getMonth() + 1)) return 'Expired';
    if (final && d.length < 2) return 'Add the month';
    if (final && d.length < 4) return 'Add the year';
    return '';
  }
  function onExpiry(advance) {
    let d = exp.value.replace(/\D/g, '').slice(0, 4);
    if (d.length === 1 && d > '1') d = '0' + d;     // "4" can only mean April
    exp.value = d.length > 2 ? d.slice(0, 2) + '/' + d.slice(2) : d;
    const shown = (d + 'MMYY'.slice(d.length)).replace(/^(..)/, '$1/');
    $('cExp').textContent = shown; $('cExp').classList.toggle('empty', !d);
    const msg = checkExpiry(false);
    err('expErr', exp, msg);
    if (advance && d.length === 4 && !msg) cvc.focus();
  }
  function onCvc() {
    const d = cvc.value.replace(/\D/g, '').slice(0, BRANDS[brand].cvc);
    cvc.value = d;
    paint(cvcSlots, d); paint(cidSlots, d);
    if (d.length === BRANDS[brand].cvc) err('cvcErr', cvc, '');
  }
  function onName() {
    const v = nameIn.value.trim();
    $('cName').textContent = v || 'Your name'; $('cName').classList.toggle('empty', !v);
    if (v) err('nameErr', nameIn, '');
  }
  // after Paid, the next edit makes the button payable again
  $('form').addEventListener('input', () => { if (pay.dataset.state === 'done') pay.dataset.state = 'idle'; });
  num.addEventListener('input', e => onNumber(e.inputType !== 'deleteContentBackward'));
  exp.addEventListener('input', e => onExpiry(e.inputType !== 'deleteContentBackward'));
  cvc.addEventListener('input', onCvc);
  nameIn.addEventListener('input', onName);
  num.addEventListener('blur', () => { const d = num.value.replace(/\D/g, ''); if (d && d.length < BRANDS[brand].groups.reduce((a, b) => a + b)) err('numErr', num, 'A few digits short'); });
  exp.addEventListener('blur', () => exp.value && err('expErr', exp, checkExpiry(true)));
  // the code lives on the back, so the card turns over; Amex keeps it on the front
  cvc.addEventListener('focus', () => {
    if (brand === 'amex') card.classList.add('cid-on');
    else flip(true);
  });
  cvc.addEventListener('blur', () => { card.classList.remove('cid-on'); flip(false); });

  $('form').addEventListener('submit', e => {
    e.preventDefault();
    if (pay.dataset.state !== 'idle') return;   // Enter while loading does nothing
    const d = num.value.replace(/\D/g, ''), max = BRANDS[brand].groups.reduce((a, b) => a + b);
    err('numErr', num, d.length < max ? (d ? 'A few digits short' : 'Add your card number') : luhn(d) ? '' : 'Looks like a typo');
    err('expErr', exp, checkExpiry(true));
    const n = BRANDS[brand].cvc;
    err('cvcErr', cvc, cvc.value.length < n ? `Needs ${n} digits` : '');
    err('nameErr', nameIn, nameIn.value.trim() ? '' : 'Add the name on the card');
    const bad = [num, exp, cvc, nameIn].find(i => i.getAttribute('aria-invalid') === 'true');
    if (bad) return bad.focus();
    pay.dataset.state = 'loading'; $('status').textContent = 'Paying';
    setTimeout(() => { pay.dataset.state = 'done'; $('status').textContent = 'Paid $86.00'; }, 1500);
  });

  /* ---------- start, stills and demo ---------- */
  setBrand('none', false);
  render();
  function fill(n, x, c, who) {
    num.value = n; onNumber(false);
    exp.value = x; onExpiry(false);
    nameIn.value = who; onName();
    if (c) { cvc.value = c; onCvc(); }
  }
  if (still === 'front') {
    fill('4539 5260 1815 9083', '09/29', '', 'Maren Solberg');
    S.rx.x = 15; S.ry.x = -25; render();
  }
  if (still === 'back') {

  }

  if (still === 'amex') {
    fill('3714 544992 07074', '06/30', '7284', 'Ines Haugland');
    cvc.focus();
    S.rx.x = 10; S.ry.x = 18; render();
  }

  let demoTilt = params.has('demo');
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    const type = async (el, text, fn, gap = 70) => { el.focus(); for (const ch of text) { el.value += ch; fn(); await wait(gap); } };
    (function wobble(t) { if (demoTilt) { tilt = { x: Math.sin(t / 1300) * 8, y: Math.sin(t / 900) * 15 }; sync(); } requestAnimationFrame(wobble); })(0);
    (async function loop() {
      fill('', '', '', ''); cvc.value = ''; onCvc(); pay.dataset.state = 'idle'; setBrand('none');
      await wait(900);
      await type(num, '5412849340722470', () => onNumber(false), 85); exp.focus();
      await type(exp, '1128', () => onExpiry(false), 110);
      await wait(200); cvc.focus(); await wait(700);
      await type(cvc, '417', onCvc, 160); await wait(700);
      await type(nameIn, 'Jonas Eide', onName, 60); await wait(500);
      nameIn.blur(); pay.click(); await wait(2800);
      pay.dataset.state = 'idle'; num.focus(); num.value = ''; onNumber(false); await wait(400);
      await type(num, '371454499207074', () => onNumber(false), 85); await wait(600);
      cvc.focus(); await wait(1600); cvc.blur();
      await wait(1200); loop();
    })();
  }
