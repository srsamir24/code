<script>/* inlined from ../_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};
</script>

  ];
  const svg = r => `<svg viewBox="0 0 200 200" aria-hidden="true">${r.art}</svg>`;
  const TRACKS = ['Opening', 'Long Light', 'Undertow', 'The Quiet Part', 'Harbour', 'Late Train'];

  $('tiles').innerHTML = RECORDS.map((r, i) =>
    `<button class="tile" data-i="${i}" aria-label="${r.t} by ${r.a}"><div class="sleeve">${svg(r)}</div><div class="t">${r.t}</div><div class="a">${r.a}</div></button>`).join('');
  const tiles = [...document.querySelectorAll('.tile')];

  // ---------- state, all on the fixed-step clock ----------
  const DUR = 0.8;                      // the morph (group animation) length, s
  let open = -1, vt = null, vtT = 0, vtOn = false, vinylX = 0, vinylV = 0, vinylTo = 0, spin = 0, pendingClose = false, closeT = 0;

  function setDetail(i) {
    const r = RECORDS[i];
    $('heroSleeve').innerHTML = svg(r);
    $('heroTitle').textContent = r.t; $('heroArtist').textContent = `${r.a} · ${r.y}`;
    $('price').textContent = '€' + r.p; $('meta1').textContent = 'LP · ' + r.y; $('meta2').textContent = r.n + ' tracks';
    $('crumb').textContent = `crate 03 / 0${i + 1}`;
    $('label').style.setProperty('--lab', r.lab);
    $('tracks').innerHTML = [0, 1].map(k => `<div><span><b>A${k + 1}</b>${TRACKS[(i + k * 3) % 6]}</span><span>${3 + (i + k) % 3}:${String(12 + i * 7 + k * 19).slice(-2)}</span></div>`).join('');
  }
  const names = (i, on) => {             // only the one being opened carries the shared names
    const el = tiles[i];
    el.querySelector('.sleeve').style.viewTransitionName = on ? 'sleeve' : '';
    el.querySelector('.t').style.viewTransitionName = on ? 'title' : '';
  };
  $('heroSleeve').style.viewTransitionName = 'sleeve';
  $('heroTitle').style.viewTransitionName = 'title';

  function transition(update) {
    const host = document;
    if (!host.startViewTransition || reduce.matches) { update(); return; }
    vtT = 0; vtOn = true;
    vt = host.startViewTransition(update);
  }
  function openRecord(i) {
    if (open >= 0 || vtOn) return;
    open = i; setDetail(i); names(i, true); document.documentElement.dataset.dir = 'open';
    transition(() => { names(i, false); $('grid').hidden = true; $('detail').hidden = false; $('live').textContent = RECORDS[i].t; });
    $('back').focus({ preventScroll: true });
  }
  function closeRecord() {
    if (open < 0 || vtOn || pendingClose) return;
    pendingClose = true; closeT = 0; vinylTo = 0;           // the record slides home first, then the sleeve flies back
  }
  function doClose() {
    const i = open; pendingClose = false; document.documentElement.dataset.dir = 'close';
    $('vinyl').style.opacity = 0;
    transition(() => { $('detail').hidden = true; $('grid').hidden = false; names(i, true); });
    open = -1;
    tiles[i].focus({ preventScroll: true });
  }
  tiles.forEach((el, i) => el.addEventListener('click', () => openRecord(i)));
  $('back').addEventListener('click', closeRecord);
  addEventListener('keydown', e => { if (e.key === 'Escape') closeRecord(); });

  const vtAnims = () => document.getAnimations().filter(a => (a.effect?.pseudoElement || '').startsWith('::view-transition'));
  function step(dt) {
    if (vtOn) {
      vtT += dt;
      if (vtT >= DUR + 0.02) {
        vtOn = false;
        if (CAPTURE) vtAnims().forEach(a => a.finish());
        if (open >= 0) { $('vinyl').style.opacity = 1; vinylTo = 1; }
        else tiles.forEach((_, i) => names(i, false));
      }
    }
    if (pendingClose) { closeT += dt; if (closeT >= 0.17) doClose(); }
    // the record: a stiff spring out of the sleeve, a slower one back in; it turns while it's out
    const k = vinylTo ? 260 : 900, c = vinylTo ? 20 : 52;
    vinylV += (-(vinylX - vinylTo) * k - vinylV * c) * dt; vinylX += vinylV * dt;
    spin += dt * 3.49 * Math.min(1, Math.max(0, vinylX));
  }
  function render() {
    if (CAPTURE && vtOn) {                                   // the recorder owns the clock: pin every pseudo animation to sim time
      const ms = vtT * 1000;
      for (const a of vtAnims()) { a.pause(); const tm = a.effect.getComputedTiming(); a.currentTime = Math.min(ms, tm.endTime - 0.01); }
    }
    $('vinyl').style.transform = `translateX(${(vinylX * 168).toFixed(2)}px)`;
    $('label').style.transform = $('sheen').style.transform = '';
    $('label').style.transform = `rotate(${spin.toFixed(4)}rad)`;
  }

  Reel({
    step, render,
    geom: () => {
      const r = card.getBoundingClientRect(), k = r.width / 800;
      const tc = i => { const b = tiles[i].querySelector('.sleeve').getBoundingClientRect(); return [(b.left + b.width / 2 - r.left) / k, (b.top + b.height / 2 - r.top) / k]; };
      return { left: r.left, top: r.top, k, tiles: RECORDS.map((_, i) => tc(i)), back: [80, 68] };
    },
  });
  window.__vtSettle = async () => { if (vt && vtOn) { try { await vt.ready; } catch (e) {} await new Promise(r => requestAnimationFrame(() => r())); render(); } return 1; };
  window.__state = () => ({ open, vtOn, vtT, vinylX, n: vtAnims().length });
})();
