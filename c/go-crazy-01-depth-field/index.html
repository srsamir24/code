<script>/* inlined from ../../micro-interactions/_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};

import * as THREE from 'three';

const Z = window.ZELLAP;
const CAPTURE = new URLSearchParams(location.search).has('capture');

const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
const canvas = document.getElementById('field');

// ---------- tuning ----------
const COUNT = 9000;
const FOV = 50, TAN = Math.tan(THREE.MathUtils.degToRad(FOV / 2));
const NEAR_D = 6, FAR_D = 110, FOCUS = 30;          // view-depth range of the field and the sharp plane
const RS = 0.36, UNIT = 30 * TAN;                   // the "hand": a cone around the cursor ray, RS of the half-screen at every depth
const PUSH = 150, DRAG = 13, K = 14, C = 4.2;         // push, wake drag, spring, damping (underdamped: a little wobble home)

const renderer = new THREE.WebGLRenderer({ canvas, antialias: false, alpha: false, preserveDrawingBuffer: CAPTURE });
renderer.setClearColor(0x000000, 1);
renderer.setPixelRatio(CAPTURE ? 1 : Math.min(devicePixelRatio, 2));
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(FOV, 1, 0.1, 400);
camera.position.set(0, 0, 30);

// ---------- the dots: sampled inside the camera's frustum so the screen is evenly covered at every depth ----------
let seed = 7;
const rnd = () => { seed |= 0; seed = seed + 0x6D2B79F5 | 0; let t = Math.imul(seed ^ seed >>> 15, 1 | seed); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
const base = new Float32Array(COUNT * 3), off = new Float32Array(COUNT * 3), vel = new Float32Array(COUNT * 3);
const pos = new Float32Array(COUNT * 3), energy = new Float32Array(COUNT), size = new Float32Array(COUNT), col = new Float32Array(COUNT * 3);
const pal = [Z.launchHi, Z.launch, Z.nexus, Z.growth, Z.growth, Z.scale].map(c => new THREE.Color(c));
const pale = new THREE.Color(Z.growthPale);
function palette(u, c) {                            // a slow sweep across the brand hues: teal -> indigo -> purple -> magenta
  const x = Math.min(0.9999, Math.max(0, u)) * (pal.length - 1), i = Math.floor(x);
  return c.copy(pal[i]).lerp(pal[i + 1], x - i);
}
function seedField(aspect) {
  seed = 7;
  const c = new THREE.Color();
  for (let i = 0; i < COUNT; i++) {
    const d = NEAR_D + (FAR_D - NEAR_D) * Math.pow(rnd(), 0.62);    // more dots far away, a few close to the lens
    const u = (rnd() * 2 - 1) * 1.25, v = (rnd() * 2 - 1) * 1.25;
    base[i * 3] = u * d * TAN * aspect; base[i * 3 + 1] = v * d * TAN; base[i * 3 + 2] = 30 - d;
    const band = 0.5 + 0.38 * Math.sin(u * 1.4 + v * 0.9 + 0.6) + (rnd() - 0.5) * 0.35;
    palette(band, c).lerp(pale, rnd() < 0.42 ? 0.55 + rnd() * 0.4 : rnd() * 0.25);   // about half the dots are pale lilac
    col.set([c.r, c.g, c.b], i * 3);
    size[i] = 0.55 + Math.pow(rnd(), 3) * 1.9;
  }
  off.fill(0); vel.fill(0);
}

const geo = new THREE.BufferGeometry();
const aPos = new THREE.BufferAttribute(pos, 3).setUsage(THREE.DynamicDrawUsage);
const aEnergy = new THREE.BufferAttribute(energy, 1).setUsage(THREE.DynamicDrawUsage);
geo.setAttribute('position', aPos);
geo.setAttribute('aEnergy', aEnergy);
geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
geo.setAttribute('aColor', new THREE.BufferAttribute(col, 3));
const mat = new THREE.ShaderMaterial({
  transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
  uniforms: { uScale: { value: 1 }, uWake: { value: new THREE.Color(Z.launchHi) } },
  vertexShader: /* glsl */`
    attribute float aSize; attribute float aEnergy; attribute vec3 aColor;
    uniform float uScale; uniform vec3 uWake;
    varying vec3 vColor; varying float vAlpha; varying float vSoft;
    void main() {
      vec4 mv = modelViewMatrix * vec4(position, 1.0);
      float d = -mv.z;
      float blur = clamp(abs(d - ${FOCUS.toFixed(1)}) / 55.0, 0.0, 1.0);      // depth of field: soft away from the focus plane
      float fog = pow(clamp(1.0 - (d - 12.0) / ${(FAR_D - 6).toFixed(1)}, 0.0, 1.0), 1.5) * smoothstep(${NEAR_D.toFixed(1)}, 13.0, d);
      float e = clamp(aEnergy, 0.0, 1.0);
      vColor = mix(aColor, uWake, e * 0.85);
      vAlpha = fog * (0.85 + 1.6 * e) / (1.0 + blur * 1.6);
      vSoft = 0.3 + blur * 0.6;
      gl_PointSize = uScale * aSize * (1.0 + blur * 1.2 + e * 0.9) * 95.0 / d;
      gl_Position = projectionMatrix * mv;
    }`,
  fragmentShader: /* glsl */`
    varying vec3 vColor; varying float vAlpha; varying float vSoft;
    void main() {
      float r = length(gl_PointCoord - 0.5) * 2.0;
      float a = 1.0 - smoothstep(1.0 - vSoft, 1.0, r);
      if (a <= 0.0) discard;
      gl_FragColor = vec4(vColor * vAlpha * a, 1.0);   // additive onto black
    }`,
});

// ---------- pointer: target from real pointer events, smoothed "hand" + slower parallax ----------
const ptr = { x: 0, y: 0, in: false };
const hand = { x: 0, y: 0, vx: 0, vy: 0, on: 0 }, par = { x: 0, y: 0 };
const toNdc = e => { const r = canvas.getBoundingClientRect(); ptr.x = (e.clientX - r.left) / r.width * 2 - 1; ptr.y = -((e.clientY - r.top) / r.height * 2 - 1); };

// ---------- simulation (fixed steps, so it plays the same at any frame rate) ----------
let simT = 0;
const camPos = new THREE.Vector3(), dir = new THREE.Vector3(), right = new THREE.Vector3(), up = new THREE.Vector3(), rot = new THREE.Matrix3();
function placeCamera() {
  const ph = (simT / LAP) * Math.PI * 2, still = reduce ? 0 : 1;
  camera.position.set(still * (Math.sin(ph) * 2.4 + par.x * 2.6), still * (Math.sin(ph * 2) * 0.9 + par.y * 1.8), 30 + still * Math.cos(ph) * 1.6);
  camera.lookAt(still * Math.sin(ph) * 0.8, 0, -30);
  camera.updateMatrixWorld();
}
let frozen = false;                                 // capture warm-up: settle the physics without moving the clock
function step(dt) {
  if (!frozen) simT += dt;
  const kh = 1 - Math.exp(-dt * 26), kp = 1 - Math.exp(-dt * 2.2);
  const px = hand.x, py = hand.y;
  hand.x += (ptr.x - hand.x) * kh; hand.y += (ptr.y - hand.y) * kh;
  hand.vx = (hand.x - px) / dt; hand.vy = (hand.y - py) / dt;
  hand.on += ((ptr.in && !reduce ? 1 : 0) - hand.on) * (1 - Math.exp(-dt * 8));
  par.x += (ptr.x * (ptr.in ? 1 : 0) - par.x) * kp; par.y += (ptr.y * (ptr.in ? 1 : 0) - par.y) * kp;
  placeCamera();
  camPos.copy(camera.position);
  right.setFromMatrixColumn(camera.matrixWorld, 0); up.setFromMatrixColumn(camera.matrixWorld, 1);
  dir.set(hand.x * TAN * camera.aspect, hand.y * TAN, -1).applyMatrix3(rot.setFromMatrix4(camera.matrixWorld)).normalize();
  const on = hand.on;
  for (let i = 0, j = 0; i < COUNT; i++, j += 3) {
    let ox = off[j], oy = off[j + 1], oz = off[j + 2], vx = vel[j], vy = vel[j + 1], vz = vel[j + 2];
    if (on > 0.001) {
      const wx = base[j] + ox - camPos.x, wy = base[j + 1] + oy - camPos.y, wz = base[j + 2] + oz - camPos.z;
      const t = wx * dir.x + wy * dir.y + wz * dir.z;
      const cx = wx - dir.x * t, cy = wy - dir.y * t, cz = wz - dir.z * t, r2 = cx * cx + cy * cy + cz * cz;
      const R = RS * t * TAN, sc = t * TAN / UNIT;                          // far dots get pushed further in world units, same on screen
      if (t > 0 && r2 < R * R) {
        const r = Math.sqrt(r2) + 1e-4, f = (1 - r / R) ** 2 * on;
        const push = PUSH * sc * f / r, drag = DRAG * f * t * TAN;          // the wake: the hand's screen speed at this depth
        const sx = hand.vx * camera.aspect, sy = hand.vy;
        vx += (cx * push + (right.x * sx + up.x * sy) * drag) * dt;
        vy += (cy * push + (right.y * sx + up.y * sy) * drag) * dt;
        vz += (cz * push + (right.z * sx + up.z * sy) * drag) * dt;
      }
    }
    vx += (-K * ox - C * vx) * dt; vy += (-K * oy - C * vy) * dt; vz += (-K * oz - C * vz) * dt;
    off[j] = ox + vx * dt; off[j + 1] = oy + vy * dt; off[j + 2] = oz + vz * dt;
    vel[j] = vx; vel[j + 1] = vy; vel[j + 2] = vz;
  }
}
function render() {
  for (let i = 0, j = 0; i < COUNT; i++, j += 3) {
    pos[j] = base[j] + off[j]; pos[j + 1] = base[j + 1] + off[j + 1]; pos[j + 2] = base[j + 2] + off[j + 2];
    energy[i] = Math.min(1, Math.hypot(off[j], off[j + 1], off[j + 2]) * 0.8 * UNIT / ((30 - base[j + 2]) * TAN));
  }
  aPos.needsUpdate = aEnergy.needsUpdate = true;
  renderer.render(scene, camera);
}
function resize() {
  const w = CAPTURE ? 1080 : innerWidth, h = CAPTURE ? 1920 : innerHeight;
  renderer.setSize(w, h, false);
  camera.aspect = w / h; camera.updateProjectionMatrix();
  mat.uniforms.uScale.value = h / 1000 * renderer.getPixelRatio();
  seedField(camera.aspect); placeCamera();
}
resize();
if (!CAPTURE) addEventListener('resize', () => { resize(); render(); });
