<script>
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const $ = id => document.getElementById(id);
  const root = document.documentElement, room = $('room'), lamp = $('lamp'), pool = $('pool'), bead = $('bead'), live = $('live');
  const say = t => { live.textContent = ''; setTimeout(() => live.textContent = t, 30); };

  let scale = 1;
  function resize() {
    scale = Math.min(1, innerWidth / 520, innerHeight / 730);
    room.style.transform = `scale(${scale})`;
    room.style.margin = `0 ${(520 * scale - 520) / 2}px ${(730 * scale - 730)}px`;
  }
  addEventListener('resize', resize); resize();

  /* the rope: a verlet chain. The top end hangs off the lamp, the bottom end is the bead. */
  const N = 18, SEG = 11.5;                       // rest length 195px
  const ANCHOR = { x: 318, y: 176 };              // the eyelet on the shade, at rest
  const CLICK_AT = 62, REARM_AT = 18, MAX_PULL = 96;
  const pts = Array.from({ length: N }, (_, i) => ({ x: ANCHOR.x, y: ANCHOR.y + i * SEG, ox: ANCHOR.x, oy: ANCHOR.y + i * SEG }));
  const REST = ANCHOR.y + (N - 1) * SEG;

  // the lamp is a pendulum. A tug on the cord kicks it, and all the light moves with it.
  const L = { a: 0, v: 0 };
  let armed = true, lightsOn = true;
  let hold = null;                                // where the bead is being held, or null
  let drag = null;

  function anchor() {
    const r = L.a * Math.PI / 180, dx = ANCHOR.x - 250, dy = ANCHOR.y;
    return { x: 250 + dx * Math.cos(r) - dy * Math.sin(r), y: dx * Math.sin(r) + dy * Math.cos(r) };
  }

  function step(h) {
    // lamp: pendulum with a little damping
    L.v += (-34 * L.a - 1.3 * L.v) * h; L.a += L.v * h;
    const a = anchor();
    const damp = 0.992, g = 2100 * h * h, MAX_V = 4;
    for (let i = 1; i < N; i++) {
      const p = pts[i];
      let vx = (p.x - p.ox) * damp, vy = (p.y - p.oy) * damp;
      // a stretched cord lets go all at once: cap the recoil so it can't whip up into a knot
      const sp = Math.hypot(vx, vy);
      if (sp > MAX_V) { vx *= MAX_V / sp; vy *= MAX_V / sp; }
      p.ox = p.x; p.oy = p.y;
      // the bead weighs more than the rope, so it falls harder
      p.x += vx; p.y += vy + g * (i === N - 1 ? 1.5 : 1);
    }
    pts[0].x = a.x; pts[0].y = a.y;
    if (hold) { pts[N - 1].x = hold.x; pts[N - 1].y = hold.y; }
    for (let k = 0; k < 14; k++) {
      for (let i = 0; i < N - 1; i++) {
        const p = pts[i], q = pts[i + 1];
        const dx = q.x - p.x, dy = q.y - p.y, d = Math.hypot(dx, dy) || 1;
        const diff = (d - SEG) / d;
        if (i === 0) { q.x -= dx * diff; q.y -= dy * diff; }
        else if (i === N - 2 && hold) { p.x += dx * diff; p.y += dy * diff; }
        else { p.x += dx * diff / 2; p.y += dy * diff / 2; q.x -= dx * diff / 2; q.y -= dy * diff / 2; }
      }
    }
    pts[0].x = a.x; pts[0].y = a.y;
    if (hold) { pts[N - 1].x = hold.x; pts[N - 1].y = hold.y; }
    for (let i = 1; i < N; i++) if (pts[i].y < a.y) pts[i].y = a.y;   // it hangs from the eyelet, never above it
  }

  // a smooth line through the points, not a polygon
  function ropePath() {
    let d = `M${pts[0].x.toFixed(1)} ${pts[0].y.toFixed(1)}`;
    for (let i = 1; i < N - 1; i++) {
      const mx = (pts[i].x + pts[i + 1].x) / 2, my = (pts[i].y + pts[i + 1].y) / 2;
      d += `Q${pts[i].x.toFixed(1)} ${pts[i].y.toFixed(1)} ${mx.toFixed(1)} ${my.toFixed(1)}`;
    }
    return d + `L${pts[N - 1].x.toFixed(1)} ${pts[N - 1].y.toFixed(1)}`;
  }
  const ropeEls = ['ropeShade', 'ropeBase', 'ropeTwist'].map($);
  function paint() {
    const d = ropePath();
    ropeEls.forEach(el => el.setAttribute('d', d));
    const end = pts[N - 1], prev = pts[N - 3];
    const ang = Math.atan2(end.x - prev.x, end.y - prev.y) * -180 / Math.PI;
    bead.style.transform = `translate(${end.x.toFixed(1)}px, ${(end.y + 22).toFixed(1)}px) rotate(${ang.toFixed(2)}deg)`;
    bead.style.transformOrigin = '50% 6px';
    lamp.style.transform = `rotate(${L.a.toFixed(3)}deg)`;
    pool.style.transform = `rotate(${(L.a * 1.2).toFixed(3)}deg)`;
    // the card's shadow leans away from the light, the highlight on it follows
    const card = $('card');
    // (a positive angle swings the bulb left)
    card.style.setProperty('--sx', `${(L.a * 3).toFixed(2)}px`);
    card.style.setProperty('--lc', `${(40 - L.a * 3.2).toFixed(1)}%`);
    document.body.style.setProperty('--lx', `${(46 - L.a * 1.6).toFixed(1)}%`);
  }

  // the switch sits at the end of the pull: it clicks on the way down, not on release
  function checkSwitch() {
    const pull = pts[N - 1].y - REST;
    if (armed && pull >= CLICK_AT) {
      armed = false;
      toggle();
      if (navigator.userActivation?.hasBeenActive) navigator.vibrate?.(12);
      // the catch gives: a jolt down the rope and a kick to the lamp
      pts[N - 1].oy -= 4;
      if (!reduce) L.v += (pts[N - 1].x - ANCHOR.x > 0 ? -1 : 1) * 20;
    }
    if (!armed && pull < REARM_AT) armed = true;
  }

  function setTheme(on, announce = true) {
    lightsOn = on;
    root.dataset.theme = on ? 'light' : 'dark';
    bead.setAttribute('aria-checked', on);
    $('state').textContent = on ? 'Lights on. Pull the cord for dark mode.' : 'Lights off. Pull the cord for light.';
    if (announce) say(on ? 'Lights on' : 'Lights off');
  }
  function toggle() {
    const on = !lightsOn;
    setTheme(on);
    // only a real pull counts as a choice, so an untouched page keeps following the system
    try { localStorage.setItem('pull-cord-theme', on ? 'light' : 'dark'); } catch (e) {}
    if (on && !reduce) {                         // an old bulb catches, it doesn't just appear
      room.classList.remove('flicker'); room.offsetWidth; room.classList.add('flicker');
      setTimeout(() => room.classList.remove('flicker'), 340);
    }
  }

  // physics clock: fixed 120Hz steps, sleeps when everything hangs still
  let raf = 0, last = 0, acc = 0;
  function frame(now) {
    raf = 0;
    acc += Math.max(0, Math.min(40, now - last)) / 1000; last = now;
    const h = 1 / 120;
    let steps = 0;
    while (acc >= h && steps < 8) { step(h); acc -= h; steps++; }
    checkSwitch();
    paint();
    const end = pts[N - 1];
    const speed = Math.abs(end.x - end.ox) + Math.abs(end.y - end.oy) + Math.abs(L.v) / 60;
    if (hold || tug || speed > 0.012 || Math.abs(L.a) > 0.02) raf = requestAnimationFrame(frame);
  }
  function wake() { if (!raf) { last = performance.now(); raf = requestAnimationFrame(frame); } }

  // drag: the rope only gives so far; past that the pull is damped
  function roomPoint(e) {
    const b = room.getBoundingClientRect();
    return { x: (e.clientX - b.left) / scale, y: (e.clientY - b.top) / scale };
  }
  function clampHold(x, y) {
    const a = anchor();
    let dx = x - a.x, dy = y - a.y, d = Math.hypot(dx, dy);
    const free = (N - 1) * SEG;
    if (d > free) {
      // past straight: a rubber band up to MAX_PULL
      const over = d - free;
      const eased = MAX_PULL * (1 - Math.exp(-over / (MAX_PULL * 0.9)));
      const k = (free + eased) / d;
      x = a.x + dx * k; y = a.y + dy * k;
    }
    return { x, y };
  }
  bead.addEventListener('pointerdown', e => {
    if (!e.isPrimary || e.button !== 0 || drag) return;
    bead.setPointerCapture(e.pointerId);
    const p = roomPoint(e), end = pts[N - 1];
    drag = { id: e.pointerId, ox: p.x - end.x, oy: p.y - end.y, sx: e.clientX, sy: e.clientY, moved: 0 };
    hold = { x: end.x, y: end.y };
    wake();
  });
  bead.addEventListener('pointermove', e => {
    if (!drag || e.pointerId !== drag.id) return;
    const p = roomPoint(e);
    drag.moved = Math.max(drag.moved, Math.hypot(e.clientX - drag.sx, e.clientY - drag.sy));
    hold = clampHold(p.x - drag.ox, p.y - drag.oy);
    wake();
  });
  const letGo = e => {
    if (!drag || e.pointerId !== drag.id) return;
    const tapped = drag.moved < 4 && e.type === 'pointerup';
    drag = null; hold = null; wake();            // verlet keeps the bead's speed by itself
    lastPointer = e.timeStamp;
    if (tapped) pullOnce();                      // a tap on the bead is a tug
  };
  bead.addEventListener('pointerup', letGo);
  bead.addEventListener('pointercancel', letGo);

  // keyboard: Space or Enter plays the same tug, through the same switch
  let tug = 0, lastPointer = -1e9;
  function pullOnce() {
    if (reduce) { toggle(); return; }
    if (tug) return;
    const start = performance.now();
    tug = 1;
    (function down(now) {
      const u = Math.min(1, Math.max(0, now - start) / 170);
      const y = REST + (CLICK_AT + 56) * u * u * (3 - 2 * u);     // through the rubber band, past the click
      hold = clampHold(ANCHOR.x, y);
      wake();
      // it holds at the bottom until the switch has gone, however slow the frames are
      if ((armed || u < 1) && now - start < 900) requestAnimationFrame(down);
      else { hold = null; tug = 0; wake(); }
    })(start);
  }
  // screen readers fire click with no pointer: the pointer path already handled its own
  bead.addEventListener('click', e => { if (e.timeStamp - lastPointer > 500) pullOnce(); });
  bead.addEventListener('keydown', e => {
    if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); if (!e.repeat) pullOnce(); }
  });

  // start from what the visitor chose before, else from their system
  const params = new URLSearchParams(location.search);
  const still = params.get('still');
  let saved = null;
  try { saved = localStorage.getItem('pull-cord-theme'); } catch (e) {}
  if (!still) {
    const startOn = saved ? saved === 'light' : !matchMedia('(prefers-color-scheme: dark)').matches;
    root.style.transition = 'none';
    setTheme(startOn, false);
    root.offsetWidth; root.style.transition = '';
  }
  // settle the rope before the first paint
  for (let i = 0; i < 240; i++) step(1 / 120);
  paint();

  // stills and the demo loop
  if (still === 'pull') {
    // caught at the click: rope taut and leaning, the lamp kicked off centre
    L.a = -3.4;
    // a fast pull out to the side, caught as the hand arrives: the rope is still catching up, so it bows
    const STEPS = 22;
    for (let i = 1; i <= STEPS; i++) {
      const u = i / STEPS, e = u * u * (3 - 2 * u);
      hold = clampHold(ANCHOR.x + 146 * e, REST + 96 * e);
      step(1 / 120); L.a = -3.4 * u; L.v = 0;
    }
    armed = false; paint();
  }
  if (still === 'off') {
