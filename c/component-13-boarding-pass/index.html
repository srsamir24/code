<script>
  const main = document.getElementById('main');
    
  // deterministic noise, so the QR, barcode and torn edge are the same on every load
  const rand = i => { const t = Math.sin(i * 127.1 + 311.7) * 43758.5453; return t - Math.floor(t); };

  // QR-like code: three finder squares, timing rows, one alignment square, seeded modules
  (function drawQR() {
    const n = 25, fin = (x, y) => (x < 8 && y < 8) || (x > n - 9 && y < 8) || (x < 8 && y > n - 9);
    let d = '', i = 0;
    for (let y = 0; y < n; y++) for (let x = 0; x < n; x++) {
      let on;
      if (fin(x, y)) {
        const lx = x < 8 ? x : x - (n - 7), ly = y < 8 ? y : y - (n - 7);
        const r = Math.max(Math.abs(lx - 3), Math.abs(ly - 3));
        on = lx >= 0 && ly >= 0 && lx < 7 && ly < 7 && r !== 2;
      } else if (Math.max(Math.abs(x - 18), Math.abs(y - 18)) <= 2) {
        on = Math.max(Math.abs(x - 18), Math.abs(y - 18)) !== 1;
      } else if (x === 6 || y === 6) on = (x + y) % 2 === 0;
      else on = rand(i++ + 418) > 0.52;
      if (on) d += `M${x} ${y}h1v1h-1z`;
    }
    const qr = document.getElementById('qr');
    qr.setAttribute('viewBox', `0 0 ${n} ${n}`);
    qr.setAttribute('shape-rendering', 'crispEdges');
    qr.innerHTML = `<path d="${d}" fill="currentColor"/>`;
  })();
  (function drawBars() {
    const tpl = document.getElementById('faceTpl').content;
    let x = 0, i = 0, r = '';
    while (x < 92) { const bw = 1 + Math.floor(rand(i++ + 77) * 3.2); if (i % 2) r += `<rect x="${x}" width="${bw}" height="62"/>`; x += bw; }
    tpl.getElementById('bars').innerHTML = `<g fill="currentColor">${r}</g>`;
    tpl.getElementById('bars').removeAttribute('id');
  })();

  const stub = document.getElementById('stub'), pass = document.getElementById('pass');
  const N = 30;
  let W = 360, w = W / N, FULL = 0.34 * W; // FULL: the drop at which the stub is torn all the way
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const st = { D: 0, side: 1, torn: false, p: 0 };
  let M = 0, Hm = 0, slices = [], spring = 0;

  // one shared torn line: y offset per 1.5px, used by both edges so the teeth match
  let J = [];
  const jag = (a, b, y0) => J.filter(([x]) => x >= a - 1.5 && x <= b + 1.5).map(([x, j]) => `${x}px ${(y0 + j).toFixed(2)}px`);

  function build() {
    W = stub.offsetWidth; w = W / N; FULL = 0.34 * W;
    J = [];
    for (let x = 0, i = 0; x <= W + 0.01; x += 1.5, i++) J.push([x, (rand(i) - 0.5) * 2.6 + Math.sin(i * 0.9) * 0.5]);
    stub.replaceChildren(); slices = [];
    Hm = main.offsetHeight; M = Hm - 3;
    const face = document.getElementById('faceTpl').content.firstElementChild;
    for (let j = 0; j < N; j++) {
      const x0 = j * w, el = document.createElement('div');
      el.className = 'slice';
      el.style.left = x0 + 'px';
      el.style.width = w + (j < N - 1 ? 1 : 0) + 'px'; // 1px overlap, no seams
      const f = face.cloneNode(true);
      f.style.left = -x0 + 'px'; f.style.width = W + 'px';
      const pts = jag(x0, x0 + w + 1, 3);
      f.style.setProperty('--jag', `polygon(${pts.join(',')}, ${x0 + w + 3}px 100%, ${x0 - 3}px 100%)`);
      const sheen = f.querySelector('.sheen');
      el.append(f); stub.append(el);
      slices.push({ el, sheen });
    }
    lucide.createIcons({ attrs: { 'stroke-width': 2.4 } });
  }

  // slide:js
  // D = how far the free corner has dropped. The tear runs
  // ahead of it: L = D / 0.34, and each slice is sheared
  // so the free part curves down like stiff paper.
  function draw() {
    const D = Math.max(0, st.D);
    const p = st.torn ? 1 : Math.min(1, D / FULL);
    const L = p * W, s = st.side, tip = s > 0 ? W - L : L;
    const t = x => L ? Math.min(1, Math.max(0, (s > 0 ? x - tip : tip - x) / L)) : 0;
    const drop = x => D * t(x) ** 1.6;
    // light catches the bend: brighter where the slope is steeper
    const lit = x => L ? Math.min(0.09, 0.16 * D * 1.6 * t(x) ** 0.6 / L).toFixed(3) : 0;
    let a = W, b = 0;
    slices.forEach(({ el, sheen }, j) => {
      const x0 = j * w, y0 = drop(x0), k = (drop(x0 + w) - y0) / w;
      const torn = L > 0 && (s > 0 ? x0 + w / 2 > tip : x0 + w / 2 < tip);
      if (torn) { a = Math.min(a, x0); b = Math.max(b, x0 + w); }
      el.classList.toggle('torn', torn);
      el.style.transform = y0 || k
        ? `translateY(${y0}px) skewY(${Math.atan(k)}rad)` : '';
      sheen.style.background = y0 || k ? `linear-gradient(90deg, oklch(1 0 0 / ${lit(x0)}) ${x0}px, oklch(1 0 0 / ${lit(x0 + w)}) ${x0 + w}px)` : '';
    });
    main.style.clipPath = b > a ? mainClip(a, b) : '';
    // /slide
    st.p = p;
    stub.style.filter = D > 0.5 ? `drop-shadow(0 ${D * 0.12}px ${4 + D * 0.18}px oklch(0.3 0.09 240 / ${Math.min(0.32, D * 0.005)}))` : '';
    const pct = Math.round(p * 100);
    stub.setAttribute('aria-valuenow', pct);
    stub.setAttribute('aria-valuetext', st.torn ? 'Torn off' : pct ? `Torn ${pct} percent` : 'Attached');
  }
  // the pass keeps its straight edge where the stub still holds, and a torn edge from a to b
  function mainClip(a, b) {
    const edge = jag(a, b, M);
    if (a <= 0 && b < W) return `polygon(0 0, ${W}px 0, ${W}px ${Hm}px, ${b}px ${Hm}px, ${edge.reverse().join(',')})`;
    if (a > 0) return `polygon(0 0, ${W}px 0, ${edge.reverse().join(',')}, ${a}px ${Hm}px, 0 ${Hm}px)`;
    return `polygon(0 0, ${W}px 0, ${edge.reverse().join(',')})`;
  }

  // paper resists, then gives: the first 40px of pull barely move it
  const rubber = d => d < 40 ? 7 * (1 - Math.exp(-d / 18)) : 7 * (1 - Math.exp(-40 / 18)) + (d - 40) * 0.95;

  // a spring on D keeps the release velocity, so a let-go snaps back in about 200ms
  function springTo(target, v = 0, done) {
    cancelAnimationFrame(spring); st.aim = target;
    if (reduce) { st.D = target; draw(); done && done(); return; }
    let last = performance.now();
    const step = now => {
      const dt = Math.min(0.032, (now - last) / 1000); last = now;
      v += (-520 * (st.D - target) - 38 * v) * dt;
      st.D += v * dt; draw();
      if (Math.abs(st.D - target) < 0.1 && Math.abs(v) < 2) { st.D = target; draw(); done && done(); return; }
      spring = requestAnimationFrame(step);
    };
    spring = requestAnimationFrame(step);
  }

  // drag: pointer capture, first finger only, side picked by where you pull
  let drag = null;
  stub.addEventListener('pointerdown', e => {
    if (drag || st.torn || e.button > 0) return;
    cancelAnimationFrame(spring); st.aim = null;
    const r = stub.getBoundingClientRect();
    drag = { id: e.pointerId, x: e.clientX, y: e.clientY, gx: e.clientX - r.left, gy: e.clientY - r.top, side: 0, d: 0, trail: [] };
    stub.setPointerCapture(e.pointerId);
  });
  stub.addEventListener('pointermove', e => {
    if (!drag) { peek(e); return; }
    if (e.pointerId !== drag.id) return;
    const dx = e.clientX - drag.x, dy = e.clientY - drag.y;
    if (!drag.side && Math.hypot(dx, dy) > 4)
      drag.side = Math.abs(dx) > Math.abs(dy) ? Math.sign(dx) : (drag.gx > W / 2 ? 1 : -1);
    if (!drag.side) return;
    drag.d = Math.max(0, dy) + Math.max(0, drag.side * dx) * 0.7;
    st.side = drag.side; st.D = rubber(drag.d); draw();
    const now = performance.now();
    drag.trail.push([now, drag.d]);
    while (drag.trail.length > 2 && now - drag.trail[0][0] > 90) drag.trail.shift();
    if (st.p >= 1) tearFree(velocity());
  });
  const velocity = () => {
    const t = drag && drag.trail; if (!t || t.length < 2) return 0;
    const [t0, d0] = t[0], [t1, d1] = t[t.length - 1];
    if (performance.now() - t1 > 60) return 0; // held still before letting go: no flick
    return t1 > t0 ? (d1 - d0) / (t1 - t0) : 0; // px per ms
  };
  function release(e) {
    if (!drag || e.pointerId !== drag.id) return;
    const v = velocity();
    // far enough, or a quick flick once the paper has given
    if (st.p >= 0.7 || (v > 0.45 && st.p > 0.06)) tearFree(v);
    else { drag = null; springTo(0, v * 950); }
  }
  stub.addEventListener('pointerup', release);
  stub.addEventListener('pointercancel', release);

  // a fine pointer resting on the stub lifts the near corner a little
  const fine = matchMedia('(hover: hover) and (pointer: fine)');
  let peeked = 0;
  function peek(e) {
    if (!fine.matches || st.torn || hold) return;
    const side = e.clientX - stub.getBoundingClientRect().left > W / 2 ? 1 : -1;
    if (side !== st.side && st.D > 0.2) st.D = 0;
    if (side === peeked) return;
    peeked = side; st.side = side; springTo(3.5);
  }
  stub.addEventListener('pointerleave', () => { peeked = 0; if (!drag && !st.torn && !hold) springTo(0); });

  // the rest of the perforation zips, then the stub falls with spin and gravity
  function tearFree(v = 0.4) {
    if (st.torn) return;
    const g = drag ? [drag.gx, drag.gy] : [st.side > 0 ? W - 30 : 30, 100];
    drag = null; cancelAnimationFrame(spring);
    const p0 = st.p, D0 = Math.max(st.D, 24);
    if (reduce) {
      st.torn = true; draw();

  function demo() {
