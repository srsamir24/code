<script id="component">
  // Feed it events: 'start', 'progress' {loaded, size, done}, 'done', 'fail'.
  const MILESTONES = [25, 50, 75];

  // Round the estimate. Nobody believes "1 min 47 s".
  const eta = (s) =>
    s < 10 ? 'A few seconds left'
    : s < 60 ? 'Less than a minute left'
    : s < 5400 ? `About ${Math.round(s / 60)} min left`
    : `About ${Math.round(s / 3600)} h left`;

  for (const job of document.querySelectorAll('.job')) {
    const bar = job.querySelector('.job-bar');
    const title = job.querySelector('.job-title');
    const files = job.querySelector('.job-files');
    const left = job.querySelector('.job-eta');
    const alert = job.querySelector('.job-alert');
    const retry = job.querySelector('.job-retry');
    const download = job.querySelector('.job-download');
    const live = job.querySelector('.job-live');
    const total = Number(job.dataset.files);
    const stallAfter = Number(job.dataset.stall) * 1000;
    const runTitle = title.textContent;
    let s, timer;

    const problem = (msg) => {
      job.dataset.state = 'stalled';
      alert.textContent = msg;
      left.textContent = '';
      retry.hidden = false;
    };
    const clearProblem = () => {
      if (document.activeElement === retry) title.focus();
      alert.textContent = '';
      retry.hidden = true;
    };
    const watch = () => {
      clearTimeout(timer);
      timer = setTimeout(() => problem(
        `No progress for ${stallAfter / 1000} s. Check your connection.`
      ), stallAfter);
    };
    const freshEstimate = () => Object.assign(s, { rate: null, n: 0, t0: null, last: null });

    // No size yet: no value = indeterminate.
    job.addEventListener('start', () => {
      s = { pct: -1, said: 0 };
      freshEstimate();
      clearTimeout(timer);
      clearProblem();
      job.dataset.state = 'running';
      title.textContent = runTitle;
      bar.removeAttribute('value');
      files.textContent = 'Preparing…';
      left.textContent = '';
      download.hidden = true;
    });

    job.addEventListener('progress', (e) => {
      const { loaded, size, done, at = performance.now() } = e.detail;
      const pct = Math.min(loaded / size * 100, 99.9);
      if (pct <= s.pct) return; // never go backwards

      // Smoothed speed, so the estimate doesn't jump with every chunk.
      if (s.last) {
        const r = (loaded - s.last.loaded) / (at - s.last.at);
        s.rate = s.rate === null ? r : 0.3 * r + 0.7 * s.rate;
        s.n++;
      } else s.t0 = at;
      s.last = { loaded, at };
      s.pct = pct;
      if (job.dataset.state === 'stalled') clearProblem();
      job.dataset.state = 'running';

      const shown = Math.floor(pct); // 99% until done
      bar.value = pct;
      files.textContent = `${done} of ${total} files · ${shown}%`;
      left.textContent = s.n >= 3 && at - s.t0 >= 3000
        ? eta((size - loaded) / s.rate / 1000)
        : 'Estimating…';

      // Announce milestones, not every tick.
      const m = MILESTONES.filter((x) => shown >= x).pop() || 0;
      if (m > s.said) {
        s.said = m;
        live.textContent = `${m}% done, ${done} of ${total} files.`;
      }
      watch();
    });

    job.addEventListener('done', () => {
      clearTimeout(timer);
      clearProblem();
      job.dataset.state = 'done';
      title.textContent = job.dataset.doneTitle;
      bar.value = 100;
      files.textContent = `${total} of ${total} files · 100%`;
      left.textContent = '';
      download.hidden = false;
      live.textContent = `${job.dataset.doneTitle}. ${total} of ${total} files ready.`;
    });

    job.addEventListener('fail', () => {
      clearTimeout(timer);
      problem(`Export stopped at ${Math.floor(s.pct)}%. Nothing is lost.`);
    });

    retry.addEventListener('click', () => {
      clearProblem();
      job.dataset.state = 'running';
      freshEstimate();                   // old speed is stale
      left.textContent = 'Estimating…';
      job.dispatchEvent(new CustomEvent('retry'));
      watch();
    });

    job.dispatchEvent(new CustomEvent('start'));
  }
</script>

<!-- Demo page only: a fake export that feeds the component. -->
<script>
  (() => {
    const job = document.querySelector('.job');
    const stall = document.getElementById('stall');
    const size = 80e6;
    let loaded = 0, tick;
    const step = () => {
      if (!stall.checked) {
        loaded = Math.min(size, loaded + size * (0.009 + Math.random() * 0.006));
        job.dispatchEvent(new CustomEvent('progress', { detail: { loaded, size, done: Math.floor(loaded / size * 8) } }));
        if (loaded >= size) { clearInterval(tick); job.dispatchEvent(new CustomEvent('done')); return; }
      }
    };
    document.getElementById('run').addEventListener('click', () => {
      clearInterval(tick);
      loaded = 0;
      job.dispatchEvent(new CustomEvent('start'));
      setTimeout(() => { tick = setInterval(step, 200); }, 1500);
    });
    job.addEventListener('retry', () => { stall.checked = false; });
  })();
</script>
