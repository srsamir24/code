<script>
const $ = s => document.querySelector(s);
const calm = matchMedia('(prefers-reduced-motion: reduce)').matches;
const app = $('#app'), form = $('#form'), email = $('#email'), pass = $('#pass'), eye = $('#eye'), go = $('#go');
const emailField = $('#emailField'), passField = $('#passField');
const validEmail = v => /^[^\s@]+@[^\s@]+\.[a-z]{2,}$/i.test(v.trim());

function setErr(field, box, text) {
  field.classList.toggle('bad', !!text);
  box.classList.toggle('on', !!text);
  if (text) box.querySelector('span').textContent = text;
  field.querySelector('input').setAttribute('aria-invalid', !!text);
}
const nudge = el => { if (!calm) el.animate([0, -8, 7, -5, 3, 0].map(x => ({ transform: `translateX(${x}px)` })), { duration: 340, easing: 'cubic-bezier(0.33, 0.1, 0.6, 1)' }); };

const emailProblem = () => validEmail(email.value) ? '' : email.value.includes('@') ? 'That address needs an ending, like .com' : 'Enter your email address';
const passProblem = () => pass.value.length >= 8 ? '' : 'Use at least 8 characters';

// an error clears as soon as the text is right, not on the next submit
email.addEventListener('input', () => {
  emailField.classList.toggle('good', validEmail(email.value));
  if (emailField.classList.contains('bad') && !emailProblem()) setErr(emailField, $('#emailErr'), '');
});
pass.addEventListener('input', () => { if (passField.classList.contains('bad') && !passProblem()) setErr(passField, $('#passErr'), ''); });
// tell people about a bad address when they leave the field. Not on the way to Sign in:
// the error would push that button down in the middle of the click, and submit checks it anyway
email.addEventListener('blur', e => {
  if (email.value && e.relatedTarget !== go) setErr(emailField, $('#emailErr'), emailProblem());
});

eye.addEventListener('click', () => {
  const show = pass.type === 'password';
  pass.type = show ? 'text' : 'password';
  eye.setAttribute('aria-pressed', show);
  eye.setAttribute('aria-label', show ? 'Hide password' : 'Show password');
});
['#forgot', '#join'].forEach(s => $(s).addEventListener('click', e => e.preventDefault()));

form.addEventListener('submit', e => {
  e.preventDefault();
  if (go.getAttribute('aria-busy') === 'true') return;
  const a = emailProblem(), b = passProblem();
  // on the way to the first submit an empty password is not worth a second error yet
  setErr(emailField, $('#emailErr'), a);
  setErr(passField, $('#passErr'), a && !pass.value ? '' : b);
  if (a) { nudge(emailField); email.focus(); return; }
  if (b) { nudge(passField); pass.focus(); return; }
  go.setAttribute('aria-busy', 'true');
  setTimeout(() => {
    const name = email.value.trim().split('@')[0].split(/[._-]/)[0];
    $('#hello').textContent = `You're in, ${name[0].toUpperCase()}${name.slice(1)}`;
    app.classList.add('in');
    go.setAttribute('aria-busy', 'false');
    $('#out').focus({ preventScroll: true });
  }, 600);
});
$('#out').addEventListener('click', () => {
  app.classList.remove('in');
  form.reset();
  emailField.classList.remove('good');
  pass.type = 'password';
  eye.setAttribute('aria-pressed', 'false');
  eye.setAttribute('aria-label', 'Show password');
});
</script>
