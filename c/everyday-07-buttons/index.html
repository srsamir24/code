<script>
const calm = matchMedia('(prefers-reduced-motion: reduce)').matches;
const run = (el, frames, opts) => calm ? null : el.animate(frames, opts);

// how long each one stays busy, and what it ends on
const WORK = { shrink: [700, 'done'], fill: [1000, 'done'], slide: [820, 'done'], shake: [380, 'fail'], pop: [340, 'done'] };
const WORDS = {
  shrink: { idle: 'Send', load: 'Sending', done: 'Sent' }, fill: { idle: 'Upload', load: 'Uploading', done: 'Uploaded' },
  press: { idle: 'Save', done: 'Saved' }, slide: { idle: 'Send invite', load: 'Sending invite', done: 'Invite sent' },
  shake: { idle: 'Pay $48.00', load: 'Paying', fail: 'Card declined. Try again' }, pop: { idle: 'Book table', load: 'Booking', done: 'Booked' },
};

document.querySelectorAll('.sb').forEach((sb, n) => {
  const kind = sb.dataset.kind;
  const demoBox = sb.parentElement;
  const tap = document.createElement('i');
  tap.className = 'tap';
  demoBox.append(tap);
  let token = 0, touched = false, step = 0;
  const after = (ms, fn) => { const t = token; setTimeout(() => { if (t === token) fn(); }, ms); };
  const set = s => {
    sb.dataset.state = s;
    sb.setAttribute('aria-label', WORDS[kind][s]);
    // both copies carry the same word while the fill runs, so the letters line up at its edge
    if (kind === 'fill') sb.querySelector('.over').textContent = s === 'done' ? 'Uploaded' : 'Upload';
  };

  function burst() {
    const box = demoBox.querySelector('.burst');
    box.innerHTML = '<i></i>'.repeat(12);
    [...box.children].forEach((d, k) => {
      const a = k / 12 * Math.PI * 2, x = Math.cos(a) * (k % 2 ? 110 : 88), y = Math.sin(a) * (k % 2 ? 46 : 36);
      d.style.background = k % 3 ? 'var(--hot)' : 'var(--ink)';
      run(d, [{ opacity: 0, transform: `translate(${x * 0.5}px, ${y * 0.5}px) scale(0.6)` }, { opacity: 1, transform: `translate(${x * 0.6}px, ${y * 0.6}px) scale(1)`, offset: 0.1 }, { opacity: 1, transform: `translate(${x * 0.85}px, ${y * 0.85}px) scale(1)`, offset: 0.55 }, { opacity: 0, transform: `translate(${x}px, ${y}px) scale(0.3)` }], { duration: 760, easing: 'cubic-bezier(0.16, 1, 0.3, 1)' });
    });
  }
  const FINISH = {
    shake() { run(sb, [0, -15, 14, -11, 8, -4, 0].map(x => ({ transform: `translateX(${x}px)` })), { duration: 500, delay: 180, easing: 'cubic-bezier(0.33, 0.1, 0.6, 1)' }); },
    pop() { burst(); },
  };

  function go(pressedByHand) {
    token++;
    if (kind === 'press') {
      if (pressedByHand) { set('done'); return; }
      sb.classList.add('down');
      after(220, () => { sb.classList.remove('down'); set('done'); });
      return;
    }
    set('load');
    const [ms, end] = WORK[kind];
    after(ms, () => { set(end); FINISH[kind]?.(); });
  }
  function back() { token++; sb.classList.remove('down'); set('idle'); }

  // a real <button>, so Space and Enter land here as well
  sb.addEventListener('click', ev => {
    if (ev.isTrusted) touched = true;
    if (sb.dataset.state === 'load') return;
    sb.dataset.state === 'idle' ? go(true) : back();
  });

  // 8 taps at 1500 ms is 12 s; the route ends on the state it starts in
  const start = sb.dataset.state;
  set(start);
  const route = start === 'idle' ? [go, back] : [back, go];
  function demo() {
    if (touched) return;
    const act = route[step++ % route.length];
    tap.style.left = demoBox.offsetWidth / 2 + 66 + 'px';
    tap.animate([{ opacity: 0, transform: 'scale(0.6)' }, { opacity: 1, transform: 'scale(1)', offset: 0.25 }, { opacity: 0, transform: 'scale(1.25)' }], { duration: 300, easing: 'ease-out' });
    setTimeout(() => { if (!touched) act(false); }, 60);
    setTimeout(demo, 1500);
  }
  document.fonts.ready.then(() => setTimeout(demo, 60 + n * 240));
});
</script>
