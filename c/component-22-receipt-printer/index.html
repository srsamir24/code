<script>
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const $ = id => document.getElementById(id);
  const fit = $('fit'), feedEl = $('feed'), box = $('box'), paper = $('paper'), stub = $('stub'), go = $('go');
  const WIDTH = 336, SLOT = 548, TEARLINE = 16;  // the tear runs 16px above the slot
  const TEETH = 24, FLICK = 0.11, TEAR_AT = 150;

  let scale = 1;
  function resize() {
    // 780: a corner of the receipt rises above the box during a full tear
    scale = Math.min(1, (innerWidth - 16) / 440, (innerHeight - 16) / 780);
    fit.style.transform = `scale(${scale})`;
    fit.style.margin = `${(700 * scale - 700) / 2}px ${(440 * scale - 440) / 2}px`;
  }
  addEventListener('resize', resize); resize();
  $('teethPath').setAttribute('d', 'M0 9' + Array.from({ length: 40 }, (_, i) => `L${i * 9.8 + 4.9} 0.5L${(i + 1) * 9.8} 9`).join('') + 'Z');

  // prices include 15% VAT, the Norwegian rate for food and drink. Everything below is added up, not typed.
  const ORDERS = [
    { no: '0418', time: '14:22', items: [[2, 'Filter coffee', 42], [1, 'Cardamom bun', 48], [1, 'Oat flat white', 62, 'extra shot'], [1, 'Rye sandwich', 96], [1, 'Beans, Halden blend', 189, '250 g, whole']] },
    { no: '0419', time: '14:26', items: [[1, 'Cortado', 52], [2, 'Cinnamon bun', 46], [1, 'Sparkling water', 38]] },
    { no: '0420', time: '14:31', items: [[3, 'Filter coffee', 42], [1, 'Chocolate cake', 72, 'to go'], [1, 'Oat milk, carton', 39], [2, 'Rye sandwich', 96]] },
  ];
  const kr = n => n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }).replace(',', ' ');
  function receipt(o) {
    const sum = o.items.reduce((s, [q, , p]) => s + q * p, 0);
    const vat = sum - sum / 1.15;
    // bars from the order number: decorative, not a scannable symbology
    const bars = [...(o.no + o.no + o.no)].flatMap(d => [+d % 3 + 1, +d % 2 + 2]).map((w, i) => `<i style="width:${w + (i % 3 === 0 ? 1 : 0)}px;margin-right:${(i % 4) + 1}px"></i>`).join('');
    return `<h1>Halden Kaffe</h1><p class="where">Storgata 14, 0184 Oslo</p>
      <div class="meta"><span>Order ${o.no}</span><span>3 Nov, ${o.time}</span></div>
      <div class="rows">${o.items.map(([q, n, p, note]) => `<div class="rowline"><b>${q}</b><span>${n}</span><span>${kr(q * p)}</span>${note ? `<small>${note}</small>` : ''}</div>`).join('')}</div>
      <div class="sums"><div class="sumline"><span>Includes VAT 15%</span><span>${kr(vat)}</span></div>
      <div class="total"><span>Total NOK</span><span>${kr(sum)}</span></div></div>
      <div class="code" aria-hidden="true">${bars}</div>
      <p class="thanks">Thanks. Paid by card ending 4021</p><div class="curl"></div>`;
  }

  // the edge: a seeded zigzag. The same points cut the receipt's bottom, the stub's top, and the next receipt's top.
  function edge(seed) {
    let s = seed * 9301 + 49297;
    const rnd = () => (s = (s * 9301 + 49297) % 233280) / 233280;
    return Array.from({ length: TEETH * 2 + 1 }, (_, i) => [
      i / (TEETH * 2) * WIDTH,
      (i % 2 ? 5 : 0) + (rnd() - 0.5) * 3.2,
    ]);
  }
  const edgePoly = (e, y) => e.map(([x, d]) => `${x.toFixed(1)}px ${(y + d).toFixed(1)}px`);
  function clipPaper(top, bottom, h) {
    paper.style.clipPath = `polygon(${[...edgePoly(top, 5), ...edgePoly(bottom, h - 5).reverse()].join(',')})`;
  }
  function clipStub(e) {
    stub.style.clipPath = `polygon(${[...edgePoly(e, 3), `${WIDTH}px 26px`, '0 26px'].join(',')})`;
  }

  /* state: idle (nothing out, Print) -> printing -> ready (attached, Tear off) -> torn (Print next) */
  let state = 'idle', orderIndex = 0, seed = 3;
  let topEdge = edge(seed), nextEdge = edge(seed + 1);
  let H = 0, fed = 0;                            // receipt height, how far it's out
  let printedTotal = 0;                          // paper used, for the roll
  const say = t => { live.textContent = ''; setTimeout(() => live.textContent = t, 30); };
  const live = $('live');
  clipStub(topEdge);

  // transform state
  const T = { x: 0, y: 0, r: 0, vx: 0, vy: 0, vr: 0, tx: 0, ty: 0, tr: 0, ox: 168, p: 0, vp: 0, tp: 0, side: 1, sway: 0, vsway: 0 };

  function place() {
    // the feed offset puts the paper's bottom edge at the tear line once it's fully out
    const baseY = SLOT - TEARLINE - fed;
    box.style.top = baseY + 'px';
    box.style.transformOrigin = `${T.ox}px ${H}px`;
    const swayRot = T.sway;
    box.style.transform = `translate3d(${T.x.toFixed(2)}px, ${T.y.toFixed(2)}px, 0) rotate(${(T.r + swayRot).toFixed(3)}deg)`;
  }
  function spinRoll() {
    const left = Math.max(0, 1 - printedTotal / 2500);
    const r = 19 + 19 * Math.sqrt(left);         // area of a roll, not its radius, goes down with the paper used
    $('rollPaper').setAttribute('r', r.toFixed(2));
    $('rollRings').setAttribute('r', Math.max(15, r - 7).toFixed(2));
    $('rollCore').setAttribute('transform', `rotate(${(-printedTotal / r * 57.3).toFixed(1)} 46 46)`);
  }

  function setUI(name, info, btn, on = true) {
    $('stateName').textContent = name; $('stateInfo').textContent = info;
    // aria-disabled, not disabled: the button keeps focus while it prints
    go.textContent = btn; go.setAttribute('aria-disabled', !on);
  }

  // print: the motor moves in steps, so the paper does too
  function print(instant) {
    if (state === 'printing') return;
    const o = ORDERS[orderIndex % ORDERS.length];
    paper.innerHTML = receipt(o);
    paper.style.height = 'auto';
    box.style.display = 'block';
    H = paper.offsetHeight;
    paper.style.height = H + 'px';
    clipPaper(topEdge, nextEdge, H);
    Object.assign(T, { x: 0, y: 0, r: 0, vx: 0, vy: 0, vr: 0, tx: 0, ty: 0, tr: 0, p: 0, tp: 0, vp: 0, sway: 0, vsway: 0 });
    fed = 0;
    feedEl.classList.add('printing');
    state = 'printing';
    setUI('Printing', `Order ${o.no}`, 'Printing', false);
    say(`Printing order ${o.no}`);
    place();
    if (instant || reduce) { finishPrint(o, instant); return; }
    let lastT = performance.now(), acc = 0;
    const STEP = 7, EVERY = 22;                  // 7px every 22ms
    (function step(now) {
      if (state !== 'printing') return;
      acc += now - lastT; lastT = now;
      while (acc >= EVERY && fed < H) {
        acc -= EVERY;
        fed = Math.min(H, fed + STEP);
        printedTotal += STEP;
        // each step gives the free end a tiny kick, which springs out
        T.vsway += (Math.random() - 0.5) * 2.4;
      }
      T.vsway += (-90 * T.sway - 9 * T.vsway) * 0.016; T.sway += T.vsway * 0.016;
      if (fed > 14) clipStub(nextEdge);
      place(); spinRoll();
      if (fed < H) requestAnimationFrame(step);
      else finishPrint(o);
    })(lastT);
  }
  function finishPrint(o, instant) {
    if (fed < H) printedTotal += H - fed;
    fed = H; T.sway = 0;
    feedEl.classList.remove('printing');
    clipStub(nextEdge);
    place(); spinRoll();
    state = 'ready';
    setUI('Printed', 'Pull one side to tear', 'Tear off');
    if (!instant) say(`Order ${o.no} printed. Tear it off.`);
  }

  // the tear: pull one side up and the rip runs across the teeth from that side.
  // The receipt pivots on the tip of the rip, which walks along the bottom edge.
  function setTear(p) {
    T.p = p;
    const walked = p * WIDTH;
    T.ox = T.side > 0 ? WIDTH - walked : walked;
    T.r = -T.side * p * 13;                      // right side up = anticlockwise
    T.y = -p * 6;
    if (state === 'ready') {
      const on = p > 0.02;
      $('stateName').textContent = on ? 'Tearing' : 'Printed';
      $('stateInfo').textContent = on ? `Rip is ${Math.round(p * 100)}% across` : 'Pull one side to tear';
    }
  }

  let spring = 0, drag = null, lastTooth = 0, walking = false;
  function loop(now) {
    spring = 0;
    const dt = Math.max(0, Math.min(0.032, (now - loopT) / 1000)); loopT = now;
    let moving = false;
    let t = dt;
    while (t > 0) {
      const h = Math.min(t, 1 / 240); t -= h;
      if (state === 'torn') {
        T.vx += (150 * (T.tx - T.x) - 15 * T.vx) * h; T.x += T.vx * h;
        T.vy += (150 * (T.ty - T.y) - 15 * T.vy) * h; T.y += T.vy * h;
        T.vr += (120 * (T.tr - T.r) - 13 * T.vr) * h; T.r += T.vr * h;
      } else {
        T.vp += (T.k * (T.tp - T.p) - T.c * T.vp) * h;
        setTear(Math.max(0, T.p + T.vp * h));
      }
    }
    moving = state === 'torn'
      ? Math.abs(T.vx) + Math.abs(T.vy) + Math.abs(T.vr) + Math.abs(T.tx - T.x) + Math.abs(T.ty - T.y) + Math.abs(T.tr - T.r) > 0.03
      : Math.abs(T.vp) + Math.abs(T.tp - T.p) > 0.0008;
    place();
    if (moving || drag) spring = requestAnimationFrame(loop);
  }
  let loopT = 0;
  function wake() { if (!spring) { loopT = performance.now(); spring = requestAnimationFrame(loop); } }
  T.k = 700; T.c = 48;

  function rip(vx, vy) {
    if (state === 'torn') return;
    state = 'torn'; drag = null;
    T.p = 1;
    const o = ORDERS[orderIndex % ORDERS.length];
    // velocity kept: px/ms to px/s, damped by the desk
    T.vx = vx * 600; T.vy = vy * 600; T.vr = -T.side * 40 + vx * 40;
    // it slides up the desk, further the harder you flicked, and rests at a slight angle
    T.tx = T.side * 6 + Math.max(-34, Math.min(34, vx * 50));
    T.ty = -24 + Math.max(-20, Math.min(0, vy * 30));
    T.tr = T.side * -2.6;
    topEdge = nextEdge; seed += 2; nextEdge = edge(seed);
    orderIndex++;
    const next = ORDERS[orderIndex % ORDERS.length];
    setUI('Torn off', `Next: order ${next.no}`, 'Print next');
    say(`Order ${o.no} torn off`);
    if (reduce) { T.x = T.tx; T.y = T.ty; T.r = T.tr; T.vx = T.vy = T.vr = 0; place(); return; }
    wake();
  }

  paper.addEventListener('pointerdown', e => {
    if (!e.isPrimary || e.button !== 0 || drag || walking || state !== 'ready') return;
    paper.setPointerCapture(e.pointerId);
    const b = paper.getBoundingClientRect();
    T.side = e.clientX > b.left + b.width / 2 ? 1 : -1;   // the rip starts on the side you hold
    drag = { id: e.pointerId, sx: e.clientX, sy: e.clientY, lx: e.clientX, ly: e.clientY, lt: e.timeStamp, vx: 0, vy: 0 };
    T.k = 1400; T.c = 70; wake();
  });
  paper.addEventListener('pointermove', e => {
    if (!drag || e.pointerId !== drag.id) return;
    const dt = Math.max(1, e.timeStamp - drag.lt);
    drag.vx = 0.6 * drag.vx + 0.4 * ((e.clientX - drag.lx) / scale / dt);
    drag.vy = 0.6 * drag.vy + 0.4 * ((e.clientY - drag.ly) / scale / dt);
    drag.lx = e.clientX; drag.ly = e.clientY; drag.lt = e.timeStamp;
    // pulling up or out to your side both count, sliding back does not
    const pull = Math.max(0, -(e.clientY - drag.sy) / scale + T.side * (e.clientX - drag.sx) / scale * 0.6);
    T.tp = Math.min(1, pull / TEAR_AT);
    const tooth = Math.floor(T.tp * TEETH);
    if (tooth !== lastTooth) { lastTooth = tooth; if (navigator.userActivation?.hasBeenActive) navigator.vibrate?.(3); }   // a tick per tooth, where phones allow it
    if (T.tp >= 1) { const d = drag; drag = null; rip(d.vx, d.vy); return; }
    wake();
  });
  const release = e => {
    if (!drag || e.pointerId !== drag.id) return;
    const d = drag; drag = null;
    const speed = Math.hypot(d.vx, d.vy);
    if (T.p > 0.55 || (speed > FLICK && e.timeStamp - d.lt < 90 && d.vy < 0 && T.p > 0.04)) { rip(d.vx, d.vy); return; }
    T.tp = 0; T.k = 900; T.c = 60; wake();       // let go early: it snaps back in under 200ms
  };
  paper.addEventListener('pointerup', release);
  paper.addEventListener('pointercancel', release);

  // the button is the same three steps, so keyboard and touch can do it all
  function autoTear() {
    if (reduce) { T.side = 1; setTear(1); rip(0, 0); T.x = T.tx; T.y = T.ty; T.r = T.tr; T.vx = T.vy = T.vr = 0; place(); return; }
    if (walking) return;
    T.side = 1; T.tp = 0; T.k = 70; T.c = 12;
    walking = true;
    const start = performance.now();
    (function walk(now) {
      if (state !== 'ready') { walking = false; return; }
      const u = Math.max(0, Math.min(1, (now - start) / 420));
      setTear(u * u * (3 - 2 * u)); place();
