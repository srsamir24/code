  12: { v: [30000, 32100, 34900, 30900, 36700, 41000, 43100, 45900, 42300, 38800, 45000, 48300], label: i => new Date(Date.UTC(2025, 10 + i, 1)).toLocaleDateString('en-GB', { timeZone: 'UTC', month: 'long', year: 'numeric' }), short: i => new Date(Date.UTC(2025, 10 + i, 1)).toLocaleDateString('en-GB', { timeZone: 'UTC', month: 'short' }), pct: 31.7, down: false, vs: 'vs the year before', avg: 36.9, ret: 38 },
};
const sum = a => a.reduce((x, y) => x + y, 0);
const money = n => n.toLocaleString('en-US');
const DIGITS = Array.from({ length: 10 }, (_, d) => `<i>${d}</i>`).join('');

/* A roller right-aligns the value onto a template, so the comma keeps its column. */
function roller(el, template) {
  const cells = [...template].map(ch => {
    const sl = document.createElement('span');
    sl.className = 'sl';
    if (ch === '#') { sl.innerHTML = `<span class="st">${DIGITS}</span>`; sl.dataset.d = 1; }
    else { sl.textContent = ch; }
    el.append(sl);
    return sl;
  });
  return {
    cells,
    set(text) {
      const pad = template.length - text.length;
      cells.forEach((sl, k) => {
        const ch = text[k - pad];
        const off = ch === undefined;
        const appearing = sl.classList.contains('off') && !off;
        sl.classList.toggle('off', off);
        if (sl.dataset.d && !off) {
          const st = sl.firstChild;
          // a column that opens up starts on its digit, so it never rolls through a leading zero
          if (appearing) { st.style.transition = 'none'; st.style.transform = `translateY(${-Number(ch) * 10}%)`; st.offsetWidth; st.style.transition = ''; }
          st.style.setProperty('--d', (cells.length - k) * 28 + 'ms');
          st.style.transform = `translateY(${-Number(ch) * 10}%)`;
        }
      });
    },
  };
}
const rollers = { total: roller($('#total'), '###,###'), orders: roller($('#orders'), '##,###'), avg: roller($('#avg'), '##.##'), ret: roller($('#ret'), '##') };

/* Column widths depend on the font, so they are measured once it has loaded. */
function measureColumns() {
  Object.values(rollers).forEach(r => r.cells.forEach(sl => {
    const probe = document.createElement('span');
    probe.style.cssText = 'position:absolute;visibility:hidden;white-space:pre';
    probe.textContent = sl.dataset.d ? '0' : sl.textContent;
    sl.parentElement.append(probe);
    sl.style.setProperty('--w', probe.getBoundingClientRect().width + 'px');
    probe.remove();
  }));
}

/* chart */
const chart = $('#chart'), tip = $('#tip'), box = $('#box');
let period = 30, W = 0, H = 0, xs = [], ys = [], targetY = [], fromY = [], t0 = 0, raf = 0, hoverI = -1;
const TOP = 50, BOTTOM = 10, SIDE = 8;

const xAt = (i, n) => SIDE + (W - SIDE * 2) * i / (n - 1);
function yFor(values) {
  const hi = Math.max(...values), lo = Math.min(...values) * 0.72;
  return values.map(v => TOP + (H - TOP - BOTTOM) * (1 - (v - lo) / (hi - lo)));
}
// a smooth line that never overshoots its points
function linePath(px, py) {
  const n = px.length, m = [];
  for (let i = 0; i < n; i++) {
    const a = i ? (py[i] - py[i - 1]) / (px[i] - px[i - 1]) : 0;
    const b = i < n - 1 ? (py[i + 1] - py[i]) / (px[i + 1] - px[i]) : 0;
    m[i] = !i ? b : i === n - 1 ? a : a * b <= 0 ? 0 : 2 * a * b / (a + b);
  }
  let d = `M${px[0]} ${py[0]}`;
  for (let i = 1; i < n; i++) {
    const h = (px[i] - px[i - 1]) / 3;
    d += `C${px[i - 1] + h} ${py[i - 1] + m[i - 1] * h} ${px[i] - h} ${py[i] - m[i] * h} ${px[i]} ${py[i]}`;
  }
  return d;
}
function paint() {
  const d = linePath(xs, ys);
  $('#ln').setAttribute('d', d);
  $('#area').setAttribute('d', `${d}L${xs[xs.length - 1]} ${H - BOTTOM}L${xs[0]} ${H - BOTTOM}Z`);
  const last = xs.length - 1;
  $('#now').setAttribute('cx', xs[last]);
  $('#now').setAttribute('cy', ys[last]);
}
const sampleY = (oldX, oldY, x) => {
  if (x <= oldX[0]) return oldY[0];
  for (let i = 1; i < oldX.length; i++) if (x <= oldX[i]) {
    const u = (x - oldX[i - 1]) / (oldX[i] - oldX[i - 1]);
    return oldY[i - 1] + (oldY[i] - oldY[i - 1]) * u;
  }
  return oldY[oldY.length - 1];
};
const easeOut = u => 1 - Math.pow(1 - u, 3);
// the old line is resampled onto the new points, then each point travels, left to right
function redraw(values) {
  const n = values.length, oldX = xs, oldY = ys;
  xs = values.map((_, i) => xAt(i, n));
  targetY = yFor(values);
  fromY = xs.map(x => sampleY(oldX, oldY, x));
  ys = fromY.slice();
  t0 = performance.now();
  cancelAnimationFrame(raf);
  if (calm) { ys = targetY.slice(); paint(); return; }
  const step = () => {
    const now = performance.now();
    let busy = false;
    ys = xs.map((x, i) => {
      const u = Math.min(1, Math.max(0, (now - t0 - (x / W) * 240) / 560));
      if (u < 1) busy = true;
      return fromY[i] + (targetY[i] - fromY[i]) * easeOut(u);
    });
    paint();
    if (busy) raf = requestAnimationFrame(step);
  };
  raf = requestAnimationFrame(step);
}
function layout() {
  W = chart.clientWidth; H = chart.clientHeight;
  $('#svg').setAttribute('viewBox', `0 0 ${W} ${H}`);
  [TOP, (TOP + H - BOTTOM) / 2, H - BOTTOM].forEach((y, i) => {
    const g = $('#g' + (i + 1));
    g.setAttribute('x1', 0); g.setAttribute('x2', W); g.setAttribute('y1', y); g.setAttribute('y2', y);
  });
  const v = DATA[period].v;
  xs = v.map((_, i) => xAt(i, v.length));
  ys = targetY = yFor(v);
  paint();
}

/* tooltip on a real data point */
function showPoint(i) {
  const d = DATA[period];
  hoverI = i;
  const x = xs[i], y = targetY[i];
  $('#hair').setAttribute('x1', x); $('#hair').setAttribute('x2', x); $('#hair').setAttribute('y1', y); $('#hair').setAttribute('y2', H - BOTTOM);
  $('#spot').setAttribute('cx', x); $('#spot').setAttribute('cy', y);
  chart.classList.add('on');
  tip.querySelector('small').textContent = d.label(i);
  tip.querySelector('b').textContent = '$' + money(d.v[i]);
  const tw = tip.offsetWidth, th = tip.offsetHeight;
  const left = Math.max(0, Math.min(W - tw, x - tw / 2));
  // always above the point: TOP leaves room for it even on the highest one
  tip.style.transform = `translate(${left}px, ${y - th - 10}px)`;
  tip.classList.add('on');
}
function hidePoint() { hoverI = -1; chart.classList.remove('on'); tip.classList.remove('on'); }
chart.addEventListener('pointermove', e => {
  const n = DATA[period].v.length, r = chart.getBoundingClientRect();
  const i = Math.max(0, Math.min(n - 1, Math.round((e.clientX - r.left - SIDE) / (W - SIDE * 2) * (n - 1))));
  if (i !== hoverI) showPoint(i);
});
chart.addEventListener('pointerleave', hidePoint);
chart.addEventListener('keydown', e => {
  const d = { ArrowRight: 1, ArrowLeft: -1 }[e.key];
  if (!d) return;
  e.preventDefault();
  const n = DATA[period].v.length;
  showPoint(hoverI < 0 ? (d > 0 ? 0 : n - 1) : (hoverI + d + n) % n);
});
chart.addEventListener('blur', hidePoint);

/* period */
function fillStats(p) {
  const d = DATA[p], total = sum(d.v);
  rollers.total.set(money(total));
  rollers.orders.set(money(Math.round(total / d.avg)));
  rollers.avg.set(d.avg.toFixed(2));
  rollers.ret.set(String(d.ret));
  $('#pct').textContent = `${d.pct.toFixed(1)}%`;
  $('#pill').classList.toggle('down', d.down);
  $('#pill').setAttribute('aria-label', `${d.down ? 'Down' : 'Up'} ${d.pct.toFixed(1)} percent`);
  $('#vs').textContent = d.vs;
  const labs = $('#xlabs').children, n = d.v.length;
  labs[0].textContent = d.short(0);
  labs[1].textContent = d.short(Math.floor((n - 1) / 2));
  labs[2].textContent = d.short(n - 1);
  const order = ['7', '30', '12'].indexOf(String(p));
  $('#pick').style.transform = `translateX(${order * 59.33}px)`;
  document.querySelectorAll('#periods button').forEach(b => b.setAttribute('aria-pressed', b.dataset.p === String(p)));
}
function setPeriod(p) {
  if (p === period) return;
  period = p;
  hidePoint();
  fillStats(p);
  redraw(DATA[p].v);
  if (!calm) $('#xlabs').animate([{ opacity: 0 }, { opacity: 1 }], { duration: 260, easing: 'cubic-bezier(0.23, 1, 0.32, 1)' });
}
document.querySelectorAll('#periods button').forEach(b => b.addEventListener('click', () => setPeriod(Number(b.dataset.p))));
