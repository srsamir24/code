const PAGES = {
  'Today': ['3 things due', [['Send the launch brief', 'Due 11:00'], ['Review checkout copy', 'Due 14:30'], ['Approve the new icons', 'Due 17:00'], ['Client call moved', 'Now 13:30'], ['Pay Halden Studio', 'Due Friday']]],
  'Inbox': ['12 unread', [['Priya Raman', 'The pricing page is ready'], ['Tom Becker', 'Can we move Thursday?'], ['Ana Costa', 'Final logo files'], ['Jonas Weber', 'Contract signed'], ['Lea Hart', 'Two quick questions']]],
  'Spring launch': ['Ships in 9 days', [['Landing page', 'In review'], ['Launch email', 'In progress'], ['Press kit', 'Not started'], ['Pricing update', 'In review'], ['Demo video', 'In progress']]],
  'Mobile app': ['Version 2.4', [['Onboarding flow', 'In review'], ['Push settings', 'In progress'], ['Offline mode', 'Blocked'], ['Dark mode', 'Done'], ['App Store text', 'In review']]],
  'Brand refresh': ['Phase 2 of 3', [['Colour tokens', 'Done'], ['Type scale', 'In review'], ['Icon set', 'In progress'], ['Photo style', 'Not started'], ['Slide template', 'Not started']]],
  'Calendar': ['Wednesday', [['Design review', '10:00'], ['Client call', '13:00'], ['Team demo', '15:30'], ['Budget check', '16:30'], ['Gym', '18:00']]],
  'Invoices': ['2 overdue', [['Halden Studio', '4,200 overdue'], ['Oak & Ivy', '1,850 overdue'], ['Lumen Labs', '6,400 paid'], ['North Pier', '2,300 paid'], ['Fern Works', '980 paid']]],
};
const $ = s => document.querySelector(s);
const app = $('#app'), nav = $('#nav'), mark = $('#mark'), page = $('#page'), tip = $('#tip');
const projects = $('#projects'), kids = $('#kids'), fold = $('#fold');
const items = [...nav.querySelectorAll('.item[data-page]')];
let current = items[0];

const isKid = el => el.classList.contains('kid');
const shown = el => (isKid(el) && (!kids.classList.contains('open') || app.classList.contains('rail'))) ? projects : el;

function placeMark() {
  const el = shown(current);
  projects.classList.toggle('held', el === projects);
  const a = el.getBoundingClientRect(), b = nav.getBoundingClientRect();
  mark.style.transform = `translateY(${a.top - b.top}px)`;
  mark.style.height = a.height + 'px';
}
function render(name, first) {
  const [sub, rows] = PAGES[name];
  page.innerHTML = `<div class="${first ? '' : 'in'}"><h2>${name}</h2><small>${sub}</small><ul>${rows.map(([a, b]) => `<li>${a}<span>${b}</span></li>`).join('')}</ul></div>`;
}
function go(el) {
  current.removeAttribute('aria-current');
  current = el;
  el.setAttribute('aria-current', 'page');
  placeMark();
  render(el.dataset.page);
}
function toggleKids(open) {
  kids.classList.toggle('open', open);
  projects.setAttribute('aria-expanded', open);
  kids.querySelectorAll('.kid').forEach(k => k.tabIndex = open ? 0 : -1);
  follow(360);
}
function follow(ms) {
  const end = performance.now() + ms;
  const tick = () => { placeMark(); if (performance.now() < end) requestAnimationFrame(tick); };
  nav.classList.add('solo');
  tick();
  clearTimeout(follow.done);
  follow.done = setTimeout(() => { nav.classList.remove('solo'); }, ms + 20);
}
items.forEach(el => el.addEventListener('click', () => go(el)));
projects.addEventListener('click', () => {
  if (app.classList.contains('rail')) setRail(false);
  toggleKids(!kids.classList.contains('open'));
});
function setRail(on) {
  app.classList.add('toggled');
  app.classList.toggle('rail', on);
  fold.setAttribute('aria-label', on ? 'Expand sidebar' : 'Collapse sidebar');
  fold.setAttribute('aria-expanded', !on);
  tip.classList.remove('on');
  follow(460);
}
fold.addEventListener('click', () => setRail(!app.classList.contains('rail')));

// names come back as tooltips when the rail is closed
nav.addEventListener('pointerover', e => {
  const item = e.target.closest('.item');
  if (!item || !app.classList.contains('rail')) { tip.classList.remove('on'); return; }
  const a = item.getBoundingClientRect(), s = app.getBoundingClientRect();
  tip.textContent = item.querySelector('.name').textContent;
  tip.style.top = a.top - s.top + 8 + 'px';
  tip.classList.add('on');
});
nav.addEventListener('pointerleave', () => tip.classList.remove('on'));

render('Today', true);
document.fonts.ready.then(placeMark);
</script>
