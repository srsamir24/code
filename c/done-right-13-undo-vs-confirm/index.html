<script id="component">
  for (const root of document.querySelectorAll('.files')) {
    const list = root.querySelector('.files-list');
    const title = root.querySelector('.files-title');
    const live = root.querySelector('.files-live');
    const toast = root.querySelector('.toast');
    const toastText = toast.querySelector('.toast-text');
    const dialog = root.querySelector('.confirm');
    const ms = Number(root.dataset.undoMs);
    const nameOf = (row) => row.dataset.name;
    let last = null;          // the archived row + where it was
    let target = null;        // the row the dialog is about

    // Focus: the next file, else the previous, else the heading.
    const focusAfter = (row) => {
      const rows = [...list.children];
      const i = rows.indexOf(row);
      const next = rows[i + 1] || rows[i - 1];
      (next ? next.querySelector('[data-act="archive"]') : title).focus();
    };

    // The undo window waits while the toast is hovered or focused.
    let timer = null, left = 0, since = 0;
    let hover = false, inside = false;
    const tick = () => {
      const hold = hover || inside;
      if (hold && timer) {
        clearTimeout(timer);
        timer = null;
        left -= Date.now() - since;
      } else if (!hold && !timer && !toast.hidden) {
        since = Date.now();
        timer = setTimeout(commit, left);
      }
    };
    const hide = () => {
      clearTimeout(timer);
      timer = null;
      hover = inside = false;
      toast.hidden = true;
    };
    const commit = () => {
      hide();
      if (last) root.dispatchEvent(
        new CustomEvent('archive', { detail: nameOf(last.row) }));
      last = null;
    };
    toast.addEventListener('mouseenter', () => { hover = true; tick(); });
    toast.addEventListener('mouseleave', () => { hover = false; tick(); });
    toast.addEventListener('focusin', () => { inside = true; tick(); });
    toast.addEventListener('focusout', () => { inside = false; tick(); });

    // Reversible: do it now, offer Undo.
    const archive = (row) => {
      const name = nameOf(row);
      if (last) commit();
      last = { row, index: [...list.children].indexOf(row) };
      focusAfter(row);
      row.remove();
      toastText.textContent = `Archived “${name}”`;
      live.textContent = `Archived ${name}. Press Control Z to undo.`;
      toast.hidden = false;
      left = ms;
      tick();
    };
    const undo = () => {
      if (!last) return;
      const { row, index } = last;
      last = null;
      list.insertBefore(row, list.children[index] || null);
      row.querySelector('[data-act="archive"]').focus();
      hide();
      live.textContent = `Restored ${nameOf(row)}.`;
    };
    toast.querySelector('.toast-undo').addEventListener('click', undo);
    document.addEventListener('keydown', (e) => {
      if (!last || !(e.ctrlKey || e.metaKey) || e.key.toLowerCase() !== 'z') return;
      if (e.target.closest('input, textarea, [contenteditable]')) return;
      e.preventDefault();
      undo();
    });

    // Irreversible: ask first, with the real verb.
    list.addEventListener('click', (e) => {
      const btn = e.target.closest('[data-act]');
      if (!btn) return;
      const row = btn.closest('.file');
      if (btn.dataset.act === 'archive') return archive(row);
      target = row;
      dialog.querySelector('.confirm-name').textContent = nameOf(row);
      dialog.returnValue = '';
      dialog.showModal();
    });
    // "Keep file" and Esc change nothing. Only "delete" deletes.
    dialog.addEventListener('close', () => {
      const row = target;
      target = null;
      if (dialog.returnValue !== 'delete') return;
      if (last) commit();
      focusAfter(row);
      row.remove();
      root.dispatchEvent(new CustomEvent('delete', { detail: nameOf(row) }));
      live.textContent = `Deleted ${nameOf(row)} forever.`;
    });
  }
</script>
