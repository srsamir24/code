<script>/* inlined from ../_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};

<script>
(() => {
  const $ = id => document.getElementById(id);
  const reduce = matchMedia('(prefers-reduced-motion: reduce)');
  const card = $('card'), cv = $('sw'), ctx = cv.getContext('2d');

  // ---------- two faces of the same card: light underneath, dark revealed from the switch ----------
  for (const [id, word, sub] of [['faceL', 'Light', 'theme · paper white'], ['faceD', 'Dark', 'theme · lamp off']]) {
    const f = $(id); f.append($('tpl').content.cloneNode(true));
    f.querySelector('.word').textContent = word; f.querySelector('.sub').textContent = sub;
  }

  // switch geometry (canvas units, 320 x 440; centre of the rocker)
  const OX = 160, OY = 234, PW = 236, PH = 344, CW = 152, CH = 264, RW = 136, L = 120;
  const SW_X = 440 + OX, SW_Y = 96 + OY;            // rocker centre in card units
  const TILT = 0.2, CONC = 24, DOME = 6, H0 = -2, OBL = 0.2, OBLX = 0.16;  // max tilt (rad), concave dish depth, pivot height, view obliqueness
  const LX = -0.35, LY = -0.75, LZ = 0.56;           // the room light: from the top left
  const A = 70, PUSH = 1.7, DAMP = 6, REST = 0.3;

  let x = 1, v = 0, push = 0, held = 0, heldT = 0, keyT = 0;   // x = +1: top half pressed (light), -1: bottom pressed (dark)
  let dark = false, from = 0, rev = 1, nud = 0, nudV = 0;   // rev: 0..1 reveal of the current theme over the previous one
  const REV_T = 0.5, R_MAX = 720;

  // ---------- input ----------
  const local = e => { const r = card.getBoundingClientRect(), k = 800 / r.width; return { x: (e.clientX - r.left) * k, y: (e.clientY - r.top) * k }; };
  const rock = $('rock');
  rock.addEventListener('pointerdown', e => {
    e.preventDefault(); rock.setPointerCapture(e.pointerId);
    held = local(e).y < SW_Y ? 1 : -1;               // top half pushes towards light, bottom towards dark
  });
  const up = () => { held = 0; };
  rock.addEventListener('pointerup', up);
  rock.addEventListener('pointercancel', up);
  rock.addEventListener('click', e => { if (e.detail === 0) { held = 0; keyT = 0.16; } });   // keyboard: Space / Enter
  function setTheme(d) {
    if (d === dark) return;
    from = dark ? 1 : 0;
    dark = d; rev = reduce.matches ? 1 : 0;
    $('faceL').style.zIndex = d ? 0 : 1; $('faceD').style.zIndex = d ? 1 : 0;
    rock.setAttribute('aria-checked', d);
    $('live').textContent = d ? 'Dark theme' : 'Light theme';
  }

  // ---------- physics ----------
  function step(dt) {
    if (keyT > 0) { keyT -= dt; push = x > 0 ? -1 : 1; if (keyT <= 0) push = 0; }
    else { heldT = held ? heldT + dt : 0; push = held * Math.min(1, 0.35 + heldT / 0.12); }   // a finger leans in, it doesn't hit
    const f = Math.tanh(5 * x) * (1 - 0.45 * x * x);     // detent: pushes away from the middle, holds it on the stop
    v += ((f + push * PUSH) * A - v * DAMP) * dt;
    x += v * dt;
    if (Math.abs(x) > 1) {                                // end stop: a little bounce, the card feels the click
      const hit = Math.abs(v); x = Math.sign(x); v *= -REST;
      if (hit > 2 && !reduce.matches) nudV += hit * 9;
    }
    if ((x < 0) !== dark) setTheme(x < 0);
    if (rev < 1) rev = Math.min(1, rev + dt / REV_T);
    nudV += (-nud * 1500 - nudV * 26) * dt; nud += nudV * dt;
  }

  // ---------- render ----------
  const mix = (a, b, t) => a.map((c, i) => c + (b[i] - c) * t);
  const rgb = c => `rgb(${c.map(Math.round).join(',')})`;
  const clamp = (u, a = 0, b = 1) => Math.min(b, Math.max(a, u));
  const face = document.createElement('canvas'); face.width = 640; face.height = 880;
  const fx = face.getContext('2d');
  const rr = (c, x0, y0, w, h, r) => { c.beginPath(); c.roundRect(x0, y0, w, h, r); };

  function sun(c, cx, cy, s) {
    c.beginPath(); c.arc(cx, cy, 9 * s, 0, 7); c.fill();
    for (let i = 0; i < 8; i++) { const a = i * Math.PI / 4;
      c.beginPath(); c.moveTo(cx + Math.cos(a) * 15 * s, cy + Math.sin(a) * 15 * s); c.lineTo(cx + Math.cos(a) * 21 * s, cy + Math.sin(a) * 21 * s); c.stroke(); }
  }
  const mc = document.createElement('canvas'); mc.width = mc.height = 128;
  const mx = mc.getContext('2d');
  function moon(c, cx, cy, s) {                       // a crescent: one disc, minus a smaller one
    mx.clearRect(0, 0, 128, 128); mx.globalCompositeOperation = 'source-over';
    mx.fillStyle = c.fillStyle; mx.beginPath(); mx.arc(64, 64, 34, 0, 7); mx.fill();
    mx.globalCompositeOperation = 'destination-out';
    mx.beginPath(); mx.arc(84, 50, 28, 0, 7); mx.fill();
    c.drawImage(mc, cx - 32 * s, cy - 32 * s, 64 * s, 64 * s);
  }

  function render() {
    const th = x * TILT, sn = Math.sin(th), cs = Math.cos(th);
    // reveal over the whole card
    const r = (ease(rev) * R_MAX).toFixed(1);
    const top = dark ? $('faceD') : $('faceL');
    top.style.clipPath = rev >= 1 ? 'none' : `circle(${r}px at ${SW_X}px ${SW_Y}px)`;
    (dark ? $('faceL') : $('faceD')).style.clipPath = 'none';
    $('nudge').style.transform = `translateY(${nud.toFixed(3)}px)`;
    for (const h of document.querySelectorAll('.hint')) h.textContent = held ? 'let it snap' : 'press the raised side';

    ctx.setTransform(2, 0, 0, 2, 0, 0); ctx.clearRect(0, 0, 320, 440);
    // the switch is wiped by the same circle as the card: old theme underneath, new theme inside the circle
    const D = dark ? 1 : 0;
    if (rev >= 1) drawSwitch(D, th, sn, cs);
    else {
      drawSwitch(from, th, sn, cs);
      ctx.save(); ctx.beginPath(); ctx.arc(OX, OY, +r, 0, 7); ctx.clip();
      drawSwitch(D, th, sn, cs);
      ctx.restore();
    }
  }
  function drawSwitch(dk, th, sn, cs) {
    const hv = vv => { const a = Math.sqrt(vv * vv + 36), u = (a - L / 2) / (L / 2);   // two shallow, slightly domed faces in a V
      return H0 + CONC * a / L + DOME * (1 - u * u) + vv * sn; };
    const h = hv;                                                      // height of the face above the plate
    const sy = (vv, hh) => OY + vv * cs - hh * OBL;                    // screen y (higher = shifted up a touch)
    const plate = mix([240, 236, 229], [56, 52, 47], dk), plateHi = mix([255, 255, 253], [96, 90, 82], dk);
    const plateLo = mix([214, 207, 196], [24, 22, 20], dk);

    // wall plate, lit from the top left
    ctx.save();
    ctx.shadowColor = `rgba(40,28,12,${(0.22 + 0.2 * dk).toFixed(3)})`; ctx.shadowBlur = 26; ctx.shadowOffsetX = 7; ctx.shadowOffsetY = 14;
    let g = ctx.createLinearGradient(OX - PW / 2, OY - PH / 2, OX + PW / 2, OY + PH / 2);
    g.addColorStop(0, rgb(plateHi)); g.addColorStop(0.5, rgb(plate)); g.addColorStop(1, rgb(plateLo));
    ctx.fillStyle = g; rr(ctx, OX - PW / 2, OY - PH / 2, PW, PH, 46); ctx.fill();
    ctx.restore();
    ctx.lineWidth = 1.5; ctx.strokeStyle = `rgba(255,255,255,${(0.75 - 0.5 * dk).toFixed(3)})`;
    rr(ctx, OX - PW / 2 + 1.5, OY - PH / 2 + 1.5, PW - 3, PH - 3, 45); ctx.stroke();

    // the raised end throws its shadow onto the plate, down and to the right
    ctx.save();
    rr(ctx, OX - PW / 2, OY - PH / 2, PW, PH, 46); ctx.clip();
    ctx.filter = 'blur(7px)';
    ctx.fillStyle = `rgba(30,18,6,${(0.30 + 0.15 * dk).toFixed(3)})`;
    ctx.beginPath();
    const N = 40, ptsR = [], ptsL = [];
    for (let i = 0; i <= N; i++) {
      const vv = -L + 2 * L * i / N, hh = Math.max(0, h(vv)) * 1.15;
      ptsR.push([OX + RW / 2 + hh * 0.49, sy(vv, 0) + hh * 0.83]); ptsL.push([OX - RW / 2 + hh * 0.49, sy(vv, 0) + hh * 0.83]);
    }
    [...ptsR, ...ptsL.reverse()].forEach(([px, py], i) => i ? ctx.lineTo(px, py) : ctx.moveTo(px, py));
    ctx.closePath(); ctx.fill();
    ctx.restore();

    // the cavity: dark floor, walls lit where they face the light
    const cx0 = OX - CW / 2, cy0 = OY - CH / 2;
    ctx.save();
    rr(ctx, cx0, cy0, CW, CH, 24); ctx.fillStyle = rgb(mix([46, 40, 34], [12, 11, 10], dk)); ctx.fill();
    ctx.clip();
    // inner lip: bottom and right walls face the light; how much shows depends on how deep the rocker end sits there
    const botDepth = clamp((12 - h(L)) / 40), topDepth = clamp((12 - h(-L)) / 40);
    g = ctx.createLinearGradient(0, cy0 + CH - 22, 0, cy0 + CH);
    g.addColorStop(0, 'rgba(255,244,228,0)'); g.addColorStop(1, `rgba(255,244,228,${(0.10 + 0.55 * botDepth * (1 - 0.6 * dk)).toFixed(3)})`);
    ctx.fillStyle = g; ctx.fillRect(cx0, cy0 + CH - 22, CW, 22);
    g = ctx.createLinearGradient(0, cy0, 0, cy0 + 26);
    g.addColorStop(0, `rgba(0,0,0,${(0.25 + 0.45 * topDepth).toFixed(3)})`); g.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = g; ctx.fillRect(cx0, cy0, CW, 26);
    g = ctx.createLinearGradient(cx0 + CW - 10, 0, cx0 + CW, 0);
    g.addColorStop(0, 'rgba(255,244,228,0)'); g.addColorStop(1, `rgba(255,244,228,${(0.22 * (1 - 0.6 * dk)).toFixed(3)})`);
    ctx.fillStyle = g; ctx.fillRect(cx0 + CW - 10, cy0, 10, CH);
    ctx.restore();

    // ---- the rocker, drawn into its own layer row by row ----
    fx.setTransform(2, 0, 0, 2, 0, 0); fx.clearRect(0, 0, 320, 440);
    const body = mix([246, 242, 236], [118, 110, 100], dk), side = mix([178, 170, 158], [40, 37, 33], dk);
    const xl = OX - RW / 2;
    // its body first (seen where the face lifts away from it), then the face
    for (let vv = -L; vv <= L; vv += 0.5) {
      const hh = h(vv), y0 = sy(vv, hh), y1 = sy(vv, -26);
      fx.fillStyle = rgb(side); fx.fillRect(xl - hh * OBLX, y0, RW + (hh + 26) * OBLX, Math.max(0.5, y1 - y0));
    }
    for (let vv = -L; vv <= L; vv += 0.5) {
      const s = (hv(vv + 0.25) - hv(vv - 0.25)) / 0.5;   // slope along the rocker (+ = rising towards the bottom)
      const n = 1 / Math.hypot(1, s);
      const diff = clamp((-LY * s + LZ) * n * 1.0);
      const gloss = Math.exp(-(((s - 0.03) / 0.11) ** 2)) * (0.20 - 0.08 * dk);   // the window's reflection: where the face is level
      const k = 0.36 + 0.72 * diff;
      const c = body.map(ch => Math.min(255, ch * k + 255 * gloss));
      fx.fillStyle = rgb(c);
      fx.fillRect(xl - h(vv) * OBLX, sy(vv, h(vv)) - 0.3, RW, 1.1);
    }
    // rounded long edges: left catches the light, right falls away
    fx.globalCompositeOperation = 'source-atop';
    g = fx.createLinearGradient(xl, 0, xl + RW, 0);
    g.addColorStop(0, 'rgba(255,255,255,.35)'); g.addColorStop(0.07, 'rgba(255,255,255,0)');
    g.addColorStop(0.88, 'rgba(0,0,0,0)'); g.addColorStop(1, `rgba(0,0,0,${(0.22 + 0.15 * dk).toFixed(3)})`);
    fx.fillStyle = g; fx.fillRect(xl, 0, RW, 440);
    // printed icons, on the face (they ride the tilt)
    const lightOn = clamp((x + 1) / 2);
    const ink = on => { const g = dk > 0.5 ? [205, 197, 185] : [120, 112, 100], c = mix(g, [217, 80, 43], on * on * (3 - 2 * on));
      return `rgba(${c.map(Math.round).join(',')},${(0.6 + 0.4 * on).toFixed(3)})`; };   // follows the tilt, not the state
    fx.lineWidth = 3.2; fx.lineCap = 'round';
    fx.fillStyle = fx.strokeStyle = ink(lightOn); sun(fx, OX - h(-L * 0.52) * OBLX, sy(-L * 0.52, h(-L * 0.52)), cs * 0.95);
    fx.fillStyle = fx.strokeStyle = ink(1 - lightOn); moon(fx, OX - 3 - h(L * 0.52) * OBLX, sy(L * 0.52, h(L * 0.52)), cs * 0.95);
    fx.globalCompositeOperation = 'source-over';
    // the rocker sits inside the cavity: anything below plate level is hidden by the lip
    ctx.save();
    ctx.beginPath(); ctx.roundRect(cx0, cy0 - 10, CW, CH + 10, 24); ctx.clip();
    ctx.shadowColor = 'rgba(20,12,4,.35)'; ctx.shadowBlur = 8; ctx.shadowOffsetX = 2; ctx.shadowOffsetY = 4;
    ctx.drawImage(face, 0, 0, 320, 440);
    ctx.restore();
  }
  const ease = u => 1 - (1 - clamp(u)) ** 3;

  Reel({
    step, render,
    geom: () => {
      const r = card.getBoundingClientRect(), k = r.width / 800;
      return { left: r.left, top: r.top, k, sx: SW_X, sy: SW_Y, half: L * 0.5 };
    },
  });
  document.fonts.ready.then(render);
})();
