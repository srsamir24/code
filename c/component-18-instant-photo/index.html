      : mode === 'hang' ? 'Pull it out'
      : pct < 3 ? 'Shake it' : `Developing ${pct}%`;
    photo.setAttribute('aria-label', p >= 1 ? `Instant print: ${SCENES[sceneI].name}` : `Instant print, ${pct} percent developed`);
  }
  function draw() {
    const speed = Math.hypot(vx, vy);
    const ry = Math.max(-18, Math.min(18, vx * 0.0065)), rx = Math.max(-16, Math.min(16, -vy * 0.0065));
    photo.style.transform = `translate3d(${x}px, ${y}px, 0) rotate(${r}deg) rotateX(${rx}deg) rotateY(${ry}deg)`;
    const held = hand && mode === 'free' ? 1 : 0;
    $('lift').style.transform = `translate(${6 + held * 12 + speed * 0.002}px, ${8 + held * 18 + speed * 0.003}px)`;
    $('lift').style.opacity = mode === 'free' ? 1 : 0.55;
    $('gloss').style.transform = `translate(${(r * 7 + ry * 3).toFixed(1)}%, ${(rx * 2).toFixed(1)}%)`;
    // the part of the print still inside the camera is cut at the slot
    photo.style.clipPath = mode === 'free' ? 'none' : `inset(${Math.max(0, SLOT - y)}px -40px -80px -40px)`;
    trail.push([x, y, r]); if (trail.length > 10) trail.shift();
    ghost($('g1'), trail[Math.max(0, trail.length - 4)], speed * 0.00011);
    ghost($('g2'), trail[0], speed * 0.00006);
    if (Math.abs(p - shownP) >= 0.003 || (p >= 1 && shownP < 1)) { develop(p); shownP = p; }
    chin();
  }
  function ghost(el, t, op) {
    if (reduce || mode !== 'free') { el.style.opacity = 0; return; }
    el.style.opacity = Math.min(0.26, op);
    el.style.transform = `translate3d(${t[0]}px, ${t[1]}px, 0) rotate(${t[2]}deg)`;
  }

  function kick() { if (!loop) { last = performance.now(); loop = requestAnimationFrame(tick); } }
  function tick(now) {
    const dt = Math.min(0.032, (now - last) / 1000); last = now;
    if (mode === 'free') {
      const tx = hand ? hand.tx : HOME.x, ty = hand ? hand.ty : HOME.y;
      const k = hand ? 900 : 300, c = hand ? 52 : 26;
      const ax = k * (tx - x) - c * vx, ay = k * (ty - y) - c * vy;
      vx += ax * dt; vy += ay * dt; x += vx * dt; y += vy * dt;
      // held off-centre the print hangs from your finger and swings when you move it
      const gx = hand ? hand.gx : 0, gy = hand ? hand.gy : 0;
      const hang = hand ? Math.max(-24, Math.min(24, -gx * 0.16)) : REST_R;
      const torque = (ax * gy - ay * gx) * 0.00045;
      vr += (-170 * (r - hang) - 13 * vr + torque) * dt;
      r = Math.max(-34, Math.min(34, r + vr * dt));
      // shake: every change of speed feeds the chemistry
      const jerk = Math.hypot(ax - lastAx, ay - lastAy) * dt;
      lastAx = ax; lastAy = ay;
      if (hand && p < 1) p = Math.min(1, p + Math.min(jerk * 7.8e-6, dt * 0.55));
    }
    // film develops by itself too, only slowly
    if (mode === 'free' && p < 1) p = Math.min(1, p + dt / (reduce ? 8 : 60));
    if (p >= 1 && shownP < 1) say(`Developed: ${SCENES[sceneI].name}`);
    draw();
    const rest = mode === 'free' && !hand && Math.hypot(x - HOME.x, y - HOME.y) < 0.15 && Math.hypot(vx, vy) < 2 && Math.abs(r - REST_R) < 0.05 && Math.abs(vr) < 0.5;
    if (rest && p >= 1) { x = HOME.x; y = HOME.y; r = REST_R; draw(); loop = 0; return; }
    if (mode === 'idle' || mode === 'hang') { if (!hand) { loop = 0; return; } }
    loop = requestAnimationFrame(tick);
  }

  // ---------- shutter: flash, motor, the print feeds out ----------
  function shoot() {
    if (mode === 'feed') return;
    const old = mode !== 'idle' ? photo.cloneNode(true) : null;
    if (old) {                                 // the last print drops off the table
      old.removeAttribute('id'); old.inert = true; old.style.zIndex = 0;
      old.querySelector('canvas').getContext('2d').drawImage(film, 0, 0);
      old.style.transform = photo.style.transform; old.style.clipPath = 'none';
      scene.append(old);
      old.animate(reduce ? [{ opacity: 1 }, { opacity: 0 }]
        : [{ transform: photo.style.transform, opacity: 1 }, { transform: `translate3d(${x}px, ${innerHeight / U}px, 0) rotate(14deg)`, opacity: 1 }],
        { duration: reduce ? 200 : 320, easing: 'cubic-bezier(0.32, 0, 0.67, 0)' }).onfinish = () => old.remove();
      sceneI = (sceneI + 1) % SCENES.length;
    }
    left = left > 1 ? left - 1 : 8;
    $('count').textContent = left;
    if (!reduce) $('flash').animate([{ opacity: 0.55 }, { opacity: 0 }], { duration: 260, easing: 'cubic-bezier(0.23, 1, 0.32, 1)' });
    if (!reduce) $('flashBar').animate([{ filter: 'brightness(1.5)' }, { filter: 'brightness(1)' }], { duration: 320 });
    $('hint').classList.add('gone');
    p = 0; shownP = -1; chinPct = -1;
    loadScene(sceneI); develop(0); shownP = 0;
    x = 0; vx = vy = 0; r = 0; vr = 0; trail.length = 0;
    photo.classList.remove('free', 'done');
    photo.style.visibility = 'visible'; photo.tabIndex = 0;
    mode = 'feed'; say('Picture taken. The print is coming out.');
    $('shutter').setAttribute('aria-disabled', 'true');
    const y0 = SLOT - 292, y1 = HANG_Y;
    if (reduce) { y = y1; mode = 'hang'; draw(); $('shutter').setAttribute('aria-disabled', 'false'); kick(); return; }
    const t0 = performance.now(), D = 1250;
    const motor = now => {
      const t = Math.min(1, (now - t0) / D);
      // an uneven little motor: steps, with the camera shaking along
      y = y0 + (y1 - y0) * (t + Math.sin(t * 68) * 0.006 * (1 - t));
      camera.style.translate = t < 1 ? `${Math.sin(now * 0.11) * 0.5}px ${Math.cos(now * 0.14) * 0.6}px` : '';
      draw();
      if (t < 1) { requestAnimationFrame(motor); return; }
      mode = 'hang'; $('shutter').setAttribute('aria-disabled', 'false'); draw();
      say('Print is out. Pull it free.');
    };
    requestAnimationFrame(motor);
  }
  $('shutter').addEventListener('click', shoot);

  // pulling it out: the rollers hold, then let go
  function freeIt(v = 0) {
    if (mode !== 'hang') return;
    mode = 'free'; photo.classList.add('free');
    vy = Math.max(v, reduce ? 0 : 500); vr = -60;
    say('Free. Shake it to develop.');
    kick();
  }

  // ---------- drag ----------
  const toScene = e => {
    const b = $('fit').getBoundingClientRect();
    return [(e.clientX - b.left) / U, (e.clientY - b.top) / U];
  };
  photo.addEventListener('pointerdown', e => {
    if (hand || !e.isPrimary || e.button > 0 || mode === 'feed' || mode === 'idle') return;
    e.preventDefault(); photo.focus({ preventScroll: true });
    const [sx, sy] = toScene(e);
    hand = { id: e.pointerId, sx, sy, ox: sx - x, oy: sy - y, tx: x, ty: y, gx: sx - (81 + x + 119), gy: sy - (y + 117), t0: performance.now(), travel: 0, py: sy };
    photo.setPointerCapture(e.pointerId);
    kick();
  });
  photo.addEventListener('pointermove', e => {
    if (!hand || e.pointerId !== hand.id) return;
    const [sx, sy] = toScene(e);
    hand.travel = Math.max(hand.travel, Math.hypot(sx - hand.sx, sy - hand.sy));
    if (mode === 'hang') {
      // only downwards, and with a resistance like rollers
      const d = Math.max(0, sy - hand.sy);
      y = HANG_Y + 20 * (1 - Math.exp(-d / 40));
      draw();
      if (d > 58) { const g = { id: hand.id, sx, sy }; hand = null; freeIt(); grabFree(g.id, sx, sy); }
      return;
    }
    hand.tx = sx - hand.ox; hand.ty = sy - hand.oy;
    kick();
  });
  function grabFree(id, sx, sy) {
    hand = { id, sx, sy, ox: sx - x, oy: sy - y, tx: x, ty: y, gx: sx - (81 + x + 119), gy: sy - (y + 117), t0: performance.now(), travel: 99 };
  }
  function release(e) {
    if (!hand || e.pointerId !== hand.id) return;
    const { travel } = hand;
    hand = null;
    if (mode === 'hang') {
      if (travel < 6) { freeIt(); return; }                // a tap pulls it for you
      y = HANG_Y; draw(); return;                          // let go early: the rollers keep it
    }
    kick();
  }
  photo.addEventListener('pointerup', release);
  photo.addEventListener('pointercancel', release);
  photo.addEventListener('contextmenu', e => e.preventDefault());

  // keyboard: Enter or Space pulls, then each press is one shake
  photo.addEventListener('keydown', e => {
    if (!['Enter', ' ', 'ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key)) return;
    e.preventDefault();
    if (mode === 'hang') { freeIt(); return; }
    if (mode !== 'free') return;
    const dir = e.key === 'ArrowLeft' ? -1 : e.key === 'ArrowRight' ? 1 : (shakeSide = -shakeSide);
    vx += dir * (reduce ? 0 : 1000); vy += reduce ? 0 : -260; vr += dir * (reduce ? 0 : 150);
    if (p < 1) p = Math.min(1, p + 0.07);
    kick();
  });
  let shakeSide = 1;

  // the glint in the lens leans with a fine pointer
  if (matchMedia('(hover: hover) and (pointer: fine)').matches && !reduce) {
    const gl = $('glint');
    addEventListener('pointermove', e => {
      gl.setAttribute('transform', `translate(${((e.clientX / innerWidth - 0.5) * 12).toFixed(1)} ${((e.clientY / innerHeight - 0.5) * 12).toFixed(1)})`);
    });
  }

  // ---------- stills and demo ----------
  const params = new URLSearchParams(location.search);
  document.fonts.ready.then(() => {
    const still = params.get('still');
    $('hint').classList.toggle('gone', Boolean(still));
    if (!still) { loadScene(0); develop(0); }
    if (still) {
      loadScene(0); photo.style.visibility = 'visible'; photo.tabIndex = 0; $('count').textContent = left = 7;
    }
    if (still === 'shake') {
      mode = 'free'; photo.classList.add('free');
      p = +(params.get('p') || 0.43); x = 12; y = 166; r = -13; vx = 950; vy = -420;
      hand = { tx: x, ty: y, gx: 0, gy: 0 };
      draw(); hand = null;
      $('g1').style.opacity = 0.2; $('g1').style.transform = 'translate3d(36px, 188px, 0) rotate(-15deg)';
      $('g2').style.opacity = 0.11; $('g2').style.transform = 'translate3d(56px, 204px, 0) rotate(-17deg)';
    }
    if (still === 'done') {
      mode = 'free'; photo.classList.add('free');
      p = 1; x = HOME.x; y = HOME.y - 18; r = REST_R; draw();
    }
    if (still === 'feed') {
      mode = 'hang'; p = 0; y = HANG_Y; draw();
    }
    if (params.has('demo')) {
      const wait = ms => new Promise(res => setTimeout(res, ms));
      (async function run() {
        await wait(900); shoot();
        await wait(1800); freeIt();
        await wait(600);
        for (let i = 0; i < 22; i++) { vx += (i % 2 ? -1 : 1) * 1100; vy -= 200; vr += (i % 2 ? -1 : 1) * 180; p = Math.min(1, p + 0.05); kick(); await wait(170); }
        await wait(2600);
        run();
      })();
    }
  });
</script>
