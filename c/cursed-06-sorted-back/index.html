<script>/* inlined from ../_kit/reel.js */
/* Shared runtime for the Cursed UI pages.
   - a simulation clock (K.now, ms) advanced by K.tick(ms): rAF in the browser, stepped by the recorder (?manual)
   - tweens, easing, the hook <-> outro swap (window.__outro), and the reel cursor driven by real pointer events
   - opts.embedded: a private clock for a component mounted elsewhere (the game): no reel cursor, no globals, own rAF loop,
     K.stop() ends it. Pointer position is still tracked (K.px/K.py) */
window.ReelKit = function (opts) {
  if (opts.embedded) return ReelKit.embedded(opts);
  const qs = new URLSearchParams(location.search);
  const REEL = qs.has('reel'), MANUAL = qs.has('manual');
  if (REEL) document.body.classList.add('reel');

  const K = { REEL, MANUAL, now: 0, px: -999, py: -999, down: false };
  const tweens = [];
  K.ease = {
    lin: t => t,
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,
    back: t => { const c = 1.7; return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2); }
  };
  K.tween = (dur, fn, e = K.ease.out, delay = 0) => { const tw = { t0: K.now + delay, dur, fn, e }; tweens.push(tw); return tw; };
  K.clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  K.lerp = (a, b, t) => a + (b - a) * t;
  /* critically-ish damped spring step: z = damping ratio */
  K.spring = (s, target, k, z, dt) => { s.v += (k * (target - s.x) - 2 * Math.sqrt(k) * z * s.v) * dt; s.x += s.v * dt; };

  /* ---------- cursor ---------- */
  document.body.insertAdjacentHTML('beforeend', `<div id="cursor">
    <div id="ring"></div>
    <svg id="c-arrow" width="26" height="34" viewBox="0 0 26 34"><path d="M2 2 L2 26 L8.2 20.4 L12.4 30.4 L16.6 28.6 L12.5 18.9 L20.8 18.6 Z" fill="#111" stroke="#fff" stroke-width="2" stroke-linejoin="round"/></svg>
    <svg id="c-hand" width="30" height="34" viewBox="0 0 30 34" style="display:none;left:-9px;top:-2px"><path d="M10 15V5.2a2.6 2.6 0 0 1 5.2 0V13 M15.2 12.6a2.5 2.5 0 0 1 5 0v1.8 M20.2 13.6a2.5 2.5 0 0 1 5 0v6.2c0 6.6-4 11.2-10 11.2-4.6 0-6.7-2.2-9.4-6.5L3 19.6a2.5 2.5 0 0 1 4.2-2.7L10 20.8V15" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
    <svg id="c-grab" width="30" height="30" viewBox="0 0 30 30" style="display:none;left:-15px;top:-13px"><path d="M8.2 11.4a2.4 2.4 0 0 1 4.8 0v-1a2.4 2.4 0 0 1 4.8 0v1a2.4 2.4 0 0 1 4.8 0v1.4a2.4 2.4 0 0 1 4.6.9v4.8c0 5.6-3.8 9.4-9.4 9.4-4.2 0-6.4-1.8-8.6-5.2l-2.4-3.8a2.3 2.3 0 0 1 3.4-3z M13 11.4v4 M17.8 11.4v4 M22.6 12.8v3" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
  </div>`);
  const $ = id => document.getElementById(id);
  const cur = $('cursor'), ring = $('ring'), shapes = { arrow: $('c-arrow'), hand: $('c-hand'), grab: $('c-grab') };
  let curScale = 1;
  function moveCursor() {
    if (!REEL) return;
    const el = document.elementFromPoint(K.px, K.py);
    const kind = (opts.cursorKind && opts.cursorKind(el)) || 'arrow';
    for (const k in shapes) shapes[k].style.display = k === kind ? '' : 'none';
    cur.style.transform = `translate(${K.px}px,${K.py}px) scale(${curScale})`;
  }
  K.moveCursor = moveCursor;
  window.addEventListener('pointermove', e => { K.px = e.clientX; K.py = e.clientY; moveCursor(); });
  window.addEventListener('pointerleave', () => { K.px = K.py = -999; });
  window.addEventListener('pointerdown', () => {
    K.down = true;
    if (!REEL) return;
    K.tween(90, p => { curScale = 1 - .14 * p; moveCursor(); });
    K.tween(300, p => { curScale = .86 + .14 * p; moveCursor(); }, K.ease.back, 120);
    K.tween(480, p => { ring.style.opacity = (1 - p) * .9; ring.style.transform = `scale(${.4 + p * 1.1})`; });
  });
  window.addEventListener('pointerup', () => { K.down = false; moveCursor(); });

  /* ---------- hook <-> outro ---------- */
  let outroOn = false;
  window.__outro = on => {
    if (on === outroOn) return; outroOn = on;
    const hook = document.querySelector('.hook'), outro = document.querySelector('.outro');
    K.tween(440, p => { const q = on ? p : 1 - p;
      hook.style.opacity = K.clamp(1 - q / .45, 0, 1);   // strictly sequential: never two lines at once
      const o = K.clamp((q - .55) / .45, 0, 1);
      outro.style.opacity = o; outro.style.transform = `translateY(${(1 - o) * 10}px)`; }, K.ease.lin);
  };

  /* ---------- clock ---------- */
  function tick(ms) {
    const n = Math.max(1, Math.ceil(ms / 4));
    for (let i = 0; i < n; i++) { K.now += ms / n; opts.update(ms / n / 1000); }
    for (let i = tweens.length - 1; i >= 0; i--) {
      const tw = tweens[i];
      if (K.now < tw.t0) continue;
      const p = tw.dur <= 1 ? 1 : Math.min(1, (K.now - tw.t0) / tw.dur);
      tw.fn(tw.e(p));
      if (p >= 1) tweens.splice(i, 1);
    }
    opts.render();
  }
  window.__tick = tick;
  K.start = () => {
    if (!MANUAL) {
      let last = performance.now();
      requestAnimationFrame(function loop(t) { tick(Math.min(50, t - last)); last = t; requestAnimationFrame(loop); });
    } else tick(16);
  };
  return K;
};

ReelKit.embedded = function (opts) {
  const K = { REEL: false, MANUAL: false, embedded: true, now: 0, px: -999, py: -999, down: false };
  const tweens = [];
  K.ease = {
    lin: t => t,
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,

  K.spring = (s, target, k, z, dt) => { s.v += (k * (target - s.x) - 2 * Math.sqrt(k) * z * s.v) * dt; s.x += s.v * dt; };
  K.moveCursor = () => {};
  const mv = e => { K.px = e.clientX; K.py = e.clientY; };
  const dn = () => { K.down = true; }, up = () => { K.down = false; };
  window.addEventListener('pointermove', mv); window.addEventListener('pointerdown', dn); window.addEventListener('pointerup', up);
  function tick(ms) {
    const n = Math.max(1, Math.ceil(ms / 4));
    for (let i = 0; i < n; i++) { K.now += ms / n; opts.update(ms / n / 1000); }
    for (let i = tweens.length - 1; i >= 0; i--) {
      const tw = tweens[i];
      if (K.now < tw.t0) continue;

      tw.fn(tw.e(p));
      if (p >= 1) tweens.splice(i, 1);
    }
    opts.render();
  }
  K.tick = tick;
  let raf = 0, alive = false;
  K.start = () => {
    if (alive) return; alive = true;
    let last = performance.now();
    raf = requestAnimationFrame(function loop(t) { if (!alive) return; tick(Math.min(50, t - last)); last = t; raf = requestAnimationFrame(loop); });
  };
  K.stop = () => {
    alive = false; cancelAnimationFrame(raf);
    window.removeEventListener('pointermove', mv); window.removeEventListener('pointerdown', dn); window.removeEventListener('pointerup', up);
  };
  return K;
};

(() => {
  const $ = id => document.getElementById(id);
  const list = $('list'), sortBy = $('sortBy');
  const RH = 64;
  const ITEMS = [
    { id: 'ai', t: 'Add AI to the footer', m: 'Exec ask &middot; 8 pts', a: ['JD', '#7a6cf0'] },

  const RECOMMENDED = ITEMS.map(i => i.id);
  const SORT_DELAY = [1000, 600, 300, 60, 0, 0];   // ms after you drop it: every attempt, it gives up pretending sooner
  const SORT_MS = [600, 450, 350, 260, 220, 220];
  const USERS = [312, 640, 1204, 2391, 4870, 9958, 12480];

  const rows = {};
  for (const it of ITEMS) {
    const el = document.createElement('li');
    el.className = 'row' + (it.id === 'bug' ? ' bug' : '');
    el.innerHTML = `<span class="grip" aria-hidden="true"><svg width="10" height="16" viewBox="0 0 10 16">${[2, 8, 14].map(y => `<circle cx="2" cy="${y}" r="1.6" fill="currentColor"/><circle cx="8" cy="${y}" r="1.6" fill="currentColor"/>`).join('')}</svg></span>

    el.setAttribute('role', 'option');
    list.appendChild(el);
    rows[it.id] = { el, y: { x: 0, v: 0 }, k: 420, lift: { x: 0, v: 0 }, tw: false, ghost: false };
  }

  let order = [...RECOMMENDED], dragging = null, grabDY = 0, dragY = 0, attempt = 0, users = 0;
  const timers = [];
  const later = (ms, fn) => timers.push({ at: K.now + ms, fn });

  const K = ReelKit({
    cursorKind: el => dragging ? 'grab' : (el && el.closest('.list')) ? 'hand' : 'arrow',
    update, render
  });

  function listTop() { return list.getBoundingClientRect().top + 1; }
  function rowAt(y) { const i = Math.floor((y - listTop()) / RH); return order[i]; }

  window.addEventListener('pointerdown', e => {
    const id = rowAt(e.clientY);
    const lr = list.getBoundingClientRect();
    if (!id || e.clientX < lr.left || e.clientX > lr.right) return;
    dragging = id; dragY = rows[id].y.x; grabDY = e.clientY - dragY; K.moveCursor();
  });
  window.addEventListener('pointerup', () => {
    if (!dragging) return;
    const id = dragging; dragging = null; K.moveCursor();
    order = placed(id);
    $('footText').textContent = 'Order saved.';
    swapText(sortBy, 'You');
    const n = attempt++;
    later(SORT_DELAY[Math.min(n, SORT_DELAY.length - 1)], () => sortBack(n));
  });

  function placed(id) {
    const rest = order.filter(x => x !== id);
    const i = K.clamp(Math.round(dragY / RH), 0, rest.length);
    rest.splice(i, 0, id); return rest;
  }

  /* quietly: soft springs, no toast, the footer still says "Order saved." */
  function sortBack(n) {
    if (dragging) return;
    order = [...RECOMMENDED];
    const ms = SORT_MS[Math.min(n, SORT_MS.length - 1)];
    // every row moves at once on one eased curve; the bug row is lifted by an invisible hand
    order.forEach((id, i) => {
      const r = rows[id], y0 = r.y.x, y1 = i * RH;
      if (Math.abs(y1 - y0) < .5) return;
      r.tw = true; r.ghost = id === 'bug';
      K.tween(ms, p => { r.y.x = K.lerp(y0, y1, p); r.y.v = 0; if (p >= 1) { r.tw = false; r.ghost = false; } }, K.ease.inOut);
    });
    swapText(sortBy, 'Recommended');
    users = Math.min(users + 1, USERS.length - 1);
    const u = $('users');
    later(ms * .6, () => { u.textContent = USERS[users].toLocaleString('en-US'); K.tween(420, p => u.style.transform = `scale(${1 + .18 * (1 - p)})`); });
  }

  function swapText(el, txt) {
    if (el.textContent === txt) return;
    el.textContent = txt;                                   // one step, no dissolve
    const back = txt === 'Recommended';
    K.tween(700, p => { el.style.color = back && p < .75 ? '#c23a31' : ''; el.style.transform = `scale(${1 + .12 * (1 - p)})`; });

    for (let i = timers.length - 1; i >= 0; i--) if (K.now >= timers[i].at) timers.splice(i, 1)[0].fn();
    let view = order;
    if (dragging) {
      dragY = K.clamp(K.py - grabDY, -8, (order.length - 1) * RH + 8);
      view = placed(dragging);
    }

      if (id === dragging) { r.y.x = dragY; r.y.v = 0; }
      else if (!r.tw) K.spring(r.y, i * RH, r.k, .9, dt);
      K.spring(r.lift, id === dragging || r.ghost ? 1 : 0, 500, .7, dt);
    });

      r.el.classList.toggle('first', Math.abs(r.y.x) < 2);
      r.el.querySelector('.rank').textContent = K.clamp(Math.round(r.y.x / RH), 0, 3) + 1;   // the badge follows where the row actually is
      r.el.style.zIndex = id === dragging ? 3 : l > .05 ? 2 : 1;
    }
  }

  window.__grip = id => { const r = rows[id || 'bug'].el.querySelector('.grip').getBoundingClientRect(); return [r.left + r.width / 2, r.top + r.height / 2]; };

  };

  order.forEach((id, i) => rows[id].y.x = i * RH);
  K.start();
})();
