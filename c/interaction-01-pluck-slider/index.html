<script>
(() => {
  const params = new URLSearchParams(location.search);
  const CAPTURE = params.has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)');

  // ---------- geometry ----------
  const W = 700, X0 = 26, X1 = 674, REST = 196, MAXD = 200;
  const N = 121;                      // points on the string
  const DX = (X1 - X0) / (N - 1);
  const DT = 1 / 1440;                // fixed physics step (seconds)
  const FRICTION = 0.9993;            // per step
  const y = new Float32Array(N), v = new Float32Array(N);

  const $ = id => document.getElementById(id);
  const svg = $('svg'), str = $('str'), shadow = $('shadow'), shine = $('shine');
  const beadG = $('beadG'), beadShadow = $('beadShadow'), valEl = $('val'), hzEl = $('hz');

  let value = +(params.get('v') || 35);   // 0..100
  let held = false, pluckY = 0;

  // tension -> wave speed (px/s). More tension, faster ring.
  const speed = () => 2100 + value * 26;
  const tension = () => (speed() / DX) ** 2;          // c² / dx²
  const beadX = () => X0 + (X1 - X0) * value / 100;

  // pegs + ticks
  const NS = 'http://www.w3.org/2000/svg';
  for (const x of [X0 - 4, X1 + 4]) {
    const g = document.createElementNS(NS, 'g');
    g.innerHTML = `<circle cx="${x}" cy="${REST + 7}" r="14" fill="rgba(60,40,15,.18)" filter="url(#softer)"/>
      <circle cx="${x}" cy="${REST}" r="12" fill="url(#peg)" stroke="rgba(60,50,35,.35)" stroke-width="1"/>
      <line x1="${x - 6}" y1="${REST - 4}" x2="${x + 6}" y2="${REST + 4}" stroke="rgba(60,50,35,.45)" stroke-width="1.4" stroke-linecap="round"/>`;
    $('pegs').appendChild(g);
  }
  const ticks = [];
  for (let i = 0; i <= 50; i++) {
    const l = document.createElementNS(NS, 'line');
    const x = X0 + (X1 - X0) * i / 50;
    l.setAttribute('x1', x); l.setAttribute('x2', x);
    l.setAttribute('stroke', i % 10 === 0 ? 'rgba(40,32,20,.42)' : 'rgba(40,32,20,.20)');
    l.setAttribute('stroke-width', i % 10 === 0 ? 1.4 : 1);
    l.setAttribute('stroke-linecap', 'round');
    $('ticks').appendChild(l); ticks.push({ l, x, major: i % 10 === 0 });
  }
  for (const [t, x] of [['0', X0], ['50', (X0 + X1) / 2], ['100', X1]]) {
    const tx = document.createElementNS(NS, 'text');
    tx.setAttribute('x', x); tx.setAttribute('y', 438); tx.setAttribute('text-anchor', 'middle');
    tx.setAttribute('font-family', 'Geist Mono'); tx.setAttribute('font-size', 21); tx.setAttribute('fill', '#9A948A');
    tx.textContent = t; $('ticks').appendChild(tx);
  }

  // ---------- physics ----------
  function holdShape(seconds) {
    // finger holds the string: ease toward two straight segments from the pegs to the finger
    // (easing, not snapping, so grabbing a ringing string damps it instead of teleporting it)
    const bx = beadX(), k = 1 - Math.exp(-seconds / 0.03);
    for (let i = 0; i < N; i++) {
      const x = X0 + i * DX;
      const target = x <= bx ? pluckY * (x - X0) / (bx - X0) : pluckY * (X1 - x) / (X1 - bx);
      y[i] += (target - y[i]) * k;
      v[i] = 0;
    }
  }
  function step(seconds) {
    if (held) { holdShape(seconds); return; }
    const T = tension();
    const steps = Math.round(seconds / DT);
    if (reduce.matches) {                    // no ringing: ease straight back
      const k = Math.pow(0.82, seconds * 60);
      for (let i = 0; i < N; i++) { y[i] *= k; v[i] = 0; }
      return;
    }
    for (let s = 0; s < steps; s++) {
      for (let i = 1; i < N - 1; i++) {
        const bend = y[i - 1] - 2 * y[i] + y[i + 1];
        v[i] += bend * T * DT;
        v[i] *= FRICTION;
      }
      for (let i = 1; i < N - 1; i++) y[i] += v[i] * DT;
    }
  }
  function sampleY(x) {
    const f = (x - X0) / DX, i = Math.min(N - 2, Math.max(0, Math.floor(f))), t = f - i;
    return y[i] * (1 - t) + y[i + 1] * t;
  }

  // ---------- render ----------
  function render() {
    let d = '';
    for (let i = 0; i < N; i++) d += (i ? 'L' : 'M') + (X0 + i * DX).toFixed(2) + ' ' + (REST + y[i]).toFixed(2);
    str.setAttribute('d', d); shadow.setAttribute('d', d); shine.setAttribute('d', d);
    const bx = beadX(), by = REST + sampleY(bx);
    beadG.setAttribute('transform', `translate(${bx.toFixed(2)} ${by.toFixed(2)})`);
    beadShadow.setAttribute('cx', bx.toFixed(2)); beadShadow.setAttribute('cy', (by + 18).toFixed(2));
    for (const t of ticks) {
      const near = Math.exp(-(((t.x - bx) / 34) ** 2));
      const h = (t.major ? 16 : 9) + near * 12;
      t.l.setAttribute('y1', (410 - h).toFixed(2)); t.l.setAttribute('y2', '410');
      t.l.setAttribute('stroke-opacity', (0.6 + near * 0.4).toFixed(3));
    }
    const pct = Math.round(value);
    valEl.textContent = String(pct).padStart(2, '0');
    svg.setAttribute('aria-valuenow', pct);
    hzEl.textContent = (speed() / (2 * (X1 - X0))).toFixed(2);
  }

  // ---------- input ----------
  function local(e) {
    const r = svg.getBoundingClientRect();
    return { x: (e.clientX - r.left) * W / r.width, y: (e.clientY - r.top) * 440 / r.height };
  }
  function drag(e) {
    const p = local(e);
    value = Math.min(100, Math.max(0, (p.x - X0) / (X1 - X0) * 100));
    value = Math.min(97, Math.max(3, value));             // keep the pluck point off the pegs
    const dy = p.y - REST;
    pluckY = Math.max(-MAXD, Math.min(MAXD, dy));          // the string stays under the finger
  }
  svg.addEventListener('pointerdown', e => {
    e.preventDefault(); held = true; svg.classList.add('held');
    svg.setPointerCapture(e.pointerId); drag(e);
  });
  svg.addEventListener('pointermove', e => { if (held) drag(e); });
  const release = () => { if (!held) return; held = false; svg.classList.remove('held'); };
  svg.addEventListener('pointerup', release);
  svg.addEventListener('pointercancel', release);
  svg.addEventListener('keydown', e => {                    // keyboard: nudge + a small pluck
    const d = { ArrowRight: 1, ArrowUp: 1, ArrowLeft: -1, ArrowDown: -1 }[e.key];
    if (!d) return; e.preventDefault();
    value = Math.min(97, Math.max(3, value + d * (e.shiftKey ? 10 : 1)));
    const i = Math.round((beadX() - X0) / DX); v[i] += 900 * d;
  });

  // ---------- capture-mode cursor (screenshots have no OS pointer) ----------
  const cur = $('cursor');
  if (CAPTURE) {
    cur.style.transform = 'translate(-200px,-200px)';
    addEventListener('pointerdown', () => cur.classList.add('down'), true);
    addEventListener('pointerup', () => cur.classList.remove('down'), true);
  }

  // ---------- loop ----------
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      const whole = Math.floor(acc / DT) * DT; acc -= whole;
      step(whole); render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
  } else {
    // the recorder drives time frame by frame, so every frame is exact
    window.__tick = sec => { step(sec); render(); };
    window.__geom = () => {
      const r = svg.getBoundingClientRect();
      return { left: r.left, top: r.top, scale: r.width / W, X0, X1, REST, value };
    };
    window.__overlay = o => {
      // sequential, never two text layers at once: card out, then code in (and back)
      $('card').style.opacity = o.card;
      $('card').style.transform = `translateY(${(1 - o.card) * -18}px)`;
      $('code').style.opacity = o.code;
      $('code').style.transform = `translateY(${(1 - o.code) * 22}px)`;
      $('chip').style.opacity = o.card * (1 - o.big);
      $('end').style.opacity = o.code;
      $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
      $('tint').style.opacity = o.tint;
      $('hl').className = o.hl ? 'h' : '';
      cur.style.opacity = o.card;
      $('chipText').textContent = o.chip;
      if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
      if (o.hook !== undefined) document.querySelector('.hook').style.opacity = o.hook;
    };
  }
})();
</script>
