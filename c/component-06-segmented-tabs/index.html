<script>
  const DATA = {
    'Today': { span: 'today', labels: ['6a', '9a', '12p', '3p', '6p', '9p', '12a'], v: [12, 48, 96, 81, 120, 64, 22] },
    'Week': { span: 'this week', labels: ['M', 'T', 'W', 'T', 'F', 'S', 'S'], v: [310, 284, 342, 398, 451, 205, 168] },
    'Month': { span: 'this month', labels: ['1', '5', '10', '15', '20', '25', '30'], v: [880, 1240, 1110, 1460, 1620, 1380, 1710] },
  };
  const tabs = [...document.querySelectorAll('.tab')];
  const thumb = document.querySelector('.thumb');
  const chart = document.getElementById('chart');
  const sum = document.getElementById('sum'), span = document.getElementById('span');
  const said = document.getElementById('said');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  chart.innerHTML = Array.from({ length: 7 }, (_, n) =>
    `<div class="col" style="--n:${n}"><div class="bar"></div><span class="lbl"></span></div>`).join('');
  const cols = [...chart.children];

  // the thumb follows the tab; on the way it stretches to cover both
  function moveThumb(tab, from) {
    thumb.style.setProperty('--x', tab.offsetLeft + 'px');
    thumb.style.setProperty('--w', tab.offsetWidth + 'px');
    if (!from || from === tab || reduce) return;
    const a = Math.min(from.offsetLeft, tab.offsetLeft);
    const b = Math.max(from.offsetLeft + from.offsetWidth, tab.offsetLeft + tab.offsetWidth);
    thumb.animate([
      { transform: `translateX(${from.offsetLeft}px)`, width: from.offsetWidth + 'px' },
      { transform: `translateX(${a}px)`, width: b - a + 'px', offset: 0.4 },
      { transform: `translateX(${tab.offsetLeft}px)`, width: tab.offsetWidth + 'px' },
    ], { duration: 460, easing: 'cubic-bezier(0.16, 1, 0.3, 1)' });
  }
  function countTo(to) {
    const from = +sum.dataset.v || 0, t0 = performance.now();
    sum.dataset.v = to;
    const step = t => {
      const k = reduce ? 1 : Math.min(1, (t - t0) / 600), e = 1 - Math.pow(1 - k, 3);
      sum.textContent = Math.round(from + (to - from) * e).toLocaleString('en-US');
      if (k < 1) requestAnimationFrame(step);
    };
    requestAnimationFrame(step);
  }
  function select(tab, focus) {
    const from = tabs.find(t => t.getAttribute('aria-selected') === 'true');
    tabs.forEach(t => {
      const on = t === tab;
      t.setAttribute('aria-selected', on);
      t.tabIndex = on ? 0 : -1;
    });
    chart.setAttribute('aria-labelledby', tab.id);
    if (focus) tab.focus();
    moveThumb(tab, from);
    const d = DATA[tab.textContent], max = Math.max(...d.v);
    cols.forEach((c, i) => {
      c.style.setProperty('--v', (d.v[i] / max) * 100);
      c.classList.toggle('top', d.v[i] === max);
      c.setAttribute('role', 'img');
      c.setAttribute('aria-label', `${d.labels[i]}: ${d.v[i]}`);
    });
    chart.classList.add('swap');
    setTimeout(() => {
      cols.forEach((c, i) => c.querySelector('.lbl').textContent = d.labels[i]);
      chart.classList.remove('swap');
    }, from ? 180 : 0);
    const total = d.v.reduce((a, b) => a + b, 0);
    countTo(total);
    span.textContent = d.span;
    said.textContent = `${total.toLocaleString('en-US')} visitors ${d.span}`;
  }
  tabs.forEach(t => t.addEventListener('click', () => select(t)));
  // arrow keys move between tabs (roving tabindex)
  document.querySelector('.tabs').addEventListener('keydown', e => {
    const i = tabs.indexOf(document.activeElement);
    if (i < 0) return;
    if (e.key === 'ArrowRight') select(tabs[(i + 1) % 3], true);
    if (e.key === 'ArrowLeft') select(tabs[(i + 2) % 3], true);
    if (e.key === 'Home') select(tabs[0], true);
    if (e.key === 'End') select(tabs[2], true);
  });

  const params = new URLSearchParams(location.search);
  select(tabs[+(params.get('tab') || 1)]);
  if (params.has('demo')) {
    let i = 1;
    setInterval(() => { i = (i + 1) % 3; select(tabs[i]); }, 1800);
  }
</script>
