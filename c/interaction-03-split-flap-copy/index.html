<script>/* inlined from ../_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};

<script>
(() => {
  const $ = id => document.getElementById(id);
  const reduce = matchMedia('(prefers-reduced-motion: reduce)');
  const ABC = ' ABCDEFGHIJKLMNOPQRSTUVWXYZ';           // the drum order: a flap can only go forward
  const IDLE = 'COPY  ', DONE = 'COPIED', FAIL = 'RETRY ';
  const G = 1400, KICK = 40, BOUNCE = 0.3, STAGGER = 0.06, HOLD = 1.5;
  const PI = Math.PI;

  const btn = $('btn');
  const cells = [...IDLE].map((c) => {
    const el = document.createElement('span'); el.className = 'cell';
    el.innerHTML = '<span class="half top"><b></b><i></i></span><span class="half bot"><b></b><i></i></span>' +
                   '<span class="flap top"><b></b><i></i></span><span class="flap bot"><b></b><i></i></span>';
    el.setAttribute('aria-hidden', 'true');
    btn.appendChild(el);
    const q = s => el.querySelector(s);
    return { el, ch: ABC.indexOf(c), goal: ABC.indexOf(c), a: 0, w: 0, moving: false, wait: 0, settle: false,
             hTop: q('.half.top b'), hBot: q('.half.bot b'), fTop: q('.flap.top'), fBot: q('.flap.bot'),
             fTopB: q('.flap.top b'), fBotB: q('.flap.bot b'), shTop: q('.flap.top i'), shBot: q('.flap.bot i'), shHalf: q('.half.bot i') };
  });
  const next = i => (i + 1) % ABC.length;

  let revertIn = Infinity;
  function show(word) {
    let k = 0;                                                 // stagger only the cells that change
    [...word].forEach((c, i) => {
      const cell = cells[i], g = ABC.indexOf(c);
      if (cell.goal === g) return;
      cell.goal = g;
      if (reduce.matches) { cell.ch = g; cell.moving = false; cell.a = 0; return; }
      if (!cell.moving || cell.settle) { cell.moving = true; cell.settle = false; cell.a = 0; cell.w = KICK; cell.wait = k++ * STAGGER; }
    });
  }

  async function copy(text) {
    try { await navigator.clipboard.writeText(text); return true; } catch {}
    const ta = document.createElement('textarea');            // fallback: the old execCommand route
    ta.value = text; ta.setAttribute('readonly', ''); ta.style.cssText = 'position:fixed;opacity:0';
    document.body.appendChild(ta); ta.select();
    let ok = false; try { ok = document.execCommand('copy'); } catch {}
    ta.remove(); return ok;
  }
  btn.addEventListener('click', async () => {
    show(DONE); revertIn = HOLD;                               // flip right away, feels instant
    const ok = await copy($('snip').textContent);
    $('live').textContent = ok ? 'Copied to clipboard' : 'Copy failed, try again';
    if (!ok) show(FAIL);
  });

  // ---------- physics ----------
  function step(dt) {
    if (revertIn !== Infinity && (revertIn -= dt) <= 0) { revertIn = Infinity; show(IDLE); $('live').textContent = ''; }
    for (const c of cells) {
      if (!c.moving) continue;
      if (c.wait > 0) { c.wait -= dt; continue; }
      c.w += G * dt; c.a += c.w * dt;                          // gravity
      if (c.a < PI) continue;                                  // still falling
      if (!c.settle) {
        c.ch = next(c.ch);                                     // the flap landed: one letter on
        if (c.ch !== c.goal) { c.a = 0; c.w = KICK; continue; }// next letter
        c.settle = true;                                       // that was the last one
      }
      c.a = PI; c.w *= -BOUNCE;                                // last one bounces
      if (Math.abs(c.w) < 1.2) { c.moving = false; c.w = 0; c.a = 0; c.settle = false; }
    }
  }

  // ---------- render ----------
  function render() {
    for (const c of cells) {
      const cur = ABC[c.ch], nxt = ABC[c.settle ? c.ch : next(c.ch)];
      const flipping = c.moving && c.wait <= 0;
      if (!flipping) {
        c.hTop.textContent = c.hBot.textContent = cur;
        c.fTop.style.visibility = c.fBot.style.visibility = 'hidden';
        c.shHalf.style.background = 'none';
      } else if (c.settle) {                                   // bouncing on the final letter
        c.hTop.textContent = c.hBot.textContent = cur;
        c.fTop.style.visibility = 'hidden';
        c.fBot.style.visibility = 'visible'; c.fBotB.textContent = cur;
        const lift = PI - c.a;                                 // how far the flap bounced up
        c.fBot.style.transform = `rotateX(${(lift * 180 / PI).toFixed(2)}deg)`;
        c.shBot.style.background = `rgba(0,0,0,${(Math.sin(lift) * 0.5).toFixed(3)})`;
        c.shHalf.style.background = `rgba(0,0,0,${(Math.sin(lift) * 0.35).toFixed(3)})`;
      } else {
        c.hTop.textContent = nxt; c.hBot.textContent = cur;    // behind the moving flap
        if (c.a < PI / 2) {                                    // top flap falling: old letter
          c.fTop.style.visibility = 'visible'; c.fBot.style.visibility = 'hidden';
          c.fTopB.textContent = cur;
          c.fTop.style.transform = `rotateX(${(-c.a * 180 / PI).toFixed(2)}deg)`;
          c.shTop.style.background = `rgba(0,0,0,${(Math.sin(c.a) * 0.45).toFixed(3)})`;
          c.shHalf.style.background = `rgba(0,0,0,${(Math.sin(c.a) * 0.3).toFixed(3)})`;
        } else {                                               // its back lands: new letter, bottom half
          c.fTop.style.visibility = 'hidden'; c.fBot.style.visibility = 'visible';
          c.fBotB.textContent = nxt;
          c.fBot.style.transform = `rotateX(${((PI - c.a) * 180 / PI).toFixed(2)}deg)`;
          c.shBot.style.background = `rgba(255,255,255,${(Math.sin(c.a) * 0.06).toFixed(3)})`;
          c.shHalf.style.background = `rgba(0,0,0,${(Math.sin(c.a) * 0.3).toFixed(3)})`;
        }
      }
    }
  }

  Reel({
    step, render,
    geom: () => { const b = btn.getBoundingClientRect(); return { x: b.left + b.width / 2, y: b.top + b.height / 2, w: b.width }; },
  });
})();
</script>
