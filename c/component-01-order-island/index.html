<script>
  lucide.createIcons({ attrs: { 'stroke-width': 2.2 } });
  const island = document.getElementById('island');
  const compact = island.querySelector('.compact');
  const full = island.querySelector('.full');
  const now = island.querySelector('.steps .now');
  const num = island.querySelector('.bigeta .num');
  const etaSmall = island.querySelector('[data-eta]');
  const trail = document.getElementById('done');
  const rider = document.getElementById('rider');
  const LEN = trail.getTotalLength();
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let raf = 0, tickTimer = 0;

  // rider travels along the real path, the amber line follows it, the rest stays dashed
  // slide:js
  function ride(to, ms) {
    cancelAnimationFrame(raf);
    const t0 = performance.now();
    const step = t => {
      const k = reduce ? 1 : Math.min(1, (t - t0) / ms);
      const e = 1 - Math.pow(1 - k, 3);
      const d = LEN * to * e;
      const pt = trail.getPointAtLength(d);
      rider.setAttribute('cx', pt.x); rider.setAttribute('cy', pt.y);
      trail.style.strokeDasharray = `${d} ${LEN}`;
      if (k < 1) raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
  }
  // /slide
  function setOpen(open) {
    island.classList.toggle('open', open);
    compact.setAttribute('aria-expanded', open);
    full.setAttribute('aria-hidden', !open);
    full.querySelectorAll('button').forEach(b => b.tabIndex = open ? 0 : -1);
    now.style.setProperty('--p', open ? '62%' : '40%');
    clearTimeout(tickTimer);
    if (open) {
      num.classList.remove('tick'); etaSmall.textContent = '4';
      trail.style.strokeDasharray = `0 ${LEN}`;
      ride(0.58, 2600);
      tickTimer = setTimeout(() => { num.classList.add('tick'); etaSmall.textContent = '3'; }, 2400);
    }
  }
  compact.addEventListener('click', () => setOpen(true));
  full.querySelector('[data-close]').addEventListener('click', () => { setOpen(false); compact.focus(); });
  document.addEventListener('keydown', e => { if (e.key === 'Escape' && island.classList.contains('open')) { setOpen(false); compact.focus(); } });
  document.addEventListener('pointerdown', e => { if (!island.contains(e.target)) setOpen(false); });
  setOpen(false);

  const params = new URLSearchParams(location.search);
  // ?demo: loops open / hold / close for recording
  if (params.has('demo')) {
    const loop = () => {
      setTimeout(() => setOpen(true), 900);
      setTimeout(() => setOpen(false), 5600);
      setTimeout(loop, 6800);
    };
    loop();
  }
  if (params.has('open')) setOpen(true);
</script>
