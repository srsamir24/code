<script>/* inlined from ../_kit/reel.js */
/* Shared runtime for the Cursed UI pages.
   - a simulation clock (K.now, ms) advanced by K.tick(ms): rAF in the browser, stepped by the recorder (?manual)
   - tweens, easing, the hook <-> outro swap (window.__outro), and the reel cursor driven by real pointer events
   - opts.embedded: a private clock for a component mounted elsewhere (the game): no reel cursor, no globals, own rAF loop,
     K.stop() ends it. Pointer position is still tracked (K.px/K.py) */
window.ReelKit = function (opts) {
  if (opts.embedded) return ReelKit.embedded(opts);
  const qs = new URLSearchParams(location.search);
  const REEL = qs.has('reel'), MANUAL = qs.has('manual');
  if (REEL) document.body.classList.add('reel');

  const K = { REEL, MANUAL, now: 0, px: -999, py: -999, down: false };
  const tweens = [];
  K.ease = {
    lin: t => t,
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,
    back: t => { const c = 1.7; return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2); }
  };
  K.tween = (dur, fn, e = K.ease.out, delay = 0) => { const tw = { t0: K.now + delay, dur, fn, e }; tweens.push(tw); return tw; };
  K.clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  K.lerp = (a, b, t) => a + (b - a) * t;
  /* critically-ish damped spring step: z = damping ratio */
  K.spring = (s, target, k, z, dt) => { s.v += (k * (target - s.x) - 2 * Math.sqrt(k) * z * s.v) * dt; s.x += s.v * dt; };

  /* ---------- cursor ---------- */
  document.body.insertAdjacentHTML('beforeend', `<div id="cursor">
    <div id="ring"></div>
    <svg id="c-arrow" width="26" height="34" viewBox="0 0 26 34"><path d="M2 2 L2 26 L8.2 20.4 L12.4 30.4 L16.6 28.6 L12.5 18.9 L20.8 18.6 Z" fill="#111" stroke="#fff" stroke-width="2" stroke-linejoin="round"/></svg>
    <svg id="c-hand" width="30" height="34" viewBox="0 0 30 34" style="display:none;left:-9px;top:-2px"><path d="M10 15V5.2a2.6 2.6 0 0 1 5.2 0V13 M15.2 12.6a2.5 2.5 0 0 1 5 0v1.8 M20.2 13.6a2.5 2.5 0 0 1 5 0v6.2c0 6.6-4 11.2-10 11.2-4.6 0-6.7-2.2-9.4-6.5L3 19.6a2.5 2.5 0 0 1 4.2-2.7L10 20.8V15" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
    <svg id="c-grab" width="30" height="30" viewBox="0 0 30 30" style="display:none;left:-15px;top:-13px"><path d="M8.2 11.4a2.4 2.4 0 0 1 4.8 0v-1a2.4 2.4 0 0 1 4.8 0v1a2.4 2.4 0 0 1 4.8 0v1.4a2.4 2.4 0 0 1 4.6.9v4.8c0 5.6-3.8 9.4-9.4 9.4-4.2 0-6.4-1.8-8.6-5.2l-2.4-3.8a2.3 2.3 0 0 1 3.4-3z M13 11.4v4 M17.8 11.4v4 M22.6 12.8v3" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
  </div>`);
  const $ = id => document.getElementById(id);
  const cur = $('cursor'), ring = $('ring'), shapes = { arrow: $('c-arrow'), hand: $('c-hand'), grab: $('c-grab') };
  let curScale = 1;
  function moveCursor() {
    if (!REEL) return;
    const el = document.elementFromPoint(K.px, K.py);
    const kind = (opts.cursorKind && opts.cursorKind(el)) || 'arrow';
    for (const k in shapes) shapes[k].style.display = k === kind ? '' : 'none';
    cur.style.transform = `translate(${K.px}px,${K.py}px) scale(${curScale})`;
  }
  K.moveCursor = moveCursor;
  window.addEventListener('pointermove', e => { K.px = e.clientX; K.py = e.clientY; moveCursor(); });
  window.addEventListener('pointerleave', () => { K.px = K.py = -999; });
  window.addEventListener('pointerdown', () => {
    K.down = true;
    if (!REEL) return;
    K.tween(90, p => { curScale = 1 - .14 * p; moveCursor(); });
    K.tween(300, p => { curScale = .86 + .14 * p; moveCursor(); }, K.ease.back, 120);
    K.tween(480, p => { ring.style.opacity = (1 - p) * .9; ring.style.transform = `scale(${.4 + p * 1.1})`; });
  });
  window.addEventListener('pointerup', () => { K.down = false; moveCursor(); });

  /* ---------- hook <-> outro ---------- */
  let outroOn = false;
  window.__outro = on => {
    if (on === outroOn) return; outroOn = on;
    const hook = document.querySelector('.hook'), outro = document.querySelector('.outro');
    K.tween(440, p => { const q = on ? p : 1 - p;
      hook.style.opacity = K.clamp(1 - q / .45, 0, 1);   // strictly sequential: never two lines at once
      const o = K.clamp((q - .55) / .45, 0, 1);
      outro.style.opacity = o; outro.style.transform = `translateY(${(1 - o) * 10}px)`; }, K.ease.lin);
  };

  /* ---------- clock ---------- */
  function tick(ms) {
    const n = Math.max(1, Math.ceil(ms / 4));
    for (let i = 0; i < n; i++) { K.now += ms / n; opts.update(ms / n / 1000); }
    for (let i = tweens.length - 1; i >= 0; i--) {
      const tw = tweens[i];
      if (K.now < tw.t0) continue;
      const p = tw.dur <= 1 ? 1 : Math.min(1, (K.now - tw.t0) / tw.dur);
      tw.fn(tw.e(p));
      if (p >= 1) tweens.splice(i, 1);
    }
    opts.render();
  }
  window.__tick = tick;
  K.start = () => {
    if (!MANUAL) {
      let last = performance.now();
      requestAnimationFrame(function loop(t) { tick(Math.min(50, t - last)); last = t; requestAnimationFrame(loop); });
    } else tick(16);
  };
  return K;
};

/* Cursed UI #15: an appearance picker where Dark mode is real, but locked until your actual local sunset (19:42).
   The sun crawls along its arc and the countdown ticks. Clicking Dark bounces back to Light, and every click
   costs you: +5 s on the countdown, and the sun gets pushed back up the sky. You can grab the sun and drag it
   down to the horizon: the countdown races to zero, the whole page goes dark, Dark mode switches on, and the
   moment you let go it all floats back up.

   Self-contained level. Needs ../_kit/reel.js (ReelKit clock) and component.css.
     const api = CursedUI.sunsetDarkMode.init(el, { K, onDusk: d => ... })   // el: an empty container, e.g. a .card
     api.reset(); api.destroy();
   onDusk(d) gets 0..1 while the sun is being dragged down, so a host page can darken around the card.
   Pass { K } to drive it from an outside clock (the reel page does); otherwise it runs its own rAF loop. */
(function () {
  const CU = window.CursedUI = window.CursedUI || {};
  const SUNSET = '19:42', SUNRISE = '07:10';
  const LEFT0 = 3 * 3600 + 17 * 60 + 52;               // 16:24:08 now -> 3:17:52 to go

  const W = 368, H = 114, HY = 92, CX = W / 2, R = W / 2 - 26, RY = 66;
  const ok = (a, b, t) => `oklch(${(a[0] + (b[0] - a[0]) * t).toFixed(3)} ${(a[1] + (b[1] - a[1]) * t).toFixed(3)} ${(a[2] + (b[2] - a[2]) * t).toFixed(1)})`;
  const hex = (a, b, t) => { const p = s => [1, 3, 5].map(i => parseInt(s.slice(i, i + 2), 16)); const A = p(a), B = p(b);
    return `rgb(${A.map((v, i) => Math.round(v + (B[i] - v) * t)).join(',')})`; };
  const SKY_TOP = [[.76, .085, 238], [.27, .07, 285]], SKY_MID = [[.86, .06, 200], [.5, .12, 330]], SKY_LOW = [[.93, .06, 82], [.7, .17, 42]];
  const GROUND = [[.8, .045, 120], [.24, .03, 285]], RIDGE = [[.86, .04, 115], [.31, .045, 300]], SUN = [[.97, .1, 96], [.76, .19, 40]];
  const arc = a => [CX - R * Math.cos(Math.PI * a), HY - RY * Math.sin(Math.PI * a)];
  const LOCKED = `<svg width="9" height="10" viewBox="0 0 9 10"><rect x=".8" y="4.2" width="7.4" height="5.2" rx="1.4" fill="#fff"/><path d="M2.4 4.4V3a2.1 2.1 0 0 1 4.2 0v1.4" fill="none" stroke="#fff" stroke-width="1.3"/></svg>${SUNSET}`;
  const OPEN = `<svg width="9" height="10" viewBox="0 0 9 10"><rect x=".8" y="4.2" width="7.4" height="5.2" rx="1.4" fill="#fff"/><path d="M2.4 4.4V3a2.1 2.1 0 0 1 4.2 0" fill="none" stroke="#fff" stroke-width="1.3" transform="translate(2.6 -.6)"/></svg>Open`;
  const MSG = ['Asking early costs <b>+3 s</b>.', 'Asked again. <b>+3 s</b>. The sun heard you.', 'Unlocks at your local sunset.'];   // the last one matches frame 0

  CU.sunsetDarkMode = {
    id: 'sunset-dark-mode', n: 15,
    title: 'Dark mode opens at sunset',
    goal: 'Switch to dark mode.',
    init(root, opts = {}) {
      root.classList.add('cu-sun');
      const thumb = (extra = '') => `<div class="win"><div class="side"></div><div class="body"><i style="width:80%"></i><i style="width:55%"></i><i style="width:68%"></i></div></div><div class="dots"><u></u><u></u><u></u></div>${extra}`;
      root.innerHTML = `
        <div class="app"><div class="icon" aria-hidden="true"><svg width="26" height="26" viewBox="0 0 26 26"><circle cx="13" cy="13" r="5.2" fill="#fff"/><ellipse cx="13" cy="13" rx="11" ry="4.6" fill="none" stroke="#fff" stroke-opacity=".55" stroke-width="1.4" transform="rotate(-24 13 13)"/><circle cx="22.6" cy="8.6" r="2" fill="#f2a43a"/></svg></div>
          <div><b>Orbit</b><span>Settings &middot; Appearance</span></div>
          <div class="x" aria-hidden="true"><svg width="10" height="10" viewBox="0 0 10 10"><path d="M1 1l8 8M9 1L1 9" stroke="currentColor" stroke-width="1.4" stroke-linecap="round"/></svg></div></div>
        <div class="lbl"><b>Theme</b><span>Applies to every device</span></div>
        <div class="tiles" role="radiogroup" aria-label="Theme">
          <button class="tile t-light sel" data-t="light" role="radio"><div class="thumb">${thumb()}</div><div class="cap"><i></i>Light</div></button>
          <button class="tile t-dark" data-t="dark" role="radio"><div class="thumb">${thumb(`<span class="lock" data-lock>${LOCKED}</span>`)}</div><div class="cap"><i></i>Dark</div></button>
          <button class="tile t-auto" data-t="auto" role="radio"><div class="thumb">${thumb()}<div class="half">${thumb()}</div></div><div class="cap"><i></i>Auto</div></button>
        </div>
        <div class="gate" data-gate><div class="gate-in">
          <div class="sky"><svg width="${W}" height="${H}" viewBox="0 0 ${W} ${H}" data-svg></svg>
            <div class="ticks"><span>Sunrise ${SUNRISE}</span><span>Sunset ${SUNSET}</span></div></div>
          <div class="when"><div class="cd"><small>Dark mode in</small><b data-cd></b><span class="fine" data-fine>+3 s</span></div><div class="at"><small>Unlocks at</small><b>${SUNSET}</b></div></div>
          <div class="status" data-status></div>
        </div></div>`;
      const q = s => root.querySelector(`[data-${s}]`);
      const svg = q('svg'), gate = q('gate'), cdEl = q('cd'), status = q('status'), fineEl = q('fine'), lockEl = q('lock');
      const tiles = {}; root.querySelectorAll('.tile').forEach(b => tiles[b.dataset.t] = b);
      const NS = 'http://www.w3.org/2000/svg', uid = 'cs' + Math.random().toString(36).slice(2, 7);
      const mk = (tag, attrs, parent = svg) => { const e = document.createElementNS(NS, tag); for (const k in attrs) e.setAttribute(k, attrs[k]); parent.appendChild(e); return e; };

      /* ---------- sky ---------- */
      const defs = mk('defs', {});
      const gSky = mk('linearGradient', { id: uid + 's', x1: 0, y1: 0, x2: 0, y2: 1 }, defs);
      const stops = [0, .62, 1].map(o => mk('stop', { offset: o }, gSky));
      const gGlow = mk('radialGradient', { id: uid + 'g' }, defs);
      const glowStops = [0, 1].map(o => mk('stop', { offset: o }, gGlow));

        .map(([x, y], i) => mk('circle', { cx: x, cy: y, r: i % 3 ? .9 : 1.3, fill: '#fff', opacity: 0 }));
      const glow = mk('circle', { r: 96, fill: `url(#${uid}g)` });
      mk('path', { d: `M${CX - R} ${HY}A${R} ${RY} 0 0 1 ${CX + R} ${HY}`, fill: 'none', stroke: 'rgba(255,255,255,.6)', 'stroke-width': 1, 'stroke-dasharray': '1.5 5', 'stroke-linecap': 'round' });

      const own = !opts.K;
      const K = opts.K || ReelKit({ embedded: true, update: dt => api.update(dt), render: () => api.render() });

      let sel = 'light', tries = 0, t0 = 0, grab = false, grabA = 0, msgHTML = '', fine = 0, pushT = 0, wasDark = false;
      const off = { x: 0, v: 0 }, push = { x: 0, v: 0 };

      const left = () => Math.max(0, LEFT0 + fine - Math.floor((K.now - t0) / 1000));
      const realA = () => A0 + CRAWL * (K.now - t0) / 1000 - push.x;
      function say(html) {
        if (html === msgHTML) return; msgHTML = html; status.innerHTML = html;
        K.tween(300, p => status.style.transform = `scale(${(1 + .05 * (1 - p)).toFixed(3)})`, K.ease.out);
      }
      function select(t) { sel = t; for (const k in tiles) { tiles[k].classList.toggle('sel', k === t); tiles[k].setAttribute('aria-checked', k === t); } }
      function bump(el, dx = 8) { K.tween(460, p => el.style.transform = `translateX(${(dx * Math.sin(p * Math.PI * 3) * (1 - p)).toFixed(2)}px)`, K.ease.lin); }

      function pickDark() {
        tries++;
        select('dark');
        const th = tiles.dark.querySelector('.thumb');                 // a real press, and Dark stays on long enough to see
        K.tween(80, p => th.style.transform = `scale(${(1 - .05 * p).toFixed(4)})`);
        K.tween(260, p => th.style.transform = `scale(${(.95 + .05 * p).toFixed(4)})`, K.ease.back, 90);

        say(MSG[Math.min(tries, MSG.length) - 1]);

        return Math.atan2(Math.max(0, (HY - py)) / RY, (CX - px) / R) / Math.PI;
      }
      const offs = [];
      const on = (el, ev, fn, o) => { el.addEventListener(ev, fn, o); offs.push(() => el.removeEventListener(ev, fn, o)); };

      on(tiles.auto, 'click', () => { select('auto'); later(170, () => { select('light'); bump(tiles.auto); }); say(`Auto also waits for <b>${SUNSET}</b>.`); });
      on(sunG, 'pointerdown', e => { grab = true; grabA = pointA(e.clientX, e.clientY) - (realA() + off.x); e.preventDefault(); });
      on(window, 'pointermove', e => {
        if (!grab) return;
        const ra = realA();
        off.x = K.clamp(pointA(e.clientX, e.clientY) - grabA, ra, 1.02) - ra; off.v = 0;
      });
      on(window, 'pointerup', () => {
        if (!grab) return; grab = false;
        if (off.x > .05) later(260, () => say(`Nice try. Sunset is still <b>${SUNSET}</b>.`));
      });

      function update(dt) {
        for (let i = timers.length - 1; i >= 0; i--) if (K.now >= timers[i].at) timers.splice(i, 1)[0].fn();
        if (!grab) K.spring(off, 0, 22, .85, dt);                     // floats back up over ~1.5 s
        K.spring(push, pushT, 160, .8, dt);
      }

        const ra = realA(), a = K.clamp(ra + off.x, 0, 1.03);
        const d = K.clamp(off.x / (1 - ra), 0, 1), e = d * d * (3 - 2 * d);

        const sy2 = a > 1 ? sy + (a - 1) * 400 : sy;

        const pulse = .5 + .5 * Math.sin(K.now / 2500 * Math.PI * 2);   // 2.5 s period: loops cleanly
        halo.setAttribute('r', (18 + 2.5 * pulse).toFixed(2)); halo.setAttribute('opacity', (.22 + .1 * pulse).toFixed(3));

        glow.setAttribute('cx', sx.toFixed(2)); glow.setAttribute('cy', sy2.toFixed(2));
        glowStops[0].style.stopColor = ok(SUN[0], SUN[1], e); glowStops[0].style.stopOpacity = (.55 + .3 * e).toFixed(3);
        glowStops[1].style.stopColor = ok(SUN[0], SUN[1], e); glowStops[1].style.stopOpacity = 0;
        stops[0].style.stopColor = ok(SKY_TOP[0], SKY_TOP[1], e);
        stops[1].style.stopColor = ok(SKY_MID[0], SKY_MID[1], e);
        stops[2].style.stopColor = ok(SKY_LOW[0], SKY_LOW[1], e);
        ground.style.fill = ok(GROUND[0], GROUND[1], e); ridge.style.fill = ok(RIDGE[0], RIDGE[1], e);
        stars.forEach((s, i) => s.setAttribute('opacity', K.clamp((e - .45 - i * .03) * 3, 0, 1).toFixed(3)));
        let dpath = ''; for (let i = 0; i <= 24; i++) { const [x, y] = arc(ra * i / 24); dpath += (i ? 'L' : 'M') + x.toFixed(2) + ' ' + y.toFixed(2); }
        trail.setAttribute('d', dpath);

        /* dragging the sun down races the countdown to zero (and back up when it floats home) */
        const s = Math.round(left() * (1 - K.clamp(e / .96, 0, 1))), hh = Math.floor(s / 3600), mm = Math.floor(s % 3600 / 60), ss = s % 60;
        const dark = e > .96;                                          // only at 0:00:00
        if (dark !== wasDark) {
          wasDark = dark; lockEl.innerHTML = dark ? OPEN : LOCKED;
          if (dark) { select('dark'); say('Sunset reached. Dark mode on.'); } else { select('light'); bump(tiles.dark); }
        }
        cdEl.textContent = `${hh}:${String(mm).padStart(2, '0')}:${String(ss).padStart(2, '0')}`;

        /* dragging the sun down previews the dark theme on the card itself */
        const ce = K.clamp((e - .3) / .4, 0, 1), cf = ce * ce * (3 - 2 * ce);
        root.style.background = hex('#ffffff', '#1a1a1f', cf);
        root.style.setProperty('--panel', hex('#ffffff', '#1a1a1f', cf));
        /* text flips light/dark over a narrow band around the midpoint, so it never fades into the background */
        const te = K.clamp((e - .47) / .06, 0, 1), tf = te * te * (3 - 2 * te);
        root.style.setProperty('--ink', hex('#121214', '#f2f1ee', tf));
        root.style.setProperty('--ink-2', hex('#5f5f66', '#a9a9b2', tf));
        root.style.setProperty('--ink-3', hex('#8d8c92', '#7b7b85', tf));
        root.style.setProperty('--line-2', hex('#e6e3dc', '#3a3a42', cf));
        root.style.setProperty('--line', `rgba(${Math.round(18 + 237 * e)},${Math.round(18 + 237 * e)},${Math.round(20 + 235 * e)},${(.085 + .02 * e).toFixed(3)})`);
        if (opts.onDusk) opts.onDusk(e, tf);
      }

      function reset() {
        timers.length = 0; tries = 0; grab = false; msgHTML = ''; fine = 0; pushT = 0; wasDark = false;
        off.x = off.v = push.x = push.v = 0; t0 = K.now; select('light'); say('Unlocks at your local sunset.');
        Object.values(tiles).forEach(t => t.style.transform = '');
      }

      const api = {
        update, render, reset,
        sunA: () => realA() + off.x,
        grabbing: () => grab,
        arcPoint: a => { const r = svg.getBoundingClientRect(), k = r.width / W, [x, y] = arc(a); return [r.left + x * k, r.top + y * k]; },
        tries: n => { tries = n; },
        el: { root, light: tiles.light, dark: tiles.dark, auto: tiles.auto, sun: sunG },
        destroy() { offs.forEach(f => f()); if (own) K.stop(); root.innerHTML = ''; root.classList.remove('cu-sun'); }
      };
      reset();
      if (own) K.start();
      return api;
    }
  };
})();
