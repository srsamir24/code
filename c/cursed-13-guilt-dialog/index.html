<script>/* inlined from ../_kit/reel.js */
/* Shared runtime for the Cursed UI pages.
   - a simulation clock (K.now, ms) advanced by K.tick(ms): rAF in the browser, stepped by the recorder (?manual)
   - tweens, easing, the hook <-> outro swap (window.__outro), and the reel cursor driven by real pointer events
   - opts.embedded: a private clock for a component mounted elsewhere (the game): no reel cursor, no globals, own rAF loop,
     K.stop() ends it. Pointer position is still tracked (K.px/K.py) */
window.ReelKit = function (opts) {
  if (opts.embedded) return ReelKit.embedded(opts);
  const qs = new URLSearchParams(location.search);
  const REEL = qs.has('reel'), MANUAL = qs.has('manual');
  if (REEL) document.body.classList.add('reel');

  const K = { REEL, MANUAL, now: 0, px: -999, py: -999, down: false };
  const tweens = [];
  K.ease = {
    lin: t => t,
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,
    back: t => { const c = 1.7; return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2); }
  };
  K.tween = (dur, fn, e = K.ease.out, delay = 0) => { const tw = { t0: K.now + delay, dur, fn, e }; tweens.push(tw); return tw; };
  K.clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  K.lerp = (a, b, t) => a + (b - a) * t;
  /* critically-ish damped spring step: z = damping ratio */
  K.spring = (s, target, k, z, dt) => { s.v += (k * (target - s.x) - 2 * Math.sqrt(k) * z * s.v) * dt; s.x += s.v * dt; };

  /* ---------- cursor ---------- */
  document.body.insertAdjacentHTML('beforeend', `<div id="cursor">
    <div id="ring"></div>
    <svg id="c-arrow" width="26" height="34" viewBox="0 0 26 34"><path d="M2 2 L2 26 L8.2 20.4 L12.4 30.4 L16.6 28.6 L12.5 18.9 L20.8 18.6 Z" fill="#111" stroke="#fff" stroke-width="2" stroke-linejoin="round"/></svg>
    <svg id="c-hand" width="30" height="34" viewBox="0 0 30 34" style="display:none;left:-9px;top:-2px"><path d="M10 15V5.2a2.6 2.6 0 0 1 5.2 0V13 M15.2 12.6a2.5 2.5 0 0 1 5 0v1.8 M20.2 13.6a2.5 2.5 0 0 1 5 0v6.2c0 6.6-4 11.2-10 11.2-4.6 0-6.7-2.2-9.4-6.5L3 19.6a2.5 2.5 0 0 1 4.2-2.7L10 20.8V15" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
    <svg id="c-grab" width="30" height="30" viewBox="0 0 30 30" style="display:none;left:-15px;top:-13px"><path d="M8.2 11.4a2.4 2.4 0 0 1 4.8 0v-1a2.4 2.4 0 0 1 4.8 0v1a2.4 2.4 0 0 1 4.8 0v1.4a2.4 2.4 0 0 1 4.6.9v4.8c0 5.6-3.8 9.4-9.4 9.4-4.2 0-6.4-1.8-8.6-5.2l-2.4-3.8a2.3 2.3 0 0 1 3.4-3z M13 11.4v4 M17.8 11.4v4 M22.6 12.8v3" fill="#fff" stroke="#111" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/></svg>
  </div>`);
  const $ = id => document.getElementById(id);
  const cur = $('cursor'), ring = $('ring'), shapes = { arrow: $('c-arrow'), hand: $('c-hand'), grab: $('c-grab') };
  let curScale = 1;
  function moveCursor() {
    if (!REEL) return;
    const el = document.elementFromPoint(K.px, K.py);
    const kind = (opts.cursorKind && opts.cursorKind(el)) || 'arrow';
    for (const k in shapes) shapes[k].style.display = k === kind ? '' : 'none';
    cur.style.transform = `translate(${K.px}px,${K.py}px) scale(${curScale})`;
  }
  K.moveCursor = moveCursor;
  window.addEventListener('pointermove', e => { K.px = e.clientX; K.py = e.clientY; moveCursor(); });
  window.addEventListener('pointerleave', () => { K.px = K.py = -999; });
  window.addEventListener('pointerdown', () => {
    K.down = true;
    if (!REEL) return;
    K.tween(90, p => { curScale = 1 - .14 * p; moveCursor(); });
    K.tween(300, p => { curScale = .86 + .14 * p; moveCursor(); }, K.ease.back, 120);
    K.tween(480, p => { ring.style.opacity = (1 - p) * .9; ring.style.transform = `scale(${.4 + p * 1.1})`; });
  });
  window.addEventListener('pointerup', () => { K.down = false; moveCursor(); });

  /* ---------- hook <-> outro ---------- */
  let outroOn = false;
  window.__outro = on => {
    if (on === outroOn) return; outroOn = on;
    const hook = document.querySelector('.hook'), outro = document.querySelector('.outro');
    K.tween(440, p => { const q = on ? p : 1 - p;
      hook.style.opacity = K.clamp(1 - q / .45, 0, 1);   // strictly sequential: never two lines at once
      const o = K.clamp((q - .55) / .45, 0, 1);
      outro.style.opacity = o; outro.style.transform = `translateY(${(1 - o) * 10}px)`; }, K.ease.lin);
  };

  /* ---------- clock ---------- */
  function tick(ms) {
    const n = Math.max(1, Math.ceil(ms / 4));
    for (let i = 0; i < n; i++) { K.now += ms / n; opts.update(ms / n / 1000); }
    for (let i = tweens.length - 1; i >= 0; i--) {
      const tw = tweens[i];
      if (K.now < tw.t0) continue;
      const p = tw.dur <= 1 ? 1 : Math.min(1, (K.now - tw.t0) / tw.dur);
      tw.fn(tw.e(p));
      if (p >= 1) tweens.splice(i, 1);
    }
    opts.render();
  }
  window.__tick = tick;
  K.start = () => {
    if (!MANUAL) {
      let last = performance.now();
      requestAnimationFrame(function loop(t) { tick(Math.min(50, t - last)); last = t; requestAnimationFrame(loop); });
    } else tick(16);
  };
  return K;
};

              <g data-tears fill="oklch(.72 .13 235)"><path data-tear d="M24.4 29.6q2.1 3.4 0 5q-2.1-1.6 0-5z"/><path data-tear d="M33.6 29.6q2.1 3.4 0 5q-2.1-1.6 0-5z"/></g>
            </svg>
            <div class="t" id="gd-t" data-t></div>
            <div class="m" id="gd-m" data-m></div>
          </div>
          <div class="btns"><button class="no" data-no></button><button class="yes" data-yes><span>Turn on</span></button></div>
        </div>`;
      const q = s => root.querySelector(`[data-${s}]`);
      const alert = q('alert'), scrim = q('scrim'), tEl = q('t'), mEl = q('m'), no = q('no'), yes = q('yes');
      const mascot = root.querySelector('.mascot'), faces = [...root.querySelectorAll('[data-face]')], eyes = q('eyes'), tears = [...root.querySelectorAll('[data-tear]')], ring = q('ring');

      const own = !opts.K;
      const K = opts.K || ReelKit({ embedded: true, update: (dt) => api.update(dt), render: () => api.render() });
      const timers = [];
      const later = (ms, fn) => timers.push({ at: K.now + ms, fn });

      let step = 0, open = true, busy = false, shake = 0, textA = 1, mood = [1, 0, 0, 0, 0], closeA = 1, scrimA = 1;
      const look = { share: { x: .5, v: 0 }, fs: { x: 17, v: 0 }, grey: { x: 0, v: 0 } };
      const vis = { x: 1, v: 0 };
      const sad = { x: 0, v: 0 };                     // 0..4, drives tears and the ring slipping

      function setText(i) { tEl.textContent = STEPS[i][0]; mEl.textContent = STEPS[i][1]; no.textContent = STEPS[i][2]; }
      function go(i, animate = true) {
        step = i;
        if (!animate) { setText(i); mood = mood.map((_, k) => k === i ? 1 : 0); return; }
        setText(i);                                                // swap at once under a small pop: never a blank alert
        K.tween(300, p => textA = p, K.ease.back);
        const from = mood.slice();
        K.tween(300, p => { mood = from.map((v, k) => K.lerp(v, k === i ? 1 : 0, p)); }, K.ease.inOut, 60);
      }
      function nope() {
        K.tween(420, p => shake = Math.sin(p * Math.PI * 4) * 9 * (1 - p), K.ease.lin);
      }
      function flash(btn) { btn.classList.add('press'); later(150, () => btn.classList.remove('press')); }
      function sayNo() {
        if (busy || !open) return;
        flash(no);
        if (step < STEPS.length - 1) { nope(); go(step + 1); fire('cursed:lose', 'guilt'); return; }
        busy = true; close();                                // fine. it closes.
        if (opts.reopen) later(1300, popBack);
        else fire('cursed:win');
      }
      function sayYes() {
        if (busy || !open) return;
        flash(yes); busy = true; close(); fire('cursed:lose', 'yes');
        later(900, popBack);
      }
      /* closing is sequential: the alert is gone (4 frames) before the dim lifts, so the two never overlap */
      function close() {
        open = false;
        K.tween(130, p => closeA = 1 - p, K.ease.inOut);
        K.tween(240, p => scrimA = 1 - p, K.ease.inOut, 110);
      }
      /* and it comes back hard: dim first, then the alert springs in from .85 with an overshoot */
      function popBack() {
        go(0, false); sad.x = sad.v = 0;
        look.share.x = LOOK[0][0]; look.fs.x = LOOK[0][1]; look.grey.x = LOOK[0][2]; look.share.v = look.fs.v = look.grey.v = 0;
        vis.x = vis.v = 0; closeA = 1; open = true; busy = false;
        K.tween(90, p => scrimA = p * 1.25, K.ease.out);
        K.tween(160, p => scrimA = 1.25 - .25 * p, K.ease.inOut, 90);
      }
      function closeNow() { open = false; busy = true; closeA = 0; scrimA = 0; vis.x = vis.v = 0; }
      function fire(name, why) {
        root.dispatchEvent(new CustomEvent(name, { bubbles: true, detail: { id: 'guilt-dialog', why } }));
        if (name === 'cursed:win' && opts.onWin) opts.onWin();
        if (name === 'cursed:lose' && opts.onLose) opts.onLose(why);
      }

      const offs = [];
      const on = (el, ev, fn) => { el.addEventListener(ev, fn); offs.push(() => el.removeEventListener(ev, fn)); };
      on(no, 'click', sayNo);
      root.querySelectorAll('.list li').forEach(li => on(li, 'click', () => {
        if (open) return;
        K.tween(420, p => li.style.background = `rgba(18,18,20,${(.06 * (1 - p)).toFixed(3)})`, K.ease.out);
      }));
      on(yes, 'click', sayYes);

      function update(dt) {
        for (let i = timers.length - 1; i >= 0; i--) if (K.now >= timers[i].at) timers.splice(i, 1)[0].fn();
        const L = LOOK[step];
        K.spring(look.share, L[0], 260, .78, dt);
        K.spring(look.fs, L[1], 260, .9, dt);
        K.spring(look.grey, L[2], 200, .95, dt);
        if (open) K.spring(vis, 1, 420, .5, dt);
        K.spring(sad, step, 120, .9, dt);
      }
      function render() {
        const v = vis.x;
        const op = open ? K.clamp(v * 2.4, 0, 1) : K.clamp(closeA, 0, 1);
        const sc = open ? .85 + .15 * v : .93 + .07 * K.clamp(closeA, 0, 1);
        alert.style.opacity = op.toFixed(3);
        alert.style.transform = `translateX(${shake.toFixed(2)}px) scale(${sc.toFixed(4)})`;
        alert.style.pointerEvents = open ? '' : 'none';
        scrim.style.opacity = K.clamp(scrimA, 0, 1).toFixed(3);
        scrim.style.background = `rgba(18,16,22,${(.26 + .05 * K.clamp(sad.x, 0, 4) + .12 * K.clamp(scrimA - 1, 0, .25) / .25).toFixed(3)})`;
        const tp = 1 + .06 * (1 - textA);
        tEl.style.transform = `scale(${tp.toFixed(4)})`;
        mEl.style.transform = `translateY(${((1 - textA) * 3).toFixed(2)}px)`;
        const s = look.share.x;
        no.style.flex = `0 0 ${(s * 100).toFixed(2)}%`;
        yes.style.flex = '1 1 0';
        no.style.fontSize = look.fs.x.toFixed(2) + 'px';
        const g = K.clamp(look.grey.x, 0, 1);
        no.style.color = `color-mix(in oklch, var(--tint) ${((1 - g) * 100).toFixed(1)}%, #6f6d75)`;
        yes.firstChild.style.transform = `scale(${(1 + .045 * K.clamp(sad.x, 0, 4)).toFixed(4)})`;
        faces.forEach((f, k) => f.style.opacity = mood[k].toFixed(3));
        eyes.style.opacity = K.clamp(1 - mood[4], 0, 1).toFixed(3);
        /* the mascot watches the pointer (pupils follow it) and breathes on a 2 s period, so it never freezes */
        const mr = mascot.getBoundingClientRect();
        if (mr.width && K.px > -900) {
          const ex = K.px - (mr.left + mr.width / 2), ey = K.py - (mr.top + mr.height / 2), en = Math.hypot(ex, ey) || 1;
          const k = Math.min(1, en / 120) * 1.5;
          eyes.setAttribute('transform', `translate(${(ex / en * k).toFixed(2)} ${(ey / en * k).toFixed(2)})`);
        }
        const br = Math.sin(K.now / 3000 * Math.PI * 2);
        mascot.style.transform = `translateY(${(br * 1.6).toFixed(2)}px) rotate(${(br * (1 + K.clamp(sad.x, 0, 4)) * .6).toFixed(2)}deg)`;
        /* "Turn on" fills in as the guilt grows: the preferred action gets louder every step */
        const fillA = K.clamp((sad.x - 1) / 3, 0, 1);
        yes.style.background = `color-mix(in oklch, var(--tint) ${(fillA * 100).toFixed(1)}%, transparent)`;
        yes.style.color = fillA > .45 ? '#fff' : '';
        const ph = (K.now % 1000) / 1000, cry = K.clamp(sad.x - 2.4, 0, 1.6);
        tears.forEach((t, k) => {
          const on = k === 0 ? K.clamp(cry / .6, 0, 1) : K.clamp((cry - .8) / .6, 0, 1);
          const p = (ph + k * .45) % 1;
          t.style.opacity = (on * (p < .75 ? 1 : (1 - p) * 4)).toFixed(3);
          t.style.transform = `translateY(${(p * 10).toFixed(2)}px)`;
        });
        const slip = K.clamp(sad.x - 3, 0, 1.2);
        ring.style.transform = `translate(0px,${(slip * 6).toFixed(2)}px) rotate(${(slip * 16).toFixed(2)}deg)`;
        ring.style.transformOrigin = '29px 30px';
      }
      function reset() {
        timers.length = 0; busy = false; open = true; go(0, false); shake = 0; textA = 1; closeA = 1; scrimA = 1;
        vis.x = 1; vis.v = 0; sad.x = sad.v = 0;
        look.share.x = .5; look.fs.x = 17; look.grey.x = 0; look.share.v = look.fs.v = look.grey.v = 0;
      }

      const api = {
        update, render, reset, popBack, closeNow,
        el: { root, no, yes, alert, row: i => root.querySelectorAll('.list li')[i] },
        destroy() { offs.forEach(f => f()); if (own) K.stop(); root.innerHTML = ''; root.classList.remove('cu-gd'); }
      };
      reset();
      if (own) K.start();
      return api;
    }
  };
})();
</script>
