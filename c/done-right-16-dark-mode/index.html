<script id="component">
  for (const picker of document.querySelectorAll('[data-theme-picker]')) {
    const root = document.documentElement;
    const os = matchMedia('(prefers-color-scheme: dark)');
    const hint = document.getElementById(picker.getAttribute('aria-describedby'));

    let saved = null;
    try { saved = localStorage.getItem('theme'); } catch (e) {}
    const choice = saved === 'light' || saved === 'dark' ? saved : 'system';
    picker.querySelector(`input[value="${choice}"]`).checked = true;

    // "System" needs no JS: CSS follows the OS live.
    // This only keeps the hint text honest.
    const showOS = () => {
      hint.querySelector('b').textContent = os.matches ? 'dark' : 'light';
    };
    showOS();
    os.addEventListener('change', showOS);

    picker.addEventListener('change', (e) => {
      const v = e.target.value;
      if (v === 'system') delete root.dataset.theme;
      else root.dataset.theme = v;
      try {
        if (v === 'system') localStorage.removeItem('theme');
        else localStorage.setItem('theme', v);
      } catch (e) {}   // storage blocked: still works for this visit
    });
  }
</script>
