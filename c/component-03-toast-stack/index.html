<script>
  const icons = () => lucide.createIcons({ attrs: { 'stroke-width': 2.2 } });
  icons();
  const region = document.getElementById('region');
  const count = document.getElementById('count');
  const announce = document.getElementById('announce');
  const KINDS = {
    deploy: { cls: 'ok', icon: 'check', title: 'Deployed to production', sub: 'studio-site · 41 s', act: 'Open' },
    upload: { cls: 'busy', icon: 'loader-circle', title: 'Uploading 12 photos', sub: '0 of 12', act: 'Hide', progress: true },
    delete: { cls: 'warn', icon: 'trash-2', title: 'Invoice #1048 deleted', sub: 'Undo in 7 s', act: 'Undo', undo: 7 },
  };
  let stack = [];
  const GAP = 10;

  // slide:js
  function layout() {
    let y = 0;
    stack.forEach((t, i) => {
      t.style.setProperty('--i', i); t.dataset.i = i;
      t.style.setProperty('--y', y + 'px');
      y += t.scrollHeight + GAP;
    });
    region.style.setProperty('--front-h', stack[0] ? stack[0].offsetHeight + 'px' : 'auto');
    const hidden = Math.max(0, stack.length - 3);
    region.classList.toggle('has-more', hidden > 0);
    count.textContent = hidden ? `+${hidden} more` : '';
  }
  // /slide
  function push(kind) {
    const k = KINDS[kind];
    const t = document.createElement('div');
    t.className = `toast ${k.cls} enter`;
    announce.textContent = `${k.title}. ${k.sub}`;
    t.innerHTML = `<div class="badge"><i data-lucide="${k.icon}"></i></div>
      <div class="txt"><strong>${k.title}</strong><span>${k.sub}</span></div>
      <button class="act" type="button">${k.act}</button>` + (k.progress ? '<div class="bar"><i></i></div>' : '');
    region.appendChild(t);
    icons();
    stack.unshift(t);
    layout();
    requestAnimationFrame(() => requestAnimationFrame(() => t.classList.remove('enter')));
    t.querySelector('.act').addEventListener('click', () => dismiss(t));
    drag(t);
    if (k.progress) {
      let n = 0;
      t._iv = setInterval(() => {
        n++;
        t.querySelector('.txt span').textContent = `${n} of 12`;
        t.querySelector('.bar i').style.setProperty('--p', n / 12);
        if (n === 12) {
          clearInterval(t._iv);
          setTimeout(() => {
            t.classList.replace('busy', 'ok');
            t.querySelector('.badge').innerHTML = '<i data-lucide="check"></i>';
            t.querySelector('strong').textContent = '12 photos uploaded';
            t.querySelector('.txt span').textContent = 'Added to Harbour rebrand';
            t.querySelector('.act').textContent = 'View';
            t.querySelector('.bar').classList.add('gone');
            icons();
            setTimeout(layout, 320);
          }, 300);
        }
      }, 260);
    }
    if (k.undo) {
      let s = k.undo;
      t._iv = setInterval(() => {
        if (region.classList.contains('open')) return;
        s--;
        t.querySelector('.txt span').textContent = `Undo in ${s} s`;
        if (s <= 0) { clearInterval(t._iv); dismiss(t); }
      }, 1000);
    } else {
      t._timer = setTimeout(() => dismiss(t), 8000);
    }
  }
  function dismiss(t, dx = 0) {
    if (!stack.includes(t)) return;
    clearTimeout(t._timer); clearInterval(t._iv);
    t.style.setProperty('--dx', dx + 'px');
    t.classList.add('leave');
    stack = stack.filter(x => x !== t);
    layout();
    setTimeout(() => t.remove(), 380);
  }
  // swipe sideways to dismiss
  function drag(t) {
    let x0 = null;
    t.addEventListener('pointerdown', e => { if (e.target.closest('button')) return; x0 = e.clientX; t.setPointerCapture(e.pointerId); t.classList.add('dragging'); });
    t.addEventListener('pointermove', e => { if (x0 === null) return; t.style.translate = `${e.clientX - x0}px 0`; });
    t.addEventListener('pointerup', e => {
      if (x0 === null) return;
      const dx = e.clientX - x0; x0 = null;
      t.classList.remove('dragging'); t.style.translate = '';
      if (Math.abs(dx) > 90) dismiss(t, dx > 0 ? 420 : -420);
    });
  }

  // pause timers and fan out while the pointer or focus is on the stack
  const open = on => {
    region.classList.toggle('open', on);
    stack.forEach(t => { clearTimeout(t._timer); if (!on && !t.classList.contains('warn')) t._timer = setTimeout(() => dismiss(t), 4000); });
  };
  region.addEventListener('pointerenter', () => open(true));
  region.addEventListener('pointerleave', () => open(false));
  region.addEventListener('focusin', () => open(true));
  region.addEventListener('focusout', e => { if (!region.contains(e.relatedTarget)) open(false); });
  document.addEventListener('keydown', e => { if (e.altKey && e.key.toLowerCase() === 't') { e.preventDefault(); region.focus(); } });
  document.querySelectorAll('.trigger').forEach(b => b.addEventListener('click', () => push(b.dataset.kind)));

  const params = new URLSearchParams(location.search);
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    (async function loop() {
      await wait(600); push('deploy');
      await wait(900); push('upload');
      await wait(900); push('delete');
      await wait(1400); open(true);
      await wait(2600); open(false);
      await wait(900);
      [...stack].forEach((t, i) => setTimeout(() => dismiss(t, 420), i * 140));
      await wait(1200);
      loop();
    })();
  }
  if (params.has('bare')) document.body.classList.add('bare');
  if (params.has('still')) {
    push('deploy'); push('upload'); push('delete');
    if (params.get('still') === 'open') open(true);
  }
</script>
