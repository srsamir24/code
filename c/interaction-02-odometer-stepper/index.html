<script>/* inlined from ../_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};
</script>
<script>
(() => {
  const $ = id => document.getElementById(id);
  const reduce = matchMedia('(prefers-reduced-motion: reduce)');
  const params = new URLSearchParams(location.search);
  const MIN = 0, MAX = 999;
  const K = 400, C = 17;                 // spring stiffness, damping (≈ 0.42 damping ratio: one small overshoot)
  const R = 128, STEP = Math.PI * 0.29;  // drum radius (px), 52° per digit

  let value = +(params.get('v') || 96);
  const wheels = [1, 10, 100].map((place, i) => {
    const el = $('d' + i), spans = [];
    for (let d = 0; d < 10; d++) { const s = document.createElement('span'); s.textContent = d; el.appendChild(s); spans.push(s); }
    const pos = Math.floor(value / place);
    return { place, el, spans, pos, vel: 0 };
  });

  // ---------- value ----------
  function set(next) {
    const clamped = Math.max(MIN, Math.min(MAX, next));
    if (clamped === value) {                       // at a stop: knock the ones wheel against it
      if (!reduce.matches) wheels[0].vel += 3.2 * Math.sign(next - value);
      return;
    }
    value = clamped;
    $('win').setAttribute('aria-valuenow', value);
    $('minus').setAttribute('aria-disabled', value === MIN);
    $('plus').setAttribute('aria-disabled', value === MAX);
  }

  // ---------- hold to repeat (on simulated time, so it is exact at any frame rate) ----------
  let hold = null;                                  // { dir, t, next, gap }
  function press(dir) { set(value + dir); hold = { dir, t: 0, next: 0.42, gap: 0.11 }; }
  function lift() { hold = null; }
  for (const [id, dir] of [['plus', 1], ['minus', -1]]) {
    const b = $(id);
    b.addEventListener('pointerdown', e => { if (e.button) return; e.preventDefault(); b.setPointerCapture(e.pointerId); press(dir); });
    b.addEventListener('pointerup', lift);
    b.addEventListener('pointercancel', lift);
    b.addEventListener('lostpointercapture', lift);
    b.addEventListener('click', e => { if (e.detail === 0) set(value + dir); });   // Enter / Space / assistive tech
  }
  $('win').addEventListener('keydown', e => {
    const d = { ArrowUp: 1, ArrowRight: 1, ArrowDown: -1, ArrowLeft: -1, PageUp: 10, PageDown: -10 }[e.key];
    if (d) { e.preventDefault(); set(value + d); }
    if (e.key === 'Home') { e.preventDefault(); set(MIN); }
    if (e.key === 'End') { e.preventDefault(); set(MAX); }
  });

  // ---------- physics ----------
  const btn = { plus: { el: $('plus'), p: 0, v: 0 }, minus: { el: $('minus'), p: 0, v: 0 } };
  function step(dt) {
    for (const [id, b] of Object.entries(btn)) {     // button travel: a stiff little spring
      const goal = hold && (hold.dir > 0) === (id === 'plus') ? 1 : 0;
      b.v += (goal - b.p) * 2600 * dt; b.v *= Math.exp(-60 * dt); b.p += b.v * dt;
    }
    if (hold) {
      hold.t += dt;
      while (hold.t >= hold.next) { set(value + hold.dir); hold.next += hold.gap; hold.gap = Math.max(0.055, hold.gap * 0.9); }
    }
    for (const w of wheels) {
      const target = Math.floor(value / w.place);
      if (reduce.matches) { w.pos = target; w.vel = 0; continue; }
      w.vel += (target - w.pos) * K * dt;          // spring into the detent
      w.vel *= Math.exp(-C * dt);                  // damping: one small overshoot
      w.pos += w.vel * dt;
    }
  }

  // ---------- render: digits around a drum ----------
  function render() {
    for (const b of Object.values(btn)) {
      const p = Math.max(0, b.p);
      b.el.style.transform = `translateY(${(p * 4).toFixed(2)}px) scale(${(1 - p * 0.04).toFixed(4)})`;
      b.el.style.boxShadow = `0 ${(10 - p * 8).toFixed(2)}px ${(22 - p * 16).toFixed(2)}px -8px rgba(40,30,15,${(0.22 - p * 0.1).toFixed(3)}), inset 0 ${(p * 3).toFixed(2)}px ${(p * 8).toFixed(2)}px rgba(40,30,15,${(p * 0.14).toFixed(3)}), inset 0 1px 0 rgba(255,255,255,${(0.9 - p * 0.9).toFixed(2)})`;
    }
    for (const w of wheels) {
      const p = ((w.pos % 10) + 10) % 10;
      for (let d = 0; d < 10; d++) {
        let off = d - p; off -= 10 * Math.round(off / 10);  // nearest way round, -5..5
        const a = off * STEP, s = w.spans[d];
        if (Math.abs(a) >= Math.PI / 2) { s.style.opacity = 0; continue; }
        const c = Math.cos(a);
        s.style.opacity = (c ** 3.2).toFixed(3);
        s.style.transform = `translateY(${(R * Math.sin(a)).toFixed(2)}px) scaleY(${c.toFixed(4)})`;
      }
    }
  }

  Reel({
    step, render,
    geom: () => {
      const r = id => { const b = $(id).getBoundingClientRect(); return { x: b.left + b.width / 2, y: b.top + b.height / 2 }; };
      return { plus: r('plus'), minus: r('minus'), bottom: $('card').getBoundingClientRect().bottom };
    },
  });
})();
