<script>
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const $ = id => document.getElementById(id);
  const fit = $('fit'), env = $('env'), letter = $('letter'), fold = $('fold'), flap = $('flap');
  const stamp = $('stamp'), stampShadow = $('stampShadow'), post = $('post'), after = $('after'), live = $('live');
  const say = t => { live.textContent = ''; setTimeout(() => live.textContent = t, 30); };

  function resize() {
    const scale = Math.min(1, (innerWidth - 16) / 480, (innerHeight - 16) / 720);
    fit.style.transform = `scale(${scale})`;
    fit.style.margin = `${(720 * scale - 720) / 2}px ${(480 * scale - 480) / 2}px`;
  }
  addEventListener('resize', resize); resize();

  // the postmark carries the real date and time
  function stampDate() {
    const d = new Date();
    $('postDay').textContent = d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short' }).toUpperCase();
    $('postTime').textContent = d.toLocaleTimeString('en-GB', { hour: '2-digit', minute: '2-digit' });
  }

  // three plies, each an inert copy of the form clipped to one third
  function cutPlies() {
    fold.innerHTML = '';
    const copyForm = () => {
      const c = letter.cloneNode(true);
      c.removeAttribute('id'); c.inert = true;
      c.classList.remove('nudge');
      c.querySelectorAll('[id]').forEach(n => n.removeAttribute('id'));
      c.querySelectorAll('input, textarea').forEach((n, i) => {
        const real = letter.querySelectorAll('input, textarea')[i];
        n.value = real.value; if (n.tagName === 'TEXTAREA') n.textContent = real.value;
      });
      c.style.cssText = 'position:absolute;left:0;top:0;box-shadow:none;width:392px;height:474px';
      return c;
    };
    return ['t', 'm', 'u'].map((name, i) => {
      const ply = document.createElement('div');
      ply.className = 'ply ' + name;
      const f = document.createElement('div'); f.className = 'f';
      const w = document.createElement('div'); w.className = 'copy'; w.style.top = -158 * i + 'px';
      w.appendChild(copyForm()); f.appendChild(w);
      const shade = document.createElement('div'); shade.className = 'shade'; f.appendChild(shade);
      const b = document.createElement('div'); b.className = 'b';
      const shadeBack = document.createElement('div'); shadeBack.className = 'shade'; b.appendChild(shadeBack);
      ply.append(f, b); fold.appendChild(ply);
      return { ply, shade, shadeBack };
    });
  }

  // helpers: every beat is a function of one clock, so stills and reverse need no extra code
  const clamp = (a, b, v) => Math.max(a, Math.min(b, v));
  const seg = (t, a, b) => clamp(0, 1, (t - a) / (b - a));
  const inOut = u => u < 0.5 ? 4 * u * u * u : 1 - Math.pow(-2 * u + 2, 3) / 2;
  const outC = u => 1 - Math.pow(1 - u, 3);
  // paper: slow to start, a touch past flat at the end
  const crease = u => inOut(Math.min(1, u / 0.82)) + (u > 0.82 ? Math.sin((u - 0.82) / 0.18 * Math.PI) * 0.02 : 0);
  // an underdamped spring, sampled: overshoots 1 and settles
  const spring = (u, f = 3.2, d = 5.5) => u <= 0 ? 0 : 1 - Math.exp(-d * u) * Math.cos(f * Math.PI * u);

  const T = {
    foldU: [0, 520], foldT: [120, 620], drop: [560, 980], flap: [960, 1340],
    flip: [1320, 1880], stamp: [1900, 2320], mark: [2300, 2560], end: 2800,
  };
  let plies = null;
  const LETTER_TOP = -416;                       // the letter's top, in the envelope's coordinates

  function pose(ms) {
    if (plies) {
      const [t, m, u] = plies;
      const fu = crease(seg(ms, ...T.foldU));
      const ft = crease(seg(ms, ...T.foldT));
      u.ply.style.transform = `translateZ(${0.6 + fu}px) rotateX(${fu * 178}deg)`;
      t.ply.style.transform = `translateZ(${1.2 + ft * 2}px) rotateX(${-ft * 178}deg)`;
      // both faces of a ply darken as it passes the light
      u.shade.style.opacity = Math.sin(Math.min(1, fu) * Math.PI) * 0.3;
      t.shade.style.opacity = Math.sin(Math.min(1, ft) * Math.PI) * 0.2;
      u.shadeBack.style.opacity = Math.sin(Math.min(1, fu) * Math.PI) * 0.13;
      t.shadeBack.style.opacity = Math.sin(Math.min(1, ft) * Math.PI) * 0.1;
      // the packet drops into the pocket: it starts where the letter was, ends inside the envelope
      const d = inOut(seg(ms, ...T.drop));
      const lift = Math.sin(seg(ms, 0, 620) * Math.PI) * 26;          // the paper leaves the desk as it folds
      const y = LETTER_TOP + (50 - 158 - LETTER_TOP) * d;          // the middle ply ends 50px inside the mouth
      // in front of everything until its lower edge reaches the envelope, then flat and behind the front panels
      const sink = seg(d, 0, 0.3);
      fold.style.transform = `translate3d(0, ${y}px, ${8 - 7 * sink + lift}px) rotateZ(${Math.sin(seg(ms, 0, 620) * Math.PI) * -2.4}deg) scaleZ(${1 - 0.9 * sink})`;
      fold.style.display = 'block';
      fold.style.opacity = ms >= T.flap[1] ? 0 : 1;
    }
    // flap: 180 is open, 0 is closed. It overshoots the envelope and presses flat
    const fl = spring(seg(ms, ...T.flap), 1.8, 6);
    flap.style.transform = `translateZ(${0.5 + fl * 5.5}px) rotateX(${180 - fl * 180}deg)`;

    // the envelope comes up from the bottom of the scene, centres, and flips to its address side
    const rise = inOut(seg(ms, T.flap[0] + 140, T.flip[1]));
    const fl2 = spring(seg(ms, ...T.flip), 1.5, 6);
    const dipU = seg(ms, T.stamp[0] + 120, T.stamp[0] + 460);
    const dip = dipU > 0 ? Math.exp(-6 * dipU) * Math.sin(dipU * Math.PI * 4) * 9 : 0;
    const hitU = seg(ms, T.mark[0], T.mark[0] + 240);
    const hit = hitU > 0 ? Math.exp(-7 * hitU) * Math.sin(hitU * Math.PI * 3.4) * 6 : 0;
    env.style.transform = `translateY(${rise * -96 + dip + hit}px) rotateZ(${rise * -1.4}deg) rotateY(${fl2 * 180}deg)`;

    // the stamp lands from above with its own shadow closing in under it
    const su = seg(ms, ...T.stamp);
    const s = spring(su, 2.4, 6.5);
    stamp.style.opacity = su > 0 ? 1 : 0;
    stamp.style.transform = `translate(${(1 - s) * -22}px, ${(1 - s) * -40}px) scale(${1 + (1 - s) * 0.85}) rotate(${(1 - s) * -26 + 2.6}deg)`;
    stampShadow.style.opacity = su > 0 ? 0.25 + 0.55 * outC(su) : 0;
    stampShadow.style.transform = `translate(${(1 - s) * 16}px, ${(1 - s) * 24 + 1.5}px) scale(${1 + (1 - s) * 0.5}) rotate(${(1 - s) * -26 + 2.6}deg)`;

    // the postmark bangs down and stays a little uneven
    const mu = seg(ms, ...T.mark);
    post.style.opacity = mu > 0 ? 0.16 + 0.68 * outC(mu * 3 > 1 ? 1 : mu * 3) : 0;
    post.style.transform = `scale(${1 + (1 - outC(seg(ms, T.mark[0], T.mark[0] + 90))) * 0.2}) rotate(-11deg)`;
  }

  function rest() {
    fold.style.display = 'none'; plies = null;
    letter.style.visibility = 'visible';
    letter.style.opacity = '';
    after.classList.remove('on');
    pose(0);
  }

  let running = 0;
  function play(from, to, done) {
    cancelAnimationFrame(running);
    const start = performance.now(), dur = Math.abs(to - from) / (to > from ? 1 : 1.9);
    (function frame(now) {
      const u = dur ? Math.min(1, (now - start) / dur) : 1;
      pose(from + (to - from) * u);
      if (u < 1) running = requestAnimationFrame(frame); else done?.();
    })(start);
  }

  // validation first: a bad letter never folds
  const valid = {
    name: v => v.trim().length > 1,
    mail: v => /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v.trim()),
    msg: v => v.trim().length > 0,
  };
  const FIELDS = [['name', 'fName'], ['mail', 'fMail'], ['msg', 'fMsg']];
  function check() {
    let first = null;
    FIELDS.forEach(([id, wrap]) => {
      const ok = valid[id]($(id).value);
      $(wrap).classList.toggle('bad', !ok);
      $(id).setAttribute('aria-invalid', !ok);
      if (!ok && !first) first = $(id);
    });
    return first;
  }
  FIELDS.forEach(([id, wrap]) => $(id).addEventListener('input', () => {
    if ($(wrap).classList.contains('bad') && valid[id]($(id).value)) { $(wrap).classList.remove('bad'); $(id).setAttribute('aria-invalid', 'false'); }
  }));

  function prepare() {
    $('fromName').textContent = $('name').value.trim();
    $('fromMail').textContent = $('mail').value.trim();
    stampDate();
    plies = cutPlies();
    letter.style.visibility = 'hidden';
  }

  letter.addEventListener('submit', e => {
    e.preventDefault();
    const bad = check();
    if (bad) {
      bad.focus();
      if (!reduce) { letter.classList.remove('nudge'); letter.offsetWidth; letter.classList.add('nudge'); }
      say('Check the letter: ' + bad.closest('.field').querySelector('.err').textContent);
      return;
    }
    prepare();
    say('Folding your letter');
    // your network call goes here; the animation runs while it does
    if (reduce) { pose(T.end); finished(); return; }
    play(0, T.end, finished);
  });

  function finished() {
    after.classList.add('on');
    $('afterLine').textContent = `We'll answer ${$('name').value.trim().split(' ')[0]} within two working days.`;
    say('Letter sent. We answer within two working days.');
    $('again').focus({ preventScroll: true });
  }
  $('again').addEventListener('click', () => {
    letter.reset();
    FIELDS.forEach(([id, wrap]) => { $(wrap).classList.remove('bad'); $(id).removeAttribute('aria-invalid'); });
    after.classList.remove('on');
    const unwind = () => { rest(); $('name').focus({ preventScroll: true }); };
    if (reduce) { unwind(); return; }
    // the envelope turns back, opens and hands you a blank sheet: the same clock, backwards
    plies = cutPlies();
    play(T.end, 0, unwind);
  });

  rest();

  // stills and the demo loop
  const params = new URLSearchParams(location.search);
  const still = params.get('still');
  const fill = () => {
    $('name').value = 'Maren Aas'; $('mail').value = 'maren@halden.no';
    $('msg').value = 'We need a booking page for the bakery, and it has to feel like us. Can we talk on Thursday?';
  };
  if (still) {
    document.fonts.ready.then(() => {
      fill();
      if (still === 'form') return;
      prepare();
      if (still === 'fold') pose(Number(params.get('t') || 270));
      if (still === 'sent') { pose(T.end); finished(); document.activeElement?.blur(); }
    });
  }
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    (async function loop() {
      fill(); await wait(1200); letter.requestSubmit();
      await wait(4200); $('again').click();
      await wait(2200); loop();
    })();
  }
</script>
