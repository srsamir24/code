<script>
const ICONS = {
  Home: '<path d="M4 11.5 12 4l8 7.5V20a1 1 0 0 1-1 1h-4v-6H9v6H5a1 1 0 0 1-1-1z"/>',
  Search: '<circle cx="11" cy="11" r="6.5"/><path d="m16 16 4.5 4.5"/>',
  Saved: '<path d="M12 20.5s-8-4.6-8-10.6A4.6 4.6 0 0 1 12 7a4.6 4.6 0 0 1 8 2.9c0 6-8 10.6-8 10.6z"/>',
  You: '<circle cx="12" cy="8" r="4"/><path d="M4.5 21c.8-4 3.8-6 7.5-6s6.7 2 7.5 6"/>',
};
const NAMES = Object.keys(ICONS);
const svg = n => `<svg viewBox="0 0 24 24" aria-hidden="true">${ICONS[n]}</svg>`;

document.querySelectorAll('.bar').forEach((bar, n) => {
  const kind = bar.dataset.kind;
  bar.setAttribute('role', 'tablist');
  bar.setAttribute('aria-label', `Tab bar example ${n + 1}`);
  const extra = { stretch: '<i class="pill"></i>', worm: '<i class="line"></i>', hop: '<i class="ball"></i>', rise: '<i class="plate"></i><i class="orb"></i>' }[kind] || '';
  bar.innerHTML = extra + NAMES.map(name => `<button role="tab" aria-label="${name}" aria-selected="false">${svg(name)}${
    kind === 'fill' ? `<i class="solid">${svg(name)}</i>` : kind === 'label' ? `<span>${name}</span>` : ''}</button>`).join('') + '<i class="tap"></i>';
  const tabs = [...bar.querySelectorAll('button')];
  const tap = bar.querySelector('.tap');
  const ball = bar.querySelector('.ball');
  let now = -1, touched = false;

  const edges = i => {
    const t = tabs[i], w = bar.clientWidth;
    return { l: t.offsetLeft, r: w - t.offsetLeft - t.offsetWidth, c: t.offsetLeft + t.offsetWidth / 2 };
  };
  const ballAt = c => `translate(${c}px, 13px)`;

  function select(i, instant) {
    if (i === now) return;
    const was = now; now = i;
    tabs.forEach((t, k) => t.setAttribute('aria-selected', k === i));
    const e = edges(i), right = was >= 0 && i > was;
    if (kind === 'stretch') {
      bar.style.setProperty('--tl', right ? '0.46s' : '0.24s');
      bar.style.setProperty('--tr', right ? '0.24s' : '0.46s');
      bar.style.setProperty('--l', e.l + 6 + 'px');
      bar.style.setProperty('--r', e.r + 6 + 'px');
    }
    if (kind === 'worm') {
      bar.style.setProperty('--dl', right ? '0.15s' : '0s');
      bar.style.setProperty('--dr', right ? '0s' : '0.15s');
      bar.style.setProperty('--l', e.l + (tabs[i].offsetWidth - 7) / 2 + 'px');
      bar.style.setProperty('--r', e.r + (tabs[i].offsetWidth - 7) / 2 + 'px');
    }
    if (kind === 'rise') bar.style.setProperty('--to', e.c + 'px');
    if (kind === 'hop') {
      if (was < 0 || instant) { ball.style.transform = ballAt(e.c); return; }
      const a = edges(was).c, mid = (a + e.c) / 2;
      ball.animate([
        { transform: `${ballAt(a)} scale(1, 1)`, easing: 'cubic-bezier(0.2, 0.7, 0.4, 1)' },
        { transform: `translate(${mid}px, -14px) scale(0.86, 1.14)`, easing: 'cubic-bezier(0.6, 0, 0.8, 0.4)', offset: 0.46 },
        { transform: `${ballAt(e.c)} scale(1.5, 0.55)`, easing: 'cubic-bezier(0.2, 0.8, 0.3, 1)', offset: 0.78 },
        { transform: `${ballAt(e.c)} scale(1, 1)` },
      ], { duration: 620 });
      ball.style.transform = ballAt(e.c);
    }
  }
  tabs.forEach((t, i) => t.addEventListener('click', ev => {
    if (ev.isTrusted) touched = true;
    select(i);
  }));
  bar.addEventListener('keydown', ev => {
    const d = { ArrowRight: 1, ArrowLeft: -1 }[ev.key];
    if (!d) return;
    touched = true;
    const i = (now + d + tabs.length) % tabs.length;
    select(i); tabs[i].focus();
  });

  // shows itself until someone uses it
  // bars 4 and 6 stay off the far tab, which sits under Instagram's buttons
  const route = n === 3 || n === 5 ? [2, 0, 1, 0] : [2, 0, 3, 1];
  let step = 0;
  function demo() {
    if (touched) return;
    const i = route[step++ % route.length];
    const e = edges(i);
    tap.style.left = e.c + 'px'; tap.style.top = kind === 'rise' ? '40%' : '50%';
    tap.animate([{ opacity: 0, transform: 'scale(0.6)' }, { opacity: 1, transform: 'scale(1)', offset: 0.25 }, { opacity: 0, transform: 'scale(1.25)' }], { duration: 280, easing: 'ease-out' });
    setTimeout(() => select(i), 90);
    setTimeout(demo, 1500);
  }
  document.fonts.ready.then(() => {
    select(route[route.length - 1], true);
    setTimeout(demo, 60 + n * 160);
  });
});
</script>
