<script>
  const DATA = [
    ['Suggestions', [
      ['New invoice', 'file-plus-2', ['N']],
      ['Invite a teammate', 'user-plus', ['I']],
      ['Open inbox', 'inbox', ['G', 'I']],
    ]],
    ['Projects', [
      ['Harbour rebrand', 'folder', []],
      ['Invoices, October', 'folder', []],
      ['Studio website', 'folder', []],
    ]],
    ['Settings', [
      ['Billing and plans', 'credit-card', []],
      ['Notifications', 'bell', []],
      ['Switch to dark theme', 'moon', ['T']],
    ]],
  ];
  const list = document.getElementById('list');
  const viewport = document.getElementById('viewport');
  const glide = list.querySelector('.glide');
  const q = document.getElementById('q');
  const status = document.getElementById('status');
  const rows = [];
  let gid = 0;
  for (const [group, items] of DATA) {
    const g = document.createElement('div');
    g.className = 'group'; g.textContent = group; g.id = 'g' + gid++;
    list.insertBefore(g, list.querySelector('.empty'));
    for (const [name, icon, keys] of items) {
      const el = document.createElement('div');
      el.className = 'item'; el.setAttribute('role', 'option'); el.id = 'o' + rows.length;
      el.innerHTML = `<span class="ic"><i data-lucide="${icon}"></i></span><span class="name"></span>` +
        `<span class="hint">${keys.map(k => `<kbd>${k}</kbd>`).join('')}</span>`;
      list.insertBefore(el, list.querySelector('.empty'));
      rows.push({ el, name, group: g, order: rows.length, nameEl: el.querySelector('.name') });
    }
  }
  lucide.createIcons({ attrs: { 'stroke-width': 2.1 } });

  // subsequence match with a score: word starts and runs of letters count more
  // slide:js
  function match(name, query) {
    query = query.toLowerCase().replace(/\s+/g, '');
    if (!query) return { hits: [], score: 0 };
    const n = name.toLowerCase(), hits = [];
    let from = 0, score = 0;
    for (const ch of query) {
      const i = n.indexOf(ch, from);
      if (i < 0) return null;
      if (i === 0 || n[i - 1] === ' ') score += 3;
      if (i === hits.at(-1) + 1) score += 2;
      score -= (i - from) * 0.1;
      hits.push(i); from = i + 1;
    }
    return { hits, score };
  }
  // /slide
  const esc = s => s.replace(/[&<>]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;' }[c]));

  let visible = [], active = 0;
  function render() {
    const query = q.value.trim();
    const before = viewport.offsetHeight;
    visible = [];
    const groupsOn = new Set();
    for (const r of rows) {
      const m = match(r.name, query);
      r.el.hidden = !m;
      if (!m) continue;
      r.score = m.score;
      visible.push(r); groupsOn.add(r.group);
      // letters that match in a row share one mark
      r.nameEl.innerHTML = [...r.name].map((c, i) => m.hits.includes(i) ? `<b>${esc(c)}</b>` : esc(c)).join('').replaceAll('</b><b>', '');
    }
    // best match first inside each group, groups keep their order
    for (const g of new Set(rows.map(r => r.group))) {
      const inG = visible.filter(r => r.group === g).sort((a, b) => b.score - a.score || a.order - b.order);
      let anchor = g;
      for (const r of inG) { anchor.after(r.el); anchor = r.el; }
    }
    visible = [...list.querySelectorAll('.item:not([hidden])')].map(el => rows.find(r => r.el === el));
    list.querySelectorAll('.group').forEach(g => g.hidden = !groupsOn.has(g));
    list.classList.toggle('is-empty', visible.length === 0);
    // animate the panel height to fit
    viewport.style.height = 'auto';
    const after = viewport.offsetHeight;
    viewport.style.height = before + 'px';
    viewport.offsetHeight;
    viewport.style.height = after + 'px';
    setActive(0, true);
    moreHint();
  }
  function moreHint() { list.classList.toggle('more', list.scrollTop + list.clientHeight < list.scrollHeight - 2); }
  list.addEventListener('scroll', moreHint);

  let glideAnim;
  function setActive(i, instant) {
    if (!visible.length) { q.removeAttribute('aria-activedescendant'); return; }
    const prev = visible[active] && visible[active].el;
    active = (i + visible.length) % visible.length;
    rows.forEach(r => { r.el.classList.remove('active'); r.el.setAttribute('aria-selected', 'false'); });
    const el = visible[active].el;
    el.classList.add('active'); el.setAttribute('aria-selected', 'true');
    q.setAttribute('aria-activedescendant', el.id);
    const to = el.offsetTop;
    glideAnim && glideAnim.cancel();
    glide.style.transform = `translateY(${to}px)`;
    glide.style.height = '48px';
    if (!instant && prev && prev !== el && !prev.hidden) {
      const from = prev.offsetTop, top = Math.min(from, to), span = Math.abs(to - from) + 48;
      glideAnim = glide.animate([
        { transform: `translateY(${from}px)`, height: '48px' },
        { transform: `translateY(${top}px)`, height: span + 'px', offset: 0.35 },
        { transform: `translateY(${to}px)`, height: '48px' },
      ], { duration: 420, easing: 'cubic-bezier(0.16, 1, 0.3, 1)' });
    }
    const t = el.offsetTop - 8, b = el.offsetTop + el.offsetHeight + 8;
    if (t < list.scrollTop) list.scrollTop = t;
    else if (b > list.scrollTop + list.clientHeight) list.scrollTop = b - list.clientHeight;
  }
  let statusTimer;
  function pick() {
    if (!visible.length) return;
    const r = visible[active];
    r.el.classList.add('picked');
    setTimeout(() => r.el.classList.remove('picked'), 180);
    status.querySelector('span').textContent = `Opened ${r.name}`;
    status.classList.add('show');
    clearTimeout(statusTimer);
    statusTimer = setTimeout(() => status.classList.remove('show'), 1400);
  }
  q.addEventListener('input', render);
  q.addEventListener('keydown', e => {
    if (e.key === 'ArrowDown') { e.preventDefault(); setActive(active + 1); }
    if (e.key === 'ArrowUp') { e.preventDefault(); setActive(active - 1); }
    if (e.key === 'Enter') { e.preventDefault(); pick(); }
    if (e.key === 'Escape') { q.value = ''; render(); }
  });
  list.addEventListener('mousemove', e => {
    const el = e.target.closest('.item'); if (!el) return;
    const i = visible.findIndex(r => r.el === el);
    if (i >= 0 && i !== active) setActive(i);
  });
  list.addEventListener('click', e => { if (e.target.closest('.item')) { pick(); q.focus(); } });
  viewport.style.height = viewport.offsetHeight + 'px';
  render();
  q.focus();

  // ?demo: arrows down a few rows, types "inv", opens the match, clears, loops
  const params = new URLSearchParams(location.search);
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    (async function loop() {
      q.value = ''; render(); await wait(900);
      for (let k = 0; k < 3; k++) { setActive(active + 1); await wait(420); }
      await wait(400);
      for (const ch of 'inv') { q.value += ch; render(); await wait(230); }
      await wait(600); setActive(1); await wait(600); pick(); await wait(1400);
      while (q.value) { q.value = q.value.slice(0, -1); render(); await wait(90); }
      await wait(500);
      loop();
    })();
  }
  if (params.has('q')) { q.value = params.get('q'); render(); }
</script>
