<script>
const calm = matchMedia('(prefers-reduced-motion: reduce)').matches;
const TRAVEL = 54;

document.querySelectorAll('.sw').forEach((sw, n) => {
  const knob = sw.querySelector('.knob');
  const kind = sw.dataset.kind;
  const tap = document.createElement('i');
  tap.className = 'tap';
  sw.parentElement.append(tap);
  let touched = false, step = 0;
  const isOn = () => sw.getAttribute('aria-checked') === 'true';

  function squash(on) {
    const a = on ? 0 : TRAVEL, b = on ? TRAVEL : 0, m = (a + b) / 2, over = b + (on ? 4 : -4);
    knob.style.transform = `translateX(${b}px)`;
    if (calm) return;
    knob.animate([
      { transform: `translateX(${a}px) scale(1, 1)`, easing: 'cubic-bezier(0.4, 0, 0.7, 0.6)' },
      { transform: `translateX(${m}px) scale(1.42, 0.74)`, offset: 0.34, easing: 'cubic-bezier(0.2, 0.7, 0.3, 1)' },
      { transform: `translateX(${over}px) scale(0.8, 1.18)`, offset: 0.62, easing: 'cubic-bezier(0.3, 0, 0.3, 1)' },
      { transform: `translateX(${b}px) scale(1, 1)` },
    ], { duration: 560 });
  }
  function set(on) {
    if (on === isOn()) return;
    sw.setAttribute('aria-checked', on);
    if (kind === 'squash') squash(on);
    if (kind === 'box' && !calm && on) sw.animate([{ transform: 'scale(1)' }, { transform: 'scale(0.84)', offset: 0.28 }, { transform: 'scale(1.08)', offset: 0.66 }, { transform: 'scale(1)' }], { duration: 400, easing: 'cubic-bezier(0.3, 0, 0.3, 1)' });
  }
  // a real <button>, so Space and Enter fire this as well
  sw.addEventListener('click', ev => {
    if (ev.isTrusted) touched = true;
    set(!isOn());
  });

  // 8 taps at 1500 ms is 12 s, and each one ends the way it started
  const start = isOn();
  const route = [!start, start];
  if (kind === 'squash') knob.style.transform = `translateX(${start ? TRAVEL : 0}px)`;
  function demo() {
    if (touched) return;
    const on = route[step++ % route.length];
    tap.animate([{ opacity: 0, transform: 'scale(0.6)' }, { opacity: 1, transform: 'scale(1)', offset: 0.25 }, { opacity: 0, transform: 'scale(1.25)' }], { duration: 460, easing: 'ease-out' });
    setTimeout(() => { if (!touched) set(on); }, 60);
    setTimeout(demo, 1500);
  }
  document.fonts.ready.then(() => setTimeout(demo, 60 + n * 250));
});
</script>
