/* Shared runtime for the Cursed UI pages.
   - a simulation clock (K.now, ms) advanced by K.tick(ms): rAF in the browser, stepped by the recorder (?manual)
   - tweens, easing, the hook <-> outro swap (window.__outro), and the reel cursor driven by real pointer events
   - opts.embedded: a private clock for a component mounted elsewhere (the game): no reel cursor, no globals, own rAF loop,
     K.stop() ends it. Pointer position is still tracked (K.px/K.py) */
window.ReelKit = function (opts) {
  if (opts.embedded) return ReelKit.embedded(opts);
  const qs = new URLSearchParams(location.search);
  const REEL = qs.has('reel'), MANUAL = qs.has('manual');
  if (REEL) document.body.classList.add('reel');

  const K = { REEL, MANUAL, now: 0, px: -999, py: -999, down: false };
  const tweens = [];
  K.ease = {
    lin: t => t,
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,
    back: t => { const c = 1.7; return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2); }
  };
  K.tween = (dur, fn, e = K.ease.out, delay = 0) => { const tw = { t0: K.now + delay, dur, fn, e }; tweens.push(tw); return tw; };
  K.clamp = (v, a, b) => Math.max(a, Math.min(b, v));
  K.lerp = (a, b, t) => a + (b - a) * t;
  /* critically-ish damped spring step: z = damping ratio */
  K.spring = (s, target, k, z, dt) => { s.v += (k * (target - s.x) - 2 * Math.sqrt(k) * z * s.v) * dt; s.x += s.v * dt; };

  /* ---------- cursor ---------- */

    for (let i = tweens.length - 1; i >= 0; i--) {

      if (p >= 1) tweens.splice(i, 1);
    }

      requestAnimationFrame(function loop(t) { tick(Math.min(50, t - last)); last = t; requestAnimationFrame(loop); });
    } else tick(16);
  };
  return K;
};
