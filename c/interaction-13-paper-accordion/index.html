<script>/* inlined from ../_shared/reel.js */
// Shared runtime for "Interaction #N" pages.
// Normal use: a fixed-step rAF loop drives the component (stable at any refresh rate).
// ?capture: the recorder drives time frame by frame via __tick(dt), and the overlay via __overlay(o).
window.Reel = function ({ step, render, DT = 1 / 1440, geom = () => ({}) }) {
  const CAPTURE = new URLSearchParams(location.search).has('capture');
  if (CAPTURE) document.documentElement.classList.add('reel');
  const $ = id => document.getElementById(id);
  const advance = seconds => {           // whole fixed steps only, so the sim never depends on frame timing
    const n = Math.round(seconds / DT);
    for (let i = 0; i < n; i++) step(DT);
  };
  render();
  if (!CAPTURE) {
    let last = performance.now(), acc = 0;
    const frame = now => {
      acc += Math.min(0.05, (now - last) / 1000); last = now;
      while (acc >= DT) { step(DT); acc -= DT; }
      render(); requestAnimationFrame(frame);
    };
    requestAnimationFrame(frame);
    return CAPTURE;
  }
  const cur = $('cursor');
  cur.style.transform = 'translate(-200px,-200px)';
  // a press ring that grows out of the fingertip, on simulated time (so it slows down in slow motion too)
  const rip = document.createElement('i'); rip.className = 'rip'; cur.appendChild(rip);
  let ripT = 9;
  addEventListener('pointerdown', () => { cur.classList.add('down'); ripT = 0; }, true);
  addEventListener('pointerup', () => cur.classList.remove('down'), true);
  const drawRip = () => {
    const u = Math.min(1, ripT / 0.45), e = 1 - (1 - u) ** 3;
    rip.style.transform = `scale(${(0.9 + 0.9 * e).toFixed(4)})`;
    rip.style.opacity = (u < 1 ? 0.55 * (1 - u) : 0).toFixed(3);
  };
  drawRip();
  window.__tick = sec => { advance(sec); ripT += sec; drawRip(); render(); };
  window.__geom = geom;
  window.__overlay = o => {
    // sequential, never two text layers at once: card out, then code in (and back)
    const card = $('card'), code = $('code');
    card.style.opacity = o.card;
    card.style.transform = `translateY(${(1 - o.card) * -18}px)`;
    code.style.opacity = o.code;
    code.style.transform = `translateY(${(1 - o.code) * 22}px)`;
    $('chip').style.opacity = o.chipA ?? o.card * (1 - o.big);
    $('end').style.opacity = o.code;
    $('big').style.opacity = o.big; $('big').style.transform = `scale(${0.94 + 0.06 * o.big})`;
    $('tint').style.opacity = o.tint;
    document.querySelectorAll('.hl').forEach(el => el.classList.toggle('on', !!o.hl));
    document.querySelectorAll('#code .ln').forEach((el, i) => el.classList.toggle('on', i === o.line));
    cur.style.opacity = o.cursor ?? o.card;
    $('chipText').textContent = o.chip;
    if (o.cx !== undefined) cur.style.transform = `translate(${o.cx}px, ${o.cy}px)`;
  };
  return CAPTURE;
};

(() => {
  const $ = id => document.getElementById(id);
  const reduce = matchMedia('(prefers-reduced-motion: reduce)');
  const card = $('card'), cv = $('cv'), ctx = cv.getContext('2d');
  const W = 800, H = 860, S = 2;
  const Y0 = 172, ROW = 104;                          // question rows
  const SX = 56, SW = 688, STRIP = 68, NF = 4;        // the paper sheet: four strips, three creases
  const PAD_T = 10, PAD_B = 24, F = 700, FOOT = 46;             // padding round the sheet, perspective distance
  const AC = Math.PI / 2, K = 220, C = 17;            // folded flat, spring, paper drag
  const ITEMS = [
    { q: 'Can I cancel anytime?', a: ['Yes. Cancel from Settings in two clicks:', 'no calls, no forms, no retention offers.', 'You keep access until the period ends,', 'and the receipt lands in your inbox.'] },
    { q: 'How does billing work?', a: ['One plan, billed monthly or yearly.', 'Yearly saves you two months. Prices', 'include VAT, and every invoice is a', 'real PDF you can download any time.'] },
    { q: 'Do you offer refunds?', a: ['Within 14 days, no questions asked.', 'After that we refund the unused part', 'of a yearly plan, to the card you', 'paid with. It usually takes 3 days.'] },
  ];

  let seed = 1;
  const rnd = () => { seed |= 0; seed = seed + 0x6D2B79F5 | 0; let t = Math.imul(seed ^ seed >>> 15, 1 | seed); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
  const mk = (w, h) => { const c = document.createElement('canvas'); c.width = w; c.height = h; return c; };

  // ---------- each answer printed once on its own sheet of paper (grain, faint old creases) ----------
  let sheets = null;                                  // printed once the fonts are in
  const build = () => ITEMS.map((it, j) => {
    const c = mk(SW * S, NF * STRIP * S), g = c.getContext('2d');
    g.scale(S, S);
    const bg = g.createLinearGradient(0, 0, 0, NF * STRIP);
    bg.addColorStop(0, '#F7F3EC'); bg.addColorStop(1, '#F2ECE2');
    g.fillStyle = bg; g.beginPath(); g.roundRect(0, 0, SW, NF * STRIP, 14); g.fill();
    g.save(); g.clip();
    seed = 5 + j * 31; g.lineCap = 'round';
    for (let i = 0; i < 900; i++) {
      const x = rnd() * SW, y = rnd() * NF * STRIP, a = rnd() * Math.PI, l = 6 + rnd() * 22, light = rnd() < 0.5;
      g.strokeStyle = light ? `rgba(255,255,255,${(0.3 + 0.4 * rnd()).toFixed(2)})` : `rgba(130,105,70,${(0.04 + 0.05 * rnd()).toFixed(3)})`;
      g.lineWidth = 0.6 + rnd() * 0.7;
      g.beginPath(); g.moveTo(x, y); g.quadraticCurveTo(x + Math.cos(a) * l / 2 + (rnd() - 0.5) * 4, y + Math.sin(a) * l / 2, x + Math.cos(a) * l, y + Math.sin(a) * l); g.stroke();
    }
    g.restore();
    g.strokeStyle = 'rgba(60,45,25,.10)'; g.lineWidth = 1; g.beginPath(); g.roundRect(0.5, 0.5, SW - 1, NF * STRIP - 1, 14); g.stroke();
    g.fillStyle = '#4A453E'; g.font = '400 26px "General Sans"'; g.letterSpacing = '-0.2px';
    it.a.forEach((l, i) => g.fillText(l, 34, i * STRIP + 43));
    return c;
  });

  // ---------- state ----------
  const items = ITEMS.map(() => ({ open: false, f: [...Array(NF)].map(() => ({ a: AC, v: 0 })) }));
  let act = 0, shown = 90;
  const btns = ITEMS.map((it, j) => {
    const b = document.createElement('button');
    b.className = 'q'; b.setAttribute('aria-expanded', 'false'); b.setAttribute('aria-controls', 'ans' + j);
    b.textContent = it.q;
    b.addEventListener('click', () => toggle(j));
    $('btns').appendChild(b);
    const r = document.createElement('div');
    r.className = 'sr'; r.id = 'ans' + j; r.setAttribute('role', 'region'); r.hidden = true; r.textContent = it.a.join(' ');
    $('btns').appendChild(r);
    return b;
  });
  function toggle(j) {
    const now = !items[j].open;
    items.forEach((it, k) => { it.open = k === j ? now : false; btns[k].setAttribute('aria-expanded', String(it.open)); $('ans' + k).hidden = !it.open; });
    if (act !== j) shown = items[j].f.reduce((s, f) => s + Math.min(AC, Math.max(0, f.a)), 0) / NF / AC * 90;   // a new panel: read it from where it is
    act = j;
  }

  function step(dt) {
    const target = items[act].f.reduce((s, f) => s + Math.min(AC, Math.max(0, f.a)), 0) / NF / AC * 90;
    shown += (target - shown) * (1 - Math.exp(-dt / 0.07));   // smoothed: it sweeps, never jumps
    for (const it of items) {
      const f = it.f;
      for (let i = 0; i < NF; i++) {
        // opening runs top-down (each crease waits for the one above); closing runs bottom-up
        const ready = it.open ? (!i || f[i - 1].a < 0.8) : (i === NF - 1 || f[i + 1].a > AC - 0.8);
        const want = it.open ? (ready ? 0 : AC) : (ready ? AC : Math.min(f[i].a, 0));
        if (reduce.matches) { f[i].a = it.open ? 0 : AC; f[i].v = 0; continue; }
        const k = it.open ? K : 900, c = it.open ? C : 40;                   // folding shut is quicker than opening out
        f[i].v += (want - f[i].a) * k * dt; f[i].v *= Math.exp(-c * dt); f[i].a += f[i].v * dt;
        if (f[i].a > AC) { f[i].a = AC; f[i].v = Math.min(0, -f[i].v * 0.15); }   // folded flat: it can't go further
      }
    }
  }

  // ---------- render ----------
  const openness = it => it.f.reduce((s, f) => s + Math.cos(f.a), 0) / NF;
  const panelH = it => { const o = openness(it); return o < 0.002 ? 0 : it.f.reduce((s, f) => s + STRIP * Math.cos(f.a), 0) + (PAD_T + PAD_B) * Math.min(1, o * 1.4); };
  function drawSheet(j, top) {
    if (!sheets) return;
    const it = items[j], src = sheets[j], cx = SX + SW / 2;
    let y = top + PAD_T * Math.min(1, openness(it) * 1.4), z = 0;
    const quads = [];
    for (let i = 0; i < NF; i++) {
      const a = it.f[i].a, hp = STRIP * Math.cos(a), z1 = z + (i % 2 ? -1 : 1) * STRIP * Math.sin(a);
      quads.push({ i, a, y0: y, y1: y + hp, s0: F / (F + z), s1: F / (F + z1) });
      y += hp; z = z1;
    }
    // soft contact shadow under the whole sheet, tighter as it lies flat
    const o = openness(it);
    ctx.save();
    ctx.shadowColor = `rgba(50,35,15,${(0.10 * o).toFixed(3)})`; ctx.shadowBlur = 14; ctx.shadowOffsetY = 5;
    ctx.fillStyle = 'rgba(244,239,230,1)';
    ctx.beginPath();
    quads.forEach((q, k) => { const w0 = SW / 2 * q.s0; k ? ctx.lineTo(cx - w0, q.y0) : ctx.moveTo(cx - w0, q.y0); });
    const lq = quads[NF - 1]; ctx.lineTo(cx - SW / 2 * lq.s1, lq.y1); ctx.lineTo(cx + SW / 2 * lq.s1, lq.y1);
    for (let k = NF - 1; k >= 0; k--) ctx.lineTo(cx + SW / 2 * quads[k].s0, quads[k].y0);
    ctx.closePath(); ctx.fill();
    ctx.restore();
    for (const q of quads) {
      const hp = q.y1 - q.y0;
      if (hp < 0.05) continue;
      // the strip: the printed sheet, row by row, each row as wide as its depth allows
      const N = Math.max(2, Math.ceil(STRIP * 2));
      for (let r = 0; r < N; r++) {
        const u0 = r / N, u1 = (r + 1) / N, s = q.s0 + (q.s1 - q.s0) * (u0 + u1) / 2, w = SW * s;
        ctx.drawImage(src, 0, (q.i * STRIP + u0 * STRIP) * S, SW * S, STRIP / N * S, cx - w / 2, q.y0 + u0 * hp, w, hp / N + 0.35);
      }
      // light: strips tipped towards the top light catch it, strips tipped away fall into shade, darkest at the valley
      const sn = Math.sin(q.a), up = (q.i % 2 === 0) === (q.a >= 0);
      const w0 = SW / 2 * q.s0, w1 = SW / 2 * q.s1;
      ctx.beginPath(); ctx.moveTo(cx - w0, q.y0); ctx.lineTo(cx + w0, q.y0); ctx.lineTo(cx + w1, q.y1); ctx.lineTo(cx - w1, q.y1); ctx.closePath();
      const g = ctx.createLinearGradient(0, q.y0, 0, q.y1), m = Math.abs(sn);
      if (up) { g.addColorStop(0, `rgba(255,253,248,${(0.25 * m).toFixed(3)})`); g.addColorStop(1, `rgba(255,252,245,${(0.85 * m).toFixed(3)})`); }
      else { g.addColorStop(0, `rgba(70,50,25,${(0.16 * m).toFixed(3)})`); g.addColorStop(1, `rgba(60,40,18,${(0.58 * m).toFixed(3)})`); }
      ctx.fillStyle = g; ctx.fill();
    }
    // the creases: a hairline that deepens as the fold closes; the paper remembers them even when flat
    for (let k = 1; k < NF; k++) {
      const q = quads[k], m = Math.max(Math.abs(Math.sin(quads[k - 1].a)), Math.abs(Math.sin(q.a)));
      const w = SW / 2 * q.s0 - 3;
      const valley = k % 2 === 0;
      ctx.strokeStyle = valley ? `rgba(70,50,25,${(0.07 + 0.22 * m).toFixed(3)})` : `rgba(255,255,255,${(0.35 + 0.5 * m).toFixed(3)})`;
      ctx.lineWidth = 1;
      ctx.beginPath(); ctx.moveTo(cx - w, q.y0 + 0.5); ctx.lineTo(cx + w, q.y0 + 0.5); ctx.stroke();
      if (!valley) { ctx.strokeStyle = `rgba(70,50,25,${(0.05 + 0.12 * m).toFixed(3)})`; ctx.beginPath(); ctx.moveTo(cx - w, q.y0 + 1.5); ctx.lineTo(cx + w, q.y0 + 1.5); ctx.stroke(); }
    }
  }
  function chevron(x, y, rot, on) {
    ctx.save(); ctx.translate(x, y); ctx.scale(1, Math.cos(rot) || 0.001);   // it flips over like the sheet does
    ctx.strokeStyle = on ? '#D9502B' : '#57524A'; ctx.lineWidth = 2.6; ctx.lineCap = 'round'; ctx.lineJoin = 'round';
    ctx.beginPath(); ctx.moveTo(-10, -5); ctx.lineTo(0, 5); ctx.lineTo(10, -5); ctx.stroke();
    ctx.restore();
  }
  function render() {
    ctx.setTransform(S, 0, 0, S, 0, 0);
    ctx.clearRect(0, 0, W, H);
    // the card is exactly as tall as what's in it: it grows and shrinks with the (spring-driven) sheets
    const CH = Math.min(H, Y0 + ITEMS.length * ROW + items.reduce((s, it) => s + panelH(it), 0) + FOOT);
    card.style.height = CH.toFixed(2) + 'px';
    ctx.save(); ctx.beginPath(); ctx.roundRect(0, 0, W, CH, 36); ctx.clip();
    let y = Y0;
    ctx.strokeStyle = 'rgba(40,32,20,.12)'; ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(SX, y + 0.5); ctx.lineTo(W - SX, y + 0.5); ctx.stroke();
    items.forEach((it, j) => {
      const o = openness(it);
      btns[j].style.top = y + 'px';
      ctx.fillStyle = o > 0.02 ? '#D9502B' : '#9A948A'; ctx.font = '400 22px "Geist Mono"'; ctx.fillText('0' + (j + 1), SX, y + 61);
      ctx.fillStyle = '#1D1B18'; ctx.font = '500 32px "General Sans"'; ctx.letterSpacing = '-0.5px';
      ctx.fillText(ITEMS[j].q, SX + 56, y + 63); ctx.letterSpacing = '0px';
      chevron(W - SX - 14, y + 53, Math.PI * Math.min(1, o), o > 0.02);
      y += ROW;
      const ph = panelH(it);
      if (ph > 0) drawSheet(j, y);
      y += ph;
      ctx.strokeStyle = 'rgba(40,32,20,.12)'; ctx.lineWidth = 1;
      ctx.beginPath(); ctx.moveTo(SX, y + 0.5); ctx.lineTo(W - SX, y + 0.5); ctx.stroke();
    });
    ctx.restore();
    $('num').textContent = Math.round(shown);
    $('rl').textContent = 'fold angle · 0' + (act + 1);
    $('read').classList.toggle('live', shown < 89.5);
  }

  Reel({
    step, render,
    geom: () => {
      const r = card.getBoundingClientRect(), k = r.width / W;
      return { left: r.left, top: r.top, k, y0: Y0, row: ROW, panel: NF * STRIP + PAD_T + PAD_B, chev: W - SX - 14 };
    },
  });
  document.fonts.ready.then(() => { sheets = build(); render(); });
})();
