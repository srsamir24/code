<script>
  lucide.createIcons({ attrs: { 'stroke-width': 2.2 } });
  const panel = document.getElementById('panel');
  const live = document.getElementById('live');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const STRIPS = 22, UNDO_MS = 5000, ROW_H = 80, LINE_H = 54;
  const LABEL = 'Hold<span class="more"> to delete</span>';
  let pending = null;            // { row, holders, anims, close, timer } while Undo is on offer
  const say = text => { live.textContent = ''; setTimeout(() => live.textContent = text, 30); };

  // hold: the CSS transition does the timing, transitionend means "held long enough"
  function press(btn) {
    if (btn.classList.contains('holding') || btn.closest('.gone')) return;
    btn.classList.remove('nudge');
    btn.classList.add('holding');
  }
  function letGo(btn) {
    if (!btn.classList.contains('holding')) return;
    btn.classList.remove('holding');          // fill snaps back in 180ms
    if (!reduce) { btn.offsetWidth; btn.classList.add('nudge'); }
    hint(btn);
  }
  // released early: a small shake and the label says what to do
  function hint(btn) {
    clearTimeout(btn.hintTimer);
    swapLabel(btn, 'Keep holding');
    say('Keep holding to delete');
    btn.hintTimer = setTimeout(() => swapLabel(btn, LABEL), 1400);
  }
  function swapLabel(btn, html) {
    btn.classList.add('swap');
    setTimeout(() => {
      btn.querySelectorAll('.label').forEach(l => l.innerHTML = html);
      btn.classList.remove('swap');
    }, 160);
  }

  panel.querySelectorAll('.hold').forEach(btn => {
    // primary button only: a right-click or ctrl-click never starts a delete
    btn.addEventListener('pointerdown', e => { if (e.isPrimary && e.button === 0 && !e.ctrlKey) press(btn); });
    btn.addEventListener('pointerup', () => letGo(btn));
    btn.addEventListener('pointerleave', () => letGo(btn));
    btn.addEventListener('pointercancel', () => letGo(btn));
    btn.addEventListener('contextmenu', e => e.preventDefault());
    // keyboard: holding Space or Enter works the same
    btn.addEventListener('keydown', e => {
      if (e.key !== ' ' && e.key !== 'Enter') return;
      e.preventDefault(); if (!e.repeat) press(btn);
    });
    btn.addEventListener('keyup', e => { if (e.key === ' ' || e.key === 'Enter') letGo(btn); });
    btn.addEventListener('blur', () => letGo(btn));
    btn.querySelector('.fill').addEventListener('transitionend', e => {
      if (e.propertyName === 'clip-path' && btn.classList.contains('holding')) shred(btn.closest('.row'));
    });
  });

  // the shredder: 22 inert clones of the row, each clipped to one vertical strip,
  // twice over: once above the slot (the row's bottom edge) and once below it
  function cutStrip(row, i, w) {
    const l = (w / STRIPS) * i, r = w - l - w / STRIPS;
    const strip = document.createElement('div');
    strip.className = 'strip';
    strip.style.width = w + 'px';
    const cut = document.createElement('div');
    cut.className = 'cut';
    const copy = row.cloneNode(true);
    copy.inert = true;
    copy.querySelector('.undo')?.remove();
    cut.style.clipPath = `inset(0 ${r}px 0 ${l}px)`;
    cut.appendChild(copy);
    strip.appendChild(cut);
    return strip;
  }
  function holder(cls, top, left, w, h) {
    const el = document.createElement('div');
    el.className = 'slot ' + cls; el.inert = true;
    el.style.cssText = `top:${top}px;left:${left}px;width:${w}px;height:${h}px`;
    panel.appendChild(el);
    return el;
  }

  function feed(row) {
    const w = row.offsetWidth, h = row.offsetHeight, D = h * 1.7;
    const up = holder('above', row.offsetTop, row.offsetLeft, w, h);
    const down = holder('below', row.offsetTop + h, row.offsetLeft, w, D);
    const anims = [];
    for (let i = 0; i < STRIPS; i++) {
      const cx = (w / STRIPS) * (i + 0.5);
      const splay = (i - (STRIPS - 1) / 2) * 0.5;          // outer ribbons lean out a little
      const timing = { duration: 900, delay: i * 13, fill: 'forwards' };   // a shredder feeds at a steady speed
      const top = cutStrip(row, i, w), below = cutStrip(row, i, w);
      below.style.top = -h + 'px';
      up.appendChild(top); down.appendChild(below);
      anims.push(top.animate([{ transform: 'none' }, { transform: `translateY(${D}px)` }], timing));
      // below the slot: the same strip, splayed by a skew
      // that pivots on the slot line as it feeds through
      anims.push(below.animate([
        { transform: 'none', transformOrigin: `${cx}px ${h}px` },
        { transform: `translateY(${D}px) skewX(${splay}deg)`,
          transformOrigin: `${cx}px ${h - D}px` },
      ], timing));
      anims.push(below.animate([{ opacity: 1 }, { opacity: 1, offset: 0.75 }, { opacity: 0 }], timing));
    }
    return { holders: [up, down], anims };
  }

  function shred(row) {
    if (pending) finish();
    const btn = row.querySelector('.hold');
    const hadFocus = btn === document.activeElement;
    // reduced motion: no strips, the row just crossfades to the Undo line
    const { holders, anims } = reduce ? { holders: [], anims: [] } : feed(row);
    btn.classList.remove('holding');               // after cloning, so the strips keep the red fill
    row.classList.add('gone');
    // the Undo line is uncovered as the strips go through; the row tightens once they are gone
    const close = row.animate([{ height: ROW_H + 'px' }, { height: LINE_H + 'px' }],
      { duration: reduce ? 1 : 380, delay: reduce ? 0 : 900, easing: 'cubic-bezier(0.77, 0, 0.175, 1)', fill: 'forwards' });
    const name = row.querySelector('.name b').textContent;
    row.querySelector('.undo b').textContent = name;
    setCount(-1);
    say(`${name} deleted. Undo available for 5 seconds.`);
    pending = { row, name, holders, anims, close, timer: setTimeout(finish, UNDO_MS) };
    if (hadFocus) row.querySelector('.undo button').focus();
  }

  // after 5s the line folds away for good
  function finish() {
    const { row, holders, timer } = pending; pending = null;
    clearTimeout(timer);
    holders.forEach(s => s.remove());
    const fold = row.animate([{ height: LINE_H + 'px', opacity: 1 }, { height: '0px', opacity: 0 }],
      { duration: reduce ? 1 : 300, easing: 'cubic-bezier(0.77, 0, 0.175, 1)', fill: 'forwards' });
    fold.onfinish = () => {
      const lost = row.contains(document.activeElement);
      row.remove();
      const next = panel.querySelector('.row .hold');
      if (!next) {                                            // nothing left: say so, focus the heading
        const empty = document.createElement('p');
        empty.className = 'empty'; empty.textContent = 'No projects. Nothing to delete.';
        document.getElementById('list').appendChild(empty);
        document.getElementById('title').focus();
      } else if (lost) next.focus();                          // focus never drops to the page
    };
  }

  // undo: the row opens again and the strips come back up through the slot
  function undo() {
    if (!pending) return;
    const { row, name, holders, anims, close, timer } = pending; pending = null;
    clearTimeout(timer);
    row.classList.add('restoring');
    [close, ...anims].forEach(a => { a.updatePlaybackRate(-1.8); a.play(); });
    setCount(+1);
    say(`${name} restored`);
    Promise.all([close, ...anims].map(a => a.finished)).then(() => {
      const a = document.activeElement;
      const refocus = a === document.body || row.contains(a);   // never steal focus from another row
      row.classList.remove('gone', 'restoring');
      holders.forEach(s => s.remove());
      close.cancel();
      if (refocus) row.querySelector('.hold').focus({ preventScroll: true });
    });
  }

  // the project count rolls instead of swapping
  let total = 3;
  function setCount(d) {
    const box = document.getElementById('count');
    // a roll still running is finished first, so a fast Undo never leaves two numbers
    box.getAnimations({ subtree: true }).forEach(a => a.cancel());
    [...box.children].slice(0, -1).forEach(c => c.remove());
    const old = box.firstElementChild, next = document.createElement('span');
    old.style.position = '';
    total += d; next.textContent = total;
    box.appendChild(next);
    next.style.position = 'absolute'; next.style.inset = '0';
    const dir = d < 0 ? -1 : 1;
    const t = { duration: reduce ? 1 : 320, easing: 'cubic-bezier(0.23, 1, 0.32, 1)' };
    old.animate([{ transform: 'none', opacity: 1 }, { transform: `translateY(${dir * -100}%)`, opacity: 0 }], t).onfinish = () => old.remove();
    next.animate([{ transform: `translateY(${dir * 100}%)`, opacity: 0 }, { transform: 'none', opacity: 1 }], t).onfinish = () => { next.style.position = ''; };
  }

  // every row gets an undo line; the live region above does the talking
  panel.querySelectorAll('.row').forEach(row => {
    const u = document.createElement('div');
    u.className = 'undo';
    u.innerHTML = `<span><b></b> deleted</span>
      <button type="button">Undo</button>
      <svg class="ring" viewBox="0 0 18 18" aria-hidden="true"><circle class="track" cx="9" cy="9" r="7"/><circle class="left" cx="9" cy="9" r="7"/></svg>`;
    u.querySelector('button').addEventListener('click', undo);
    row.appendChild(u);
  });

  // stills and the demo loop
  const params = new URLSearchParams(location.search);
  const still = params.get('still');
  const north = panel.querySelectorAll('.row')[2];
  if (still === 'hold') {
    const btn = north.querySelector('.hold'), fill = btn.querySelector('.fill');
    btn.classList.add('holding');
    fill.style.transition = 'none';
    fill.style.clipPath = 'inset(0 35% 0 0)';      // 65% of the way there
  }
  if (still === 'shred') {
    north.querySelector('.hold').classList.add('holding');
    shred(north);
    clearTimeout(pending.timer);
    const t = Number(params.get('t') || 520);       // about 60% through the feed
    [...pending.anims, pending.close].forEach(a => { a.pause(); a.currentTime = t; });
    const ring = north.querySelector('.ring .left');
    ring.style.animation = 'none'; ring.style.strokeDashoffset = 44 * 0.3;   // 30% drained
  }
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    const btn = north.querySelector('.hold');
    (async function loop() {
      await wait(900);
      press(btn); await wait(650); letGo(btn);
      await wait(1500);
      press(btn); await wait(2600);
      await wait(1400); undo();
      await wait(1800);
      loop();
    })();
  }
</script>
