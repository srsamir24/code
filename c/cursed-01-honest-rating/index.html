<script>
(() => {
  const qs = new URLSearchParams(location.search);
  const REEL = qs.has('reel'), MANUAL = qs.has('manual');
  if (REEL) document.body.classList.add('reel');

  const $ = id => document.getElementById(id);
  const track = $('track'), label = $('label'), note = $('note'), mirror = $('mirror'), field = $('field');
  const form = $('form'), done = $('done'), outro = $('outro');

  /* ---------- star markup ---------- */
  const STAR = 'M25 3.2l6.5 13.4 14.7 2.1-10.7 10.3 2.6 14.6L25 36.7l-13.1 6.9 2.6-14.6L3.8 18.7l14.7-2.1z';
  const starSvg = () => `<svg viewBox="0 0 50 50">
      <path d="${STAR}" fill="#faf9f6" stroke="#d9d5cc" stroke-width="1.5" stroke-linejoin="round"/>
      <g class="on"><path d="${STAR}" fill="url(#gold)" stroke="#e39a32" stroke-width="1" stroke-linejoin="round"/>
      <path d="M25 8.5l4.6 9.4" stroke="#fff" stroke-opacity=".55" stroke-width="2.2" stroke-linecap="round"/></g></svg>`;
  document.body.insertAdjacentHTML('beforeend',`<svg width="0" height="0" style="position:absolute"><defs>
    <linearGradient id="gold" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#ffdb7a"/><stop offset="1" stop-color="#f29d34"/></linearGradient></defs></svg>`);

  const GAP = 66, N = 5, LEAD = N - 1, PERIOD = GAP * N;   // the row wraps around like a carousel
  const stars = [];
  const mk = (i, twin) => {
    const b = document.createElement('button');
    b.className = 'star'; b.type = 'button';
    if (twin) { b.tabIndex = -1; b.setAttribute('aria-hidden','true'); }
    else { b.setAttribute('role','radio'); b.setAttribute('aria-label','5 stars'); }  // every star is honest about it
    b.dataset.value = '5';
    b.innerHTML = starSvg();
    track.appendChild(b);
    return b;
  };
  for (let i = 0; i < N; i++) {
    const el = mk(i, false), tw = mk(i, true);
    const rest = (i - (N - 1) / 2) * GAP;
    stars.push({ el, tw, on: el.querySelector('.on'), on2: tw.querySelector('.on'), rest, x: rest, v: 0,
                 k: 560 - (LEAD - i) * 80,          // the 5th star leads, the rest trail behind it
                 f: 0, fv: 0, delay: 0 });
  }
  const WORDS = ['Terrible', 'Meh', 'Okay', 'Good', 'Amazing'];
  let dodgeAt = 0;
  $('big').innerHTML = Array.from({length:5}, () => `<svg viewBox="0 0 50 50"><path d="${STAR}" fill="url(#gold)" stroke="#e39a32" stroke-width="1" stroke-linejoin="round"/><path d="M25 8.5l4.6 9.4" stroke="#fff" stroke-opacity=".55" stroke-width="2.2" stroke-linecap="round"/></svg>`).join('');
  const bigStars = [...$('big').children];

  /* ---------- state ---------- */
  let now = 0;                 // simulation clock (ms)
  let px = -999, py = -999;    // pointer
  let hovering = false, rated = false, focused = false;
  let lastType = -1e9;
  const tweens = [];
  const ease = {
    out: t => 1 - Math.pow(1 - t, 3),
    inOut: t => t < .5 ? 4*t*t*t : 1 - Math.pow(-2*t + 2, 3) / 2,
    back: t => { const c = 1.7; return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2); }
  };
  function tween(dur, fn, e = ease.out, delay = 0) {
    const tw = { t0: now + delay, dur, fn, e }; tweens.push(tw); return tw;
  }

  /* ---------- pointer ---------- */
  function hoverZone() {
    const r = track.getBoundingClientRect();
    return px > r.left + 6 && px < r.right - 6 && py > r.top - 10 && py < r.bottom + 10;
  }
  window.addEventListener('pointermove', e => { px = e.clientX; py = e.clientY; moveCursor(); });
  window.addEventListener('pointerleave', () => { px = py = -999; });
  track.addEventListener('pointerdown', () => rate());
  stars.forEach(s => s.el.addEventListener('keydown', e => { if (e.key === 'Enter' || e.key === ' ') rate(); }));

  function rate() {
    rated = true;
    stars.forEach((s, i) => { s.fv += 9 + (LEAD - i) * 1.5; });   // little pop
    setLabel('You picked 5. Great choice.', true);
  }
  function setLabel(text, hot) {
    if (label.textContent === text) return;
    label.textContent = text; label.classList.toggle('hot', hot);
    tween(260, p => { label.style.transform = `translateY(${(1 - p) * 5}px)`; label.style.opacity = p; });
  }

  /* ---------- the "autocorrect" ---------- */
  const FIX = { terrible:'terrific', awful:'awesome', bad:'great', hate:'love', slow:'fast', buggy:'flawless',
                worst:'best', broken:'perfect', meh:'wow', useless:'priceless', annoying:'amazing', crashes:'delights' };
  let morph = null, mark = null;   // mark = {start,len,t}
  note.addEventListener('focus', () => { focused = true; field.classList.add('focus'); });
  note.addEventListener('blur',  () => { focused = false; field.classList.remove('focus'); });
  note.addEventListener('input', () => {
    lastType = now;
    const m = note.value.match(/([A-Za-z]+)([^A-Za-z])$/);
    if (m && FIX[m[1].toLowerCase()] && !morph) {
      const start = note.value.length - m[0].length;
      let to = FIX[m[1].toLowerCase()];
      if (m[1][0] === m[1][0].toUpperCase()) to = to[0].toUpperCase() + to.slice(1);
      morph = { start, from: m[1], to, cur: m[1], t0: now + 110 };
    }
  });

  function stepMorph() {
    if (!morph) return;
    const k = Math.floor((now - morph.t0) / 52);
    if (k < 0) return;
    const { from, to, start } = morph;
    const L = Math.max(from.length, to.length);
    const word = k >= L ? to : to.slice(0, k + 1) + from.slice(k + 1, Math.max(from.length, k + 1));
    if (word !== morph.cur) {
      const v = note.value;
      note.value = v.slice(0, start) + word + v.slice(start + morph.cur.length);
      morph.cur = word;
      note.setSelectionRange(note.value.length, note.value.length);
      mark = { start, len: word.length, t: now };
    }
    if (k >= L) { mark = { start, len: to.length, t: now }; morph = null; }
  }

  function renderMirror() {
    const v = note.value;
    const esc = s => s.replace(/&/g,'&amp;').replace(/</g,'&lt;');
    let html;
    if (mark && now - mark.t < 1400) {
      const a = Math.max(0, 1 - Math.max(0, now - mark.t - 500) / 900);
      html = esc(v.slice(0, mark.start)) +
        `<mark style="background:rgba(255,205,96,${.55*a});box-shadow:inset 0 -2px 0 rgba(226,150,40,${a})">${esc(v.slice(mark.start, mark.start + mark.len))}</mark>` +
        esc(v.slice(mark.start + mark.len));
    } else html = esc(v);
    const blinkOn = (now - lastType) < 500 || Math.floor(now / 530) % 2 === 0;
    if (focused && blinkOn) html += '<span class="caret"></span>';
    mirror.innerHTML = html;
  }

  /* ---------- submit / reset ---------- */
  let doneShown = false;
  function press(el) {
    tween(110, p => el.style.transform = `scale(${1 - .04 * p})`);
    tween(380, p => el.style.transform = `scale(${.96 + .04 * p})`, ease.back, 110);
  }
  ['submit','later'].forEach(id => $(id).addEventListener('pointerdown', () => press($(id))));
  $('submit').addEventListener('click', () => { if (!rated) rate(); showDone(); });
  $('later').addEventListener('click', () => { rate(); showDone(); });   // "Not now" also counts as 5
  $('again').addEventListener('click', () => reset());

  function showDone() {
    if (doneShown) return; doneShown = true; note.blur();
    tween(260, p => { form.style.opacity = 1 - p; form.style.transform = `translateY(${-8 * p}px)`; });
    done.style.pointerEvents = 'auto';
    tween(420, p => { done.style.opacity = p; done.style.transform = `translateY(${10 * (1 - p)}px)`; }, ease.out, 160);
    bigStars.forEach((s, i) => { s.style.transform = 'scale(0)';
      tween(520, p => s.style.transform = `scale(${p}) rotate(${(1 - p) * -40}deg)`, ease.back, 260 + i * 70); });
  }
  function reset() {
    if (!doneShown) return;
    doneShown = false; done.style.pointerEvents = 'none';
    rated = false; note.value = ''; mark = morph = null;
    stars.forEach(s => { s.f = 0; s.fv = 0; });
    label.textContent = 'Tap to rate'; label.classList.remove('hot'); label.style.opacity = 1; label.style.transform = '';
    tween(260, p => { done.style.opacity = 1 - p; });
    tween(300, p => { form.style.opacity = p; form.style.transform = `translateY(${6 * (1 - p)}px)`; }, ease.out, 40);
  }
  let outroOn = false;
  window.__outro = on => {
    if (on === outroOn) return; outroOn = on;
    const hook = document.querySelector('.hook');
    tween(on ? 420 : 320, p => { const q = on ? p : 1 - p;
      hook.style.opacity = 1 - q;
      outro.style.opacity = q; outro.style.transform = `translateY(${(1 - q) * 10}px)`; });
  };
  window.__reset = reset;

  /* ---------- physics ---------- */
  function update(dt) {
    const wasHover = hovering;
    hovering = hoverZone() && !doneShown;
    const tr = track.getBoundingClientRect();
    const local = px - (tr.left + tr.width / 2);
    if (hovering && !wasHover) {
      dodgeAt = now + 260;                                       // behave honestly for a beat...
      stars.forEach((s, i) => s.delay = dodgeAt + 120 + (LEAD - i) * 45);
    }
    if (!hovering && wasHover && !rated) setLabel('Tap to rate', false);
    const dodging = hovering && now >= dodgeAt;
    const offset = dodging ? local - stars[LEAD].rest : 0;     // ...then star #5 always lands under the pointer
    const honestIdx = Math.max(0, Math.min(N - 1, Math.round(local / GAP + (N - 1) / 2)));
    if (hovering && !rated) setLabel(dodging ? '5 stars · Amazing' : WORDS[honestIdx], dodging);

    stars.forEach((s, i) => {
      const k = s.k, c = 2 * Math.sqrt(k) * .72;
      const a = k * (s.rest + offset - s.x) - c * s.v;
      s.v += a * dt; s.x += s.v * dt;
      const ft = rated || (dodging ? now >= s.delay : hovering && i <= honestIdx) ? 1 : 0;
      const fk = 340, fc = 2 * Math.sqrt(fk) * .45;
      s.fv += (fk * (ft - s.f) - fc * s.fv) * dt; s.f += s.fv * dt;
    });
  }
  const wrap = x => ((x + PERIOD / 2) % PERIOD + PERIOD) % PERIOD - PERIOD / 2;
  function render() {
    for (const s of stars) {
      const sp = Math.min(Math.abs(s.v) / 2600, .3);
