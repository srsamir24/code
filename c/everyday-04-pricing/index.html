<script>
// yearly is ten months of the monthly price, so "2 months free" is true of every plan,
// and these divide by 12 exactly: 120, 300 and 780 a year is 10, 25 and 65 a month
const PLANS = [
  { id: 'solo', name: 'Solo', line: 'For one person sending invoices', month: 12, feats: ['30 invoices a month', 'Card and bank payments', 'Email reminders'] },
  { id: 'studio', name: 'Studio', line: 'For a small team with regular clients', month: 30, feats: ['Up to 8 people', 'Unlimited invoices', 'Automatic late fees'] },
  { id: 'agency', name: 'Agency', line: 'For many clients and many hands', month: 78, feats: ['Unlimited people', 'Client payment portals', 'Priority support'] },
];
const $ = s => document.querySelector(s);
const calm = matchMedia('(prefers-reduced-motion: reduce)').matches;
const plansBox = $('#plans'), bill = $('#bill'), cta = $('#cta'), ctaText = $('#ctaText');
const TICK = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="m5 12.5 4.6 4.6L19 7.5"/></svg>';
const DIGITS = Array.from({ length: 10 }, (_, d) => `<i>${d}</i>`).join('');

plansBox.innerHTML = PLANS.map(p => `<button class="plan" id="${p.id}" aria-pressed="false">
  <span class="top"><span><span class="pname">${p.name}</span><span class="pline">${p.line}</span></span>
  <span class="price"><span class="amount"><span class="cur">$</span><span class="w"><span class="s">${DIGITS}</span></span><span class="w"><span class="s">${DIGITS}</span></span></span><span class="per"></span></span></span>
  <span class="more"><span>${p.feats.map((f, i) => `<span class="feat" style="--i:${i}">${TICK}${f}</span>`).join('')}</span></span>
</button>`).join('');

let yearly = false, chosen = 'studio', token = 0, busy = false;
const perMonth = p => yearly ? Math.round(p.month * 10 / 12) : p.month;
const perLabel = p => yearly ? `$${p.month * 10} a year` : 'per month';

// the old label stays on top of the new one while they swap, so the button is never empty
function swapLabel(html) {
  if (!calm && ctaText.innerHTML) {
    const old = ctaText.cloneNode(true);
    old.removeAttribute('id');
    old.style.pointerEvents = 'none';
    cta.append(old);
    old.animate([{ opacity: 1 }, { opacity: 0 }], { duration: 90, easing: 'linear', fill: 'forwards' }).onfinish = () => old.remove();
    ctaText.animate([{ opacity: 0, transform: 'translateY(6px)' }, { opacity: 1, transform: 'translateY(0)' }], { duration: 170, delay: 60, fill: 'backwards', easing: 'cubic-bezier(0.23, 1, 0.32, 1)' });
  }
  ctaText.innerHTML = html;
}
function paintCta(animate) {
  const p = PLANS.find(x => x.id === chosen);
  const text = `Start ${p.name}, ${yearly ? `$${p.month * 10} a year` : `$${p.month} a month`}`;
  if (animate) swapLabel(text); else ctaText.textContent = text;
}
function paintPrices() {
  PLANS.forEach((p, row) => {
    const el = document.getElementById(p.id);
    const text = String(perMonth(p)).padStart(2, '0');
    el.querySelectorAll('.s').forEach((s, k) => {
      s.style.setProperty('--d', row * 40 + k * 50 + 'ms');
      s.style.transform = `translateY(${-Number(text[k]) * 10}%)`;
    });
    el.querySelector('.per').textContent = perLabel(p);
    el.querySelector('.amount').setAttribute('aria-label', `$${perMonth(p)} a month`);
  });
}
function setPeriod(y) {
  if (y === yearly) return;
  yearly = y;
  bill.classList.toggle('yr', y);
  $('#mo').setAttribute('aria-pressed', !y);
  $('#yr').setAttribute('aria-pressed', y);
  paintPrices();
  resetCta();
  paintCta(true);
}
function choose(id) {
  if (id === chosen) return;
  chosen = id;
  PLANS.forEach(p => document.getElementById(p.id).setAttribute('aria-pressed', p.id === id));
  resetCta();
  paintCta(true);
}
function resetCta() { token++; busy = false; cta.classList.remove('busy', 'ok'); }
function start() {
  if (busy || cta.classList.contains('ok')) return;
  const t = ++token, p = PLANS.find(x => x.id === chosen);
  busy = true;
  cta.classList.add('busy');
  ctaText.innerHTML = `<svg class="ring" viewBox="0 0 26 26" aria-hidden="true"><circle cx="13" cy="13" r="10" stroke-dasharray="44 20"/></svg>Starting ${p.name}`;
  setTimeout(() => {
    if (t !== token) return;
    busy = false;
    cta.classList.remove('busy'); cta.classList.add('ok');
    swapLabel(`You're on ${p.name}`);
    setTimeout(() => { if (t === token) { cta.classList.remove('ok'); paintCta(true); } }, 1000);
  }, 800);
}

$('#mo').addEventListener('click', () => setPeriod(false));
$('#yr').addEventListener('click', () => setPeriod(true));
PLANS.forEach(p => document.getElementById(p.id).addEventListener('click', () => choose(p.id)));
cta.addEventListener('click', start);

document.getElementById('studio').setAttribute('aria-pressed', 'true');
// the first paint has no transition to run, so the wheels start on the right digit
paintPrices();
paintCta(false);
</script>
