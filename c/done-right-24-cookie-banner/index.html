<script id="component">
  // Bump `version` when the purposes change: everyone is asked again.
  const CONSENT = { version: 2, maxAge: 365 * 24 * 60 * 60 * 1000 };

  const readConsent = (key) => {
    try {
      const c = JSON.parse(localStorage.getItem(key));
      const fresh = c && Date.now() - c.at < CONSENT.maxAge;
      if (fresh && c.v === CONSENT.version) return c;
    } catch {}
    return null;   // not asked yet, new purposes, or a year old
  };

  const saveConsent = (key, choice) => {
    const c = { v: CONSENT.version, ...choice, at: Date.now() };
    try { localStorage.setItem(key, JSON.stringify(c)); } catch {}
    return c;
  };

  // A tag runs once, and only if its purpose got a yes.
  const runAllowed = (c) => {
    for (const tag of document.querySelectorAll('script[data-consent]')) {
      if (!c[tag.dataset.consent]) continue;
      const s = document.createElement('script');
      if (tag.dataset.src) s.src = tag.dataset.src;
      else s.textContent = tag.textContent;
      tag.replaceWith(s);
    }
  };

  const parts = (banner) => ({
    form: banner.querySelector('.consent-form'),
    choose: banner.querySelector('.consent-choose'),
    more: banner.querySelector('.consent-more[type="button"]'),
    save: banner.querySelector('.consent-more[value="save"]'),
    done: banner.querySelector('.consent-done'),
    msg: banner.querySelector('.consent-msg'),
    boxes: [...banner.querySelectorAll('.consent-choose input[name]')],
  });

  const showChoices = (banner, c = {}) => {
    const p = parts(banner);
    p.boxes.forEach((b) => { b.checked = !!c[b.name]; });   // never pre-ticked
    p.choose.hidden = p.save.hidden = false;
    p.more.hidden = true;
    p.more.setAttribute('aria-expanded', 'true');
  };

  for (const banner of document.querySelectorAll('.consent')) {
    const key = banner.dataset.key || 'consent';
    const p = parts(banner);
    const stored = readConsent(key);
    if (stored) runAllowed(stored);
    else banner.hidden = false;   // shown, but focus stays where it is

    p.more.addEventListener('click', () => {
      showChoices(banner);
      p.boxes[0].focus();
    });

    p.form.addEventListener('submit', (e) => {
      e.preventDefault();
      const how = e.submitter.value;   // reject | accept | save
      const choice = Object.fromEntries(p.boxes.map((b) =>
        [b.name, how === 'accept' || (how === 'save' && b.checked)]));
      const c = saveConsent(key, choice);
      runAllowed(c);   // a withdrawn yes stops on the next page load
      const on = p.boxes.filter((b) => c[b.name]).map((b) => b.closest('label').querySelector('b').textContent.toLowerCase());
      p.form.hidden = true;
      p.done.hidden = false;
      p.msg.textContent = (how === 'reject' ? 'You rejected all optional cookies.'
        : how === 'accept' ? 'You accepted all cookies.'
        : on.length ? `Saved: necessary + ${on.join(' + ')}.` : 'Saved: necessary cookies only.')
        + ' Change it any time in Cookie settings.';
      p.msg.focus();
    });

    banner.querySelector('.consent-hide').addEventListener('click', () => {
      banner.hidden = true;
    });
  }

  // Changing your mind: the same banner, your current choice ticked.
  for (const open of document.querySelectorAll('.consent-open')) {
    open.addEventListener('click', () => {
      const banner = document.getElementById(open.getAttribute('aria-controls'));
      const p = parts(banner);
      showChoices(banner, readConsent(banner.dataset.key || 'consent') || {});
      p.form.hidden = false;
      p.done.hidden = true;
      banner.hidden = false;
      banner.querySelector('.consent-title').focus();
    });
  }
</script>
