<script id="component">
  // Back goes one step up. It never cancels anything.
  const BACK = {
    what: 'billing', pause: 'what', confirm: 'what',
  };

  for (const sub of document.querySelectorAll('.sub')) {
    const panel = (s) => sub.querySelector(`.sub-panel[data-step="${s}"]`);
    const status = sub.querySelector('.sub-status');
    const trigger = panel('billing').querySelector('[data-go="what"]');
    const error = sub.querySelector('.sub-error');
    const ends = sub.dataset.ends;
    let step = 'billing';

    // Each heading is described by its "Step 1 of 2".
    for (const p of sub.querySelectorAll('.sub-panel')) {
      const count = p.querySelector('.sub-count');
      if (!count) continue;
      count.id = `${sub.id}-${p.dataset.step}-count`;
      p.querySelector('.sub-title').setAttribute('aria-describedby', count.id);
    }
    if (error) {
      error.id = `${sub.id}-pause-error`;
      sub.querySelector('.sub-months').setAttribute('aria-describedby', error.id);
    }

    // Show one step and move focus to its heading.
    const show = (next, focus) => {
      panel(step).hidden = true;
      panel(next).hidden = false;
      step = sub.dataset.step = next;
      (focus || panel(next).querySelector('.sub-title')).focus();
    };

    const say = {
      'confirm>billing': `Nothing changed. Pro renews ${ends}.`,
      'cancelled>billing': `Cancellation undone. Pro renews ${ends}.`,
      'paused>billing': `Pause ended. Pro renews ${ends}.`,
    };

    const go = (next) => {
      if (next === 'paused') {
        const pick = sub.querySelector('.sub-months :checked');
        if (!pick) {
          error.hidden = false;
          sub.querySelectorAll('.sub-months input').forEach((i) => i.setAttribute('aria-invalid', 'true'));
          sub.querySelector('.sub-months input').focus();
          return;
        }
        sub.querySelector('.sub-resume').textContent = pick.value;
      }
      const from = step;
      show(next, next === 'billing' ? trigger : null);
      if (next === 'cancelled') status.textContent = `Cancelled. You have Pro until ${ends}.`;
      else if (next === 'paused') status.textContent = `Paused. No charges until ${sub.querySelector('.sub-resume').textContent}.`;
      else status.textContent = say[`${from}>${next}`] || '';
    };

    const back = () => {
      if (!BACK[step]) return;
      const to = BACK[step];
      show(to, to === 'billing' ? trigger : null);
    };

    sub.addEventListener('click', (e) => {
      const b = e.target.closest('button');
      if (!b) return;
      if (b.classList.contains('sub-back')) back();
      else if (b.dataset.go) go(b.dataset.go);
    });
    sub.addEventListener('keydown', (e) => {
      if (e.key !== 'Escape' || !BACK[step]) return;
      e.preventDefault();
      back();   // Esc = Back. Never "cancel".
    });
    sub.addEventListener('change', (e) => {
      if (!e.target.closest('.sub-months')) return;
      error.hidden = true;
      sub.querySelectorAll('.sub-months input').forEach((i) => i.removeAttribute('aria-invalid'));
    });
  }
</script>
