<script>
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const fit = document.getElementById('fit');
  const wallet = document.getElementById('wallet');
  const live = document.getElementById('live');
  const readName = document.getElementById('readName');
  const readInfo = document.getElementById('readInfo');
  const putBackBtn = document.getElementById('putBack');

  // the library barcode, thin and thick by a fixed pattern
  document.querySelector('.bars').innerHTML = [2,1,3,1,1,4,2,1,3,2,1,1,4,1,2,3,1,2].map(w => `<i style="width:${w}px"></i>`).join('');

  let scale = 1;
  function resize() {
    scale = Math.min(1, (innerWidth - 16) / 440, (innerHeight - 16) / 660);
    fit.style.transform = `scale(${scale})`;
    fit.style.margin = `${(660 * scale - 660) / 2}px ${(440 * scale - 440) / 2}px`;
  }
  addEventListener('resize', resize); resize();

  // slots: 0 is the back of the stack, 4 the front. A card that comes back takes the front.
  const IN_POCKET = [108, 152, 196, 240, 284];   // card tops with five in the pocket
  const SQUEEZED  = [238, 254, 270, 286];        // four in the pocket, one out
  const OUT_Y = 8, OUT_ROT = -2.2;               // where a pulled card rests
  const PULL = 64;                               // lift that commits the pull
  const FLICK = 0.11;                            // px/ms: a quick flick counts at any distance

  const cards = [...wallet.querySelectorAll('.card')].map(el => ({
    el, id: el.dataset.id, name: el.dataset.name, info: el.dataset.info,
    sheen: el.querySelector('.sheen'),
    y: 0, vy: 0, x: 0, vx: 0, r: 0, vr: 0,       // state
    ty: 0, tx: 0, tr: 0, k: 300, c: 26,          // spring target
  }));
  let stack = [...cards];                        // back to front
  let out = null, drag = null;

  function layout() {
    const slots = out ? SQUEEZED : IN_POCKET;
    stack.forEach((c, i) => {
      c.el.style.zIndex = i + 1;
      if (c === drag?.card) return;
      c.ty = slots[i]; c.tx = 0; c.tr = 0;
      // cards settle front first, each softer than the one before it
      c.k = 300 - (stack.length - 1 - i) * 26; c.c = 26;
    });
    if (out && out !== drag?.card) { out.ty = OUT_Y; out.tx = 0; out.tr = OUT_ROT; out.k = 260; out.c = 21; }
    cards.forEach(c => c.el.setAttribute('aria-pressed', c === out));
    wake();
  }

  function say(text) { live.textContent = ''; setTimeout(() => live.textContent = text, 30); }
  function read() {
    readName.textContent = out ? out.name : 'Pull a card';
    readInfo.textContent = out ? out.info : 'Last one used goes to the front';
    putBackBtn.hidden = !out;
  }

  function pull(card) {
    if (out && out !== card) put(false);
    stack = stack.filter(c => c !== card);
    out = card; layout(); read();
    say(`${card.name} out. ${card.info}`);
  }
  function put(announce = true) {
    if (!out) return;
    const c = out; out = null;
    stack.push(c);                               // it goes back in at the front
    layout(); read();
    if (announce) say(`${c.name} back in the pocket`);
  }

  // one spring loop for all five; it sleeps when nothing moves
  let raf = 0, last = 0;
  function wake() { if (!raf) { last = performance.now(); raf = requestAnimationFrame(tick); } }
  function paint(c) {
    const lift = Math.max(0, 300 - c.y);
    c.el.style.transform = `translate3d(${c.x.toFixed(2)}px, ${c.y.toFixed(2)}px, 0) rotate(${c.r.toFixed(3)}deg)`;
    // the higher it is, the longer its shadow
    const s = lift / 300;
    c.el.style.boxShadow = `0 1px 1px oklch(0.15 0.03 60 / 0.35), 0 ${5 + s * 22}px ${10 + s * 34}px ${-3 - s * 8}px oklch(0.13 0.04 60 / ${0.4 + s * 0.2})`;
    c.sheen.style.transform = `translate3d(${(c.r * 26 + (c.y - 150) * 0.32 + c.x * 0.4).toFixed(1)}px, 0, 0)`;
  }
  function tick(now) {
    raf = 0;
    let dt = Math.max(0, Math.min(32, now - last)) / 1000; last = now;
    let moving = false;
    for (const c of cards) {
      if (reduce && c !== drag?.card) { c.y = c.ty; c.x = c.tx; c.r = c.tr; c.vy = c.vx = c.vr = 0; paint(c); continue; }
      let t = dt;
      while (t > 0) {
        const h = Math.min(t, 1 / 240); t -= h;
        c.vy += (c.k * (c.ty - c.y) - c.c * c.vy) * h; c.y += c.vy * h;
        c.vx += (c.k * (c.tx - c.x) - c.c * c.vx) * h; c.x += c.vx * h;
        c.vr += (c.k * (c.tr - c.r) - c.c * c.vr) * h; c.r += c.vr * h;
      }
      if (Math.abs(c.ty - c.y) + Math.abs(c.tx - c.x) + Math.abs(c.tr - c.r) + Math.abs(c.vy) + Math.abs(c.vx) + Math.abs(c.vr) > 0.02) moving = true;
      paint(c);
    }
    if (moving || drag) raf = requestAnimationFrame(tick);
  }

  // drag: the leather holds on, then lets go. Rotation comes from where you grabbed.
  function neighbours(card, amount) {
    const i = stack.indexOf(card);
    if (i < 0) return;
    const slots = out ? SQUEEZED : IN_POCKET;
    [stack[i - 1], stack[i + 1]].forEach(n => {
      if (n) n.ty = slots[stack.indexOf(n)] - amount * 0.16;  // the card next to it comes along a little
    });
  }
  function dragPose(card, dx, dy, side, home) {
    let y;
    if (card === out) {
      // out of the pocket: free downwards, damped upwards
      y = dy < 0 ? home + dy * 0.22 : home + dy;
    } else {
      // in the pocket: leather friction on the way up, a hard floor on the way down
      y = dy < 0 ? home + dy * 0.82 : home + Math.sqrt(dy) * 2.2;
    }
    // the further out, the more it can swing: the pocket walls hold the rest
    const free = Math.min(1, Math.max(0, (300 - y) / 260));
    card.ty = y;
    card.tx = dx * (0.08 + 0.42 * free);
    card.tr = (card === out ? OUT_ROT : 0) + dx * side * 0.05 * (0.2 + 0.8 * free) + side * Math.min(7, Math.abs(dy) * 0.035) * -1;
    if (card !== out) neighbours(card, Math.min(40, Math.max(0, -dy)));
  }
  let lastPointer = -1e9;
  cards.forEach(card => {
    const el = card.el;
    // assistive tech fires click without a pointer: the pointer path already handled its own
    el.addEventListener('click', e => { if (e.timeStamp - lastPointer > 500) card === out ? put() : pull(card); });
    el.addEventListener('pointerdown', e => {
      if (!e.isPrimary || e.button !== 0 || drag) return;
      el.setPointerCapture(e.pointerId);
      const box = el.getBoundingClientRect();
      drag = {
        card, id: e.pointerId, sx: e.clientX, sy: e.clientY, y0: card.y, x0: card.x,
        side: (e.clientX - (box.left + box.width / 2)) / (box.width / 2),
        lx: e.clientX, ly: e.clientY, lt: e.timeStamp, vy: 0, vx: 0, moved: 0,
      };
      card.k = 900; card.c = 52;                  // stiff under the hand
      wake();
    });
    el.addEventListener('pointermove', e => {
      if (!drag || drag.card !== card || e.pointerId !== drag.id) return;
      const dx = (e.clientX - drag.sx) / scale, dy = (e.clientY - drag.sy) / scale;
      const dt = Math.max(1, e.timeStamp - drag.lt);
      drag.vy = 0.6 * drag.vy + 0.4 * ((e.clientY - drag.ly) / scale / dt);
      drag.vx = 0.6 * drag.vx + 0.4 * ((e.clientX - drag.lx) / scale / dt);
      drag.lx = e.clientX; drag.ly = e.clientY; drag.lt = e.timeStamp;
      drag.moved = Math.max(drag.moved, Math.hypot(dx, dy));
      dragPose(card, dx, dy, drag.side, card === out ? OUT_Y : drag.y0);
      // the line under the wallet names the card as soon as it starts to lift
      if (card !== out) {
        readName.textContent = dy < -12 ? card.name : 'Pull a card';
        readInfo.textContent = dy < -12 ? card.info : 'Last one used goes to the front';
      }
      wake();
    });
    const end = e => {
      if (!drag || drag.card !== card || e.pointerId !== drag.id) return;
      const d = drag; drag = null;
      lastPointer = e.timeStamp;
      // velocity kept on release, but only if the hand was still moving
      const fresh = e.timeStamp - d.lt < 90;
      card.vy = fresh ? d.vy * 1000 : 0; card.vx = fresh ? d.vx * 1000 : 0;
      const dy = (e.clientY - d.sy) / scale;
      const flicked = fresh && Math.abs(d.vy) > FLICK;
      if (e.type === 'pointercancel') {
        layout();
      } else if (d.moved < 6) {                       // a plain tap toggles
        card === out ? put() : pull(card);
      } else if (card === out) {
        (dy > PULL || (flicked && d.vy > 0)) ? put() : layout();
      } else {
        (-dy > PULL || (flicked && d.vy < 0)) ? pull(card) : layout();
      }
      read();
    };
    el.addEventListener('pointerup', end);
    el.addEventListener('pointercancel', end);
    el.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); if (!e.repeat) card === out ? put() : pull(card); }
      if (e.key === 'ArrowUp' || e.key === 'ArrowDown') {
        e.preventDefault();
        const order = out ? [out, ...stack] : stack;
        const i = order.indexOf(card) + (e.key === 'ArrowUp' ? -1 : 1);
        order[(i + order.length) % order.length].el.focus();
      }
    });
  });
  document.addEventListener('keydown', e => { if (e.key === 'Escape' && out) { const c = out; put(); c.el.focus(); } });
  putBackBtn.addEventListener('click', () => { const c = out; put(); c?.el.focus(); });

  // start: everything at rest, nothing animates in
  layout();
  cards.forEach(c => { c.y = c.ty; c.x = c.tx; c.r = c.tr; c.vy = c.vx = c.vr = 0; paint(c); });

  // stills and the demo loop
  const params = new URLSearchParams(location.search);
  const still = params.get('still');
  const byId = id => cards.find(c => c.id === id);
  function freeze() { cancelAnimationFrame(raf); raf = 0; cards.forEach(c => { c.y = c.ty; c.x = c.tx; c.r = c.tr; c.vy = c.vx = c.vr = 0; paint(c); }); }
  if (still === 'pull') {
    // the transit card caught halfway out, leaning off the thumb
    // through the same pose function a real drag uses
    const c = byId('transit');
    layout(); freeze();
    dragPose(c, -30, -229, -0.8, IN_POCKET[stack.indexOf(c)]);
    freeze();
    readName.textContent = c.name; readInfo.textContent = c.info;
  }
  if (still === 'out') { pull(byId('debit')); freeze(); }
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    (async function loop() {
      for (const id of ['transit', 'debit', 'coffee']) {
        await wait(1200); pull(byId(id));
        await wait(1600); put();
      }
      loop();
    })();
  }
