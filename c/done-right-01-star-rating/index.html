<script id="component">
  for (const rating of document.querySelectorAll('.rating')) {
    const radios = rating.querySelectorAll('input[type="radio"]');
    const value = rating.querySelector('.rating-value');
    const error = rating.querySelector('.rating-error');

    const showValue = () => {
      const picked = rating.querySelector(':checked');
      value.textContent = picked ? `${picked.value} out of 5` : 'No rating yet';
    };
    const setError = (on) => {
      rating.toggleAttribute('data-invalid', on);
      error.hidden = !on;
      radios.forEach((r) => on ? r.setAttribute('aria-invalid', 'true') : r.removeAttribute('aria-invalid'));
    };

    rating.addEventListener('change', () => { showValue(); setError(false); });

    // Required, but with our own message instead of the browser bubble.
    rating.form?.addEventListener('submit', (event) => {
      if (rating.querySelector(':checked')) return;
      event.preventDefault();
      setError(true);
      radios[0].focus();
    });

    showValue();
  }
</script>

<!-- Demo page only: pretend to send the review. -->
