  });

    if (!drag || e.pointerId !== drag.id) return;
    const dy = (e.clientY - drag.cy) / k;
    // up and down is rubber-banded: the further you pull, the less it follows
    move(drag.x0 + (e.clientX - drag.cx) / k, drag.y0 + dy * 0.55 / (1 + Math.abs(dy) / 260));
    sample(e);
  });
  function sample(e) {
    drag.pts.push([e.timeStamp, drag.el.s.x, drag.el.s.y]);
    while (drag.pts.length > 2 && e.timeStamp - drag.pts[0][0] > 100) drag.pts.shift();
  }
  function move(x, y) {
    const s = drag.el.s;
    s.x = x; s.y = y; s.r = x * ROT * s.f;
    kick();
  }

  // velocity over the last 100 ms; a flick wins over distance
  function release(e) {
    if (!drag || (e && e.pointerId !== drag.id)) return;
    const { el, pts } = drag, s = el.s;
    const [t0, x0, y0] = pts[0], [t1, x1, y1] = pts[pts.length - 1];
    const dt = Math.max(16, t1 - t0);
    // held still before letting go: no flick
    const stale = (e ? e.timeStamp : performance.now()) - t1 > 60;
    const still0 = stale || (e && e.type === 'pointercancel');
    const vx = still0 ? 0 : (x1 - x0) / dt, vy = still0 ? 0 : (y1 - y0) / dt;
    el.classList.remove('grabbing');
    drag = null;
    let dir = 0;
    if (Math.abs(vx) > FLICK) dir = Math.sign(vx) === Math.sign(s.x) ? Math.sign(vx) : 0;
    else if (Math.abs(s.x) > COMMIT) dir = Math.sign(s.x);
    if (dir) return throwTop(dir, Math.max(Math.abs(vx), 0.9), vy);
    s.vx = vx * 1000; s.vy = vy * 1000; s.vr = s.vx * ROT * s.f;
    s.spring = true;
    kick();
  }
  deck.addEventListener('pointerup', release);
  deck.addEventListener('pointercancel', release);
  deck.addEventListener('lostpointercapture', release);

  function throwTop(dir, speed = 1.3, vy = -0.12) {
    const el = queue.shift();
    if (!el) return;
    const q = clamp(Math.abs(el.s.x) / (W * 0.6));
    rest += 1 - q;            // the cards below keep their place, then settle
    const s = el.s;
    s.spring = false;
    s.vx = dir * speed; s.vy = clamp(vy, -0.8, 0.8);
    s.vr = s.vx * ROT * s.f * 1.4;
    el.classList.add('thrown');
    el.style.zIndex = 50;
    history.push({ el, dir });
    if (reduce) { s.x = dir * 30; el.animate({ opacity: [1, 0] }, { duration: 200, fill: 'forwards' }); }
    flying.add(el);
    $('live').textContent = `${dir > 0 ? 'Saved' : 'Skipped'} ${el.rec.title}.`;
    refocus = ['saveBtn', 'skipBtn'].includes(document.activeElement.id);
    status();
    kick();
  }
  // /slide

  function tick(now) {
    const dt = Math.min(32, now - last); last = now;
    let busy = !!drag;
    if (reduce) rest = 0;
    if (Math.abs(rest) > 0.002) { rest *= Math.exp(-dt / 80); busy = true; } else rest = 0;
    const top = queue[0];
    if (top && top.s.spring) busy = spring(top.s, dt / 1000) || busy;
    flying.forEach(el => {
      const s = el.s;
      if (reduce) { if (now - (s.t0 ??= now) > 220) out(el); else busy = true; return; }
      s.x += s.vx * dt; s.y += s.vy * dt; s.r += s.vr * dt;
      s.vy += 0.0009 * dt;  // a little gravity
      if (Math.abs(s.x) > innerWidth / 2 + 420) out(el); else busy = true;
    });
    render();
    raf = busy && !frozen ? requestAnimationFrame(tick) : 0;
  }
  function kick() { if (!raf && !frozen) { last = performance.now(); raf = requestAnimationFrame(tick); } }
  // a spring with a little overshoot: stiffness 380, damping 24
  function spring(s, dt) {
    for (const k of ['x', 'y', 'r']) {
      const v = 'v' + k;
      s[v] += (-380 * s[k] - 24 * s[v]) * dt;
      s[k] += s[v] * dt;
    }
    if (reduce || (Math.abs(s.x) < 0.3 && Math.abs(s.vx) < 6 && Math.abs(s.y) < 0.3 && Math.abs(s.r) < 0.05)) {
      Object.assign(s, { x: 0, y: 0, r: 0, vx: 0, vy: 0, vr: 0, spring: false });
      return false;
    }
    return true;
  }
  function out(el) {
    flying.delete(el);
    el.getAnimations().forEach(a => a.cancel());  // before remove(), or the fade sticks
    el.remove();
    delete el.s.t0;
    if (!queue.length && !flying.size) showEnd();
  }

  // Undo: the card flies back from where it left
  function undo() {
    if (drag || !history.length) return;
    const { el } = history.pop();
    flying.delete(el);
    el.getAnimations().forEach(a => a.cancel());
    el.classList.remove('thrown');
    if (!el.isConnected) deck.append(el);
    const s = el.s;
    s.x = clamp(s.x, -innerWidth / 2 - 200, innerWidth / 2 + 200);
    s.r = clamp(s.r, -40, 40);
    s.vx = s.vy = s.vr = 0;
    if (reduce) { s.x = s.y = s.r = 0; el.animate({ opacity: [0, 1] }, { duration: 200 }); }
    s.spring = true;
    queue.unshift(el);
    status();
    if (!end.hidden && (end.contains(document.activeElement) || document.activeElement === document.body)) $('saveBtn').focus();
    hideEnd();
    $('live').textContent = `Back: ${el.rec.title}.`;
    kick();
  }

  function status() {
    const done = history.length, n = RECORDS.length;
    $('wheel').style.setProperty('--n', Math.min(done + 1, n));
    $('saveBtn').disabled = $('skipBtn').disabled = !queue.length;
    $('undoBtn').disabled = !done;
    const t = queue[0] ? ' ' + queue[0].rec.title : '';
    $('saveBtn').setAttribute('aria-label', 'Save' + t);
    $('skipBtn').setAttribute('aria-label', 'Skip' + t);
  }

  function showEnd() {
    const saved = history.filter(h => h.dir > 0).map(h => h.el.rec);
    const n = saved.length, k = Math.min(n, 5);
    $('fan').innerHTML = saved.slice(-k).map((rec, i) => {
      const o = i - (k - 1) / 2;
      return `<div class="mini" style="--i:${i};--x:${o * 70}px;--y:${Math.abs(o) * 10}px;--r:${o * 11}deg"><div class="clip"><div class="cover c-${rec.id}">${rec.art}</div></div></div>`;
    }).join('');
    $('endTitle').textContent = n ? `You saved ${n} record${n > 1 ? 's' : ''}` : 'Nothing saved this week';
    $('endText').textContent = `Skipped ${RECORDS.length - n}. Undo brings the last one back.`;
    end.hidden = false;
    end.offsetWidth;
    end.classList.add('in');
    $('live').textContent = $('endTitle').textContent + '.';
    // the Save/Skip button that had focus is now disabled: hand focus to Start over
    if (refocus) $('again').focus();
    refocus = false;
  }
  function hideEnd() { end.classList.remove('in'); end.hidden = true; }

  function reset() {
    queue.concat([...flying]).forEach(el => el.remove());
    flying.clear(); history = []; rest = 0;
    queue = RECORDS.map(make);
    queue.slice().reverse().forEach(el => deck.append(el));
    lucide.createIcons({ attrs: { 'stroke-width': 2.4 } });
    hideEnd(); status(); render();
  }

  $('saveBtn').addEventListener('click', () => !drag && throwTop(1));
  $('skipBtn').addEventListener('click', () => !drag && throwTop(-1));
  $('undoBtn').addEventListener('click', undo);
  $('again').addEventListener('click', () => {
    reset(); $('saveBtn').focus();
    $('live').textContent = `${RECORDS.length} records. ${RECORDS[0].title} first.`;
  });
  addEventListener('keydown', e => {
    if (e.target.closest('button') && (e.key === 'Enter' || e.key === ' ')) return;
    if (e.key === 'ArrowRight' && !drag) { e.preventDefault(); throwTop(1); }
    else if (e.key === 'ArrowLeft' && !drag) { e.preventDefault(); throwTop(-1); }
    else if (e.key === 'Backspace' || e.key === 'z' || e.key === 'Z') { e.preventDefault(); undo(); }
  });

  // fit: small or landscape screens scale the whole app down instead of cutting it off
  const app = document.querySelector('.app');
  function fit() {
    app.style.zoom = 1;
    k = Math.min(1, (innerWidth - 24) / 330, (innerHeight - 24) / app.offsetHeight);
    app.style.zoom = k;
    W = deck.offsetWidth; COMMIT = W * 0.4;
  }
  addEventListener('resize', fit);
  fit();
  reset();

  // carousel states and the demo loop
  const params = new URLSearchParams(location.search);
  const still = params.get('still');
  function skipInstant(dir) { throwTop(dir); out([...flying].pop()); rest = 0; }
  if (still) {
    frozen = true;
    if (still === 'drag') {
      Object.assign(queue[0].s, { x: COMMIT * 0.85, y: 6, f: 1, r: COMMIT * 0.85 * ROT });
    } else if (still === 'throw') {
      skipInstant(1);
      throwTop(-1);
      Object.assign([...flying][0].s, { x: -250, y: 34, r: -27 });
      rest = 0.12;
    } else if (still === 'done') {
      [1, -1, 1, -1, 1, -1].forEach(skipInstant);
    }
    render();
  }
  if (params.has('demo')) {
    const wait = ms => new Promise(r => setTimeout(r, ms));
    const ease = t => t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2;
    // a ghost finger: drags the top card, then lets go with a velocity
    function ghost(toX, toY, ms, f) {
      const el = queue[0];
      el.s.f = f; el.s.spring = false;
      drag = { el, id: -1, pts: [] };
      const t0 = performance.now();
      return new Promise(done => (function step(now) {
        const t = clamp((now - t0) / ms), k = ease(t);
        move(toX * k, toY * k);
        drag.pts.push([now, el.s.x, el.s.y]);
        while (drag.pts.length > 2 && now - drag.pts[0][0] > 100) drag.pts.shift();
        if (t < 1) requestAnimationFrame(step); else { release(); done(); }
      })(t0));
    }
    (async function loop() {
      await wait(900);  await ghost(150, -10, 900, 1);
      await wait(1000); await ghost(-70, 6, 700, -0.7);
      await wait(900);  await ghost(-80, 4, 140, 1);
      await wait(1000); throwTop(1);
      await wait(1000); undo();
      await wait(1300); throwTop(1);
      await wait(700);  throwTop(-1);
      await wait(700);  throwTop(1);
      await wait(700);  throwTop(-1);
      await wait(3200); reset(); loop();
    })();
  }
</script>
